79  	public IBaseResource setup() {
80  		encounter = createResource(Encounter.class);
81  		patient = getLastResource(Patient.class);
82  		if (patient == null) {
83  			// No patient was found, create one.
84  			patient = createResource(Patient.class);
85  		}
86  		encounter.setSubject(ParserUtils.toReference(patient, encounter, "subject", "patient"));
87  		account = null;
88  		location = null;
89  		return encounter;

92  	/**
93  	 * Set the patient class.
94  	 * Maps the class to the preferred terminology from patientClass if a value is found
95  	 * that maps, otherwise sets to the first coding in patientClass.
96  	 * Also adds the class to the Encounter.type. 
97  	 * @param patientClass	The patient class
98  	 */
99  	@ComesFrom(path = "Encounter.class", field = 2, table = "0004", map = "EncounterClass", comment = "Patient Class",
100 			   also = "Encounter.type")
101 	public void setPatientClass(CodeableConcept patientClass) {
102 		boolean notMapped = false;
103 		// Set to mapped value when present.
104 		// Encounter.class is of coding type with a preferred coding system, so it gets mapped
105 		encounter.setClass_(mapPatientClass(patientClass));
106 		
107 		// Otherwise set to first coding value.  It's typical that there's only one,
108 		// so we won't worry about the rest of them, unlike the mapping case above.
109 		if (!encounter.hasClass_() && patientClass.hasCoding()) {
110 			notMapped = true;
111 			encounter.setClass_(patientClass.getCodingFirstRep());
112 		}
113 		
114 		if (encounter.hasClass_()) {
115 			// We don't want to lose the original values, so we store them as the codings
116 			// that apply to coding.code
117 			CodeType code = encounter.getClass_().getCodeElement();
118 			for (Coding coding: patientClass.getCoding()) {
119 				if (notMapped) {
120 					// If it wasn't mapped, we stored the first one, so we skip it.
121 					notMapped = false;
122 					continue;
123 				}
124 				code.addExtension(PathUtils.FHIR_EXT_PREFIX + "/iso21090-SC-coding", coding);
125 			}
126 		}
127 		encounter.addType(patientClass);
128 	}
129 
130 	private Coding mapPatientClass(CodeableConcept patientClass) {
131 		for (Coding coding: patientClass.getCoding()) {
132 			if (!coding.hasCode()) {
133 				continue;
134 			}
135 			
136 			if (coding.hasSystem() && !coding.getSystem().endsWith("0004")) {
137 				continue;
138 			}
139 
140 			switch (coding.getCode()) {
141 			case "E":	//	Emergency
142 				return new Coding(Systems.V3_ACT_ENCOUNTER_CODE, "EMER", "Emergency");
143 			case "I":	//	Inpatient	
144 				return new Coding(Systems.V3_ACT_ENCOUNTER_CODE, "IMP", "inpatient");
145 			case "O":	//	Outpatient	
146 				return new Coding(Systems.V3_ACT_ENCOUNTER_CODE, "AMB", "ambulatory");
147 			default:
148 				break;
149 			}
150 		}
151 		return null;
152 	}

159 	public void setAssignedPatientLocation(Location assignedPatientLocation) {
160 		location = addResource(assignedPatientLocation);
161 		EncounterLocationComponent loc = encounter.addLocation()
162 				.setLocation(ParserUtils.toReference(assignedPatientLocation, encounter, "location"));
163 		if (encounter.hasStatus()) {
164 			if (EncounterStatus.PLANNED.equals(encounter.getStatus())) {
165 				loc.setStatus(EncounterLocationStatus.PLANNED);
166 			} else {
167 				loc.setStatus(EncounterLocationStatus.ACTIVE);
168 			}
169 		}
170 	}

362 	private LocationStatus mapBedStatus(Coding bedStatus) {
363 		// TODO Auto-generated method stub
364 		if (!bedStatus.hasCode()) {
365 			return null;
366 		}
367 		if (bedStatus.hasSystem() && bedStatus.getSystem().endsWith("0116")) {
368 			switch (bedStatus.getCode()) {
369 			case "O", "U", "I":
370 				return LocationStatus.ACTIVE;
371 			case "H", "K":
372 				return LocationStatus.SUSPENDED;
373 			case "C":
374 				return LocationStatus.INACTIVE;
375 			default:
376 				return null;
377 			}
378 		}
379 		return null;
380 	}
