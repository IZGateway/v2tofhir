32  /**
33   * Parse an ERR segment into an OperationOutcome resource.
34   * 
35   * If no OperationOutcome already exists for this message, a new one is created.
36   * One OperationOutcome.issue is created for each ERR segment.
37   * 
38   * OperationOutcome.issue.location is set from ERR-2
39   * OperationOutcome.issue.code is set from ERR-3
40   * OperationOutcome.issue.severity is set from ERR-4
41   * OperationOutcome.issue.detail is set from ERR-3 and ERR-5
42   * OperationOutcome.issue.diagnostics is set from ERR-7 and ERR-8 if present.
43   *  
44   */
45  @Produces(segment="ERR", resource = OperationOutcome.class, extra = MessageHeader.class)
46  @Slf4j
47  public class ERRParser extends AbstractSegmentParser {
48  	private static final String SUCCESS = "success";
49  	private OperationOutcomeIssueComponent issue;
50  	
51  	static {
52  		log.debug("{} loaded", ERRParser.class.getName());
53  	}
54  	private static final LinkedHashMap<String, String[]> errorCodeMap = new LinkedHashMap<>();
55  	private static final String NOT_SUPPORTED = "not-supported";
56  	private static final String[][] errorCodeMapping = {
57  			{ SUCCESS, "0", "Message accepted", "Success. Optional, as the AA conveys success. Used for systems that must always return a status code." },
58  			{ "structure", "100", "Segment sequence error", "Error: The message segments were not in the proper order, or required segments are missing." },
59  			{ "required", "101", "Required field missing", "Error: A required field is missing from a segment" },
60  			{ "value", "102", "Data type error", "Error: The field contained data of the wrong data type, e.g. an NM field contained" },
61  			{ "code-invalid", "103", "Table value not found", "Error: A field of data type ID or IS was compared against the corresponding table, and no match was found." },
62  			{ NOT_SUPPORTED, "200", "Unsupported message type", "Rejection: The Message Type is not supported." },
63  			{ NOT_SUPPORTED, "201", "Unsupported event code", "Rejection: The Event Code is not supported." },
64  			{ NOT_SUPPORTED, "202", "Unsupported processing id", "Rejection: The Processing ID is not supported." },
65  			{ NOT_SUPPORTED, "203", "Unsupported version id", "Rejection:  The Version ID is not supported." },
66  			{ "not-found", "204", "Unknown key identifier", "Rejection: The ID of the patient, order, etc., was not found. Used for transactions other than additions, e.g. transfer of a non-existent patient." },
67  			{ "conflict", "205", "Duplicate key identifier", "Rejection: The ID of the patient, order, etc., already exists. Used in response to addition transactions (Admit, New Order, etc.)." },
68  			{ "lock-error", "206", "Application record locked", "Rejection: The transaction could not be performed at the application storage level, e.g., database locked." },
69  			{ "exception", "207", "Application internal error", "Rejection: A catchall for internal errors not explicitly covered by other codes." }
70  	};
71  	private static final List<FieldHandler> fieldHandlers = new ArrayList<>();
72  	static {
73  		for (String[] mapping : errorCodeMapping) {
74  			errorCodeMap.put(mapping[1], mapping);
75  		}
76  	}

78  	/**
79  	 * Construct the ERR Segment parser for the given MessageParser
80  	 * 
81  	 * @param messageParser	The messageParser that is using this segment parser.
82  	 */
83  	public ERRParser(MessageParser messageParser) {
84  		super(messageParser, "ERR");
85  		if (fieldHandlers.isEmpty()) {
86  			FieldHandler.initFieldHandlers(this, fieldHandlers);
87  		}
88  	}
89  	
90  	@Override
91  	public List<FieldHandler> getFieldHandlers() {
92  		return fieldHandlers;
93  	}
94  	
95  	@Override 
96  	public IBaseResource setup() {
97  		OperationOutcome oo;
98  		// Create a new OperationOutcome.issue for each ERR resource
99  		oo = getFirstResource(OperationOutcome.class);
100 		if (oo == null) {
101 			oo = createResource(OperationOutcome.class);
102 			MessageHeader mh = getFirstResource(MessageHeader.class);
103 			if (mh == null) {
104 				mh = createResource(MessageHeader.class);
105 			}
106 			// If MessageHeader or OperationOutcome has to be created, 
107 			// link them in MessageHeader.response.details
108 			// Normally this is already done in MSAParser, but if processing
109 			// an incomplete message (e.g., ERR segment but w/o MSA), this ensures 
110 			// that the parser has the expected objects to work with.
111 			mh.getResponse().setDetails(ParserUtils.toReference(oo, mh, "details"));
112 		}
113 		issue = oo.addIssue();
114 		return oo;
115 	}

126 	/**
127 	 * Set the details on an operation outcome
128 	 * @param errorCode	The error code
129 	 */
130 	@ComesFrom(path="OperationOutcome.issue.details", table="0357", field=3, comment="Also sets issue.code")
131 	@ComesFrom(path="OperationOutcome.issue.details", table="0533", field=5)
132 	public void setDetails(CodeableConcept errorCode) {
133 		if (errorCode == null) {
134 			return;
135 		}
136 		for (Coding coding: errorCode.getCoding()) {
137 			issue.getDetails().addCoding(coding);
138 		}
139 		for (Coding c: errorCode.getCoding()) {
140 			// If from table 0357
141 			if (!TerminologyMapperFactory.get().v2TableUri("0357").equals(c.getSystem())) {  // Check the system.
142 				continue;
143 			}
144 			String[] map = errorCodeMap.get(c.getCode());
145 			if (map == null) {
146 				issue.setCode(IssueType.PROCESSING);
147 				continue;
148 			}
149 			try {
150 				if (SUCCESS.equals(map[0])) {
151 					// Some versions of FHIR support "success", others don't.
152 					issue.setCode(IssueType.INFORMATIONAL);
153 				} else {
154 					IssueType type = IssueType.fromCode(map[0]); 
155 					issue.setCode(type);
156 				}
157 			} catch (FHIRException fex) {
158 				getParser().warn("Unexpected FHIRException: {}", fex.getMessage(), fex);
159 			}
160 		}
161 	}
162 	
163 	/**
164 	 * Set the severity
165 	 * @param severity	Severity
166 	 */
167 	@ComesFrom(path="OperationOutcome.issue.severity", field = 4, table = "0516", map = "ErrorSeverity")
168 	public void setSeverity(CodeableConcept severity) {
169 		for (Coding coding: severity.getCoding()) {
170 			issue.getDetails().addCoding(coding);
171 		}
172 		for (Coding sev : severity.getCoding()) {
173 			if (sev == null || !sev.hasCode()) {
174 				continue;
175 			}
176 			switch (sev.getCode().toUpperCase()) {
177 			// F is not really a code in table 0516, but if it shows up, treat it as if it meant fatal.
178 			case "F":	issue.setSeverity(IssueSeverity.FATAL); break;
179 			// These are the legit codes.
180 			case "I":	issue.setSeverity(IssueSeverity.INFORMATION); break;
181 			case "W":	issue.setSeverity(IssueSeverity.WARNING); break;
182 			case "E":	// Fall through
183 			// Everything else maps to error
184 			default: 	issue.setSeverity(IssueSeverity.ERROR); break;
185 			}
186 		}
187 	}

189 	/**
190 	 * Set the diagnostic messages in MSA-7 and MSA-8 in the OperationOutcome
191 	 * @param diagnostics	The diagnostics to report
192 	 */
193 	@ComesFrom(path="OperationOutcome.issue.diagnostics", field=7, comment="Diagnostics")
194 	@ComesFrom(path="OperationOutcome.issue.diagnostics", field=8, comment="User Message")
195 	public void setDiagnostics(StringType diagnostics) {
196 		if (issue.hasDiagnostics()) {
197 			issue.setDiagnostics(issue.getDiagnostics() + "\n" + diagnostics);
198 		} else {
199 			issue.setDiagnosticsElement(diagnostics);
200 		}
201 	}
