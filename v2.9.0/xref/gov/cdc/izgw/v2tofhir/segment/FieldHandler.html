32  /**
33   * A handler for processing field conversions in a Segment of a V2 message.
34   *
35   * @author Audacious Inquiry
36   */
37  @Slf4j
38  public class FieldHandler implements Comparable<FieldHandler> {
39  	/** Set VERIFY_METHODS to true to check method initialization */
40  	private static final boolean VERIFY_METHODS = false;
41  	/** Set GET_PROPERTY to true to get the property during initialization */
42  	private static final boolean GET_PROPERTY = false;
43  	private final ComesFrom from;
44  	private final Produces produces;
45  	private final Method method;
46  	@SuppressWarnings("unused")
47  	private final Property prop;
48  	private final Class<? extends IBase> theClass;
49  	private final Class<? extends Type> theType;
50  	/**
51  	 * Create a FieldHandler from a ComesFrom annotation on a parser method.
52  	 * @param method	The method to which the ComesFrom annotation applies.
53  	 * @param from		The annotation
54  	 * @param p	The class for the parser
55  	 */
56  	public FieldHandler(Method method, ComesFrom from, Processor<Message, ? extends Structure> p) {
57  		this.method = method;
58  		this.from = from;
59  		this.produces = p.getProduces();
60  		
61  		if (GET_PROPERTY) {
62  			/**
63  			 * This still has a few small problems to fix on more complex
64  			 * paths using [] and resolve() features.
65  			 */
66  			try {
67  				prop = PathUtils.getProperty(
68  					getResourceClass(produces, from.path(), p), 
69  					from.path()
70  				);
71  			} catch (IllegalArgumentException e) {
72  				log.error(e.getMessage(), e);
73  				throw new ServiceConfigurationError(e.getMessage(), e);
74  			}
75  		} else {
76  			prop = null;
77  		}
78  		
79  		Class<?>[] params = method.getParameterTypes();
80  
81  		if (IBase.class.isAssignableFrom(params[0])) {
82  			@SuppressWarnings("unchecked")
83  			Class<? extends IBase> param = (Class<? extends IBase>) params[0];
84  			theClass = param;
85  			theType = null;
86  		} else if (Type.class.isAssignableFrom(params[0])) {
87  			@SuppressWarnings("unchecked")
88  			// Direct handling of type in Parser (e.g., for OBX Segments where Type of OBX-5 is not known
89  			// until OBX-2 is parsed.
90  			Class<? extends Type> param = (Class<? extends Type>) params[0];
91  			theClass = null;
92  			theType = param;
93  		} else {
94  			throw new ServiceConfigurationError("Method does not accept a FHIR type: " + method);
95  		}
96  		
97  		// Verify setValue works at initialization on a dummy
98  		if (VERIFY_METHODS) {
99  			IBase dummyObject = null;
100 			try {
101 				dummyObject = theClass.getDeclaredConstructor().newInstance();
102 				// Test the method call
103 				setValue(p, dummyObject);
104 			} catch (InstantiationException | IllegalAccessException | IllegalArgumentException
105 					| InvocationTargetException | NoSuchMethodException | SecurityException e) {
106 				throw new ServiceConfigurationError("Cannot create " + theClass.getSimpleName() + " for " + method);
107 			}
108 		}
109 	}

111 	private Class<? extends IBaseResource> getResourceClass(Produces produces, String path, Processor<Message, ? extends Structure> p) {
112 		Class<? extends IBaseResource> resourceClass = produces.resource();
113 		String resourceName = StringUtils.substringBefore(path, ".");
114 		// remove any array index
115 		resourceName = resourceName.contains("[") ? StringUtils.substringBefore(resourceName, "[") : resourceName; 
116 		if (!resourceClass.getSimpleName().equals(resourceName)) {
117 			for (Class<? extends IBaseResource> extraClass: produces.extra()) {
118 				if (extraClass.getSimpleName().equals(resourceName)) {
119 					resourceClass = extraClass;
120 					break;
121 				}
122 			}
123 			if (resourceClass == produces.resource()) {
124 				throw new IllegalArgumentException(
125 					"Resource " + resourceName + " is not in @Produces.resource or @Produces.extra for " + p.getClass().getSimpleName()
126 				);
127 			}
128 		}
129 		return resourceClass;
130 	}

132 	/**
133 	 * Convert fields within a segment to components in a resource
134 	 * 
135 	 * @param p	The structure parser to do the handling for
136 	 * @param segment	The segment to process
137 	 * @param r	The resource to update
138 	 */
139 	public void handle(Processor<Message, ? extends Structure> p, Segment segment, IBaseResource r) {
140 		if (!from.fixed().isEmpty()) {
141 			setFixedValue(p);
142 		} else if (from.field() != 0) {
143 			setFromFieldAndComponent(p, segment);
144 		} else {
145 			// we have neither a fixed value nor a field and component,
146 			// this is a configuration error.
147 			log.error("ComesFrom missing fixed or field value");
148 		}
149 	}
150 
151 	private void setFixedValue(Processor<Message, ? extends Structure> p) {
152 		ST st = new ST(null);
153 		try {
154 			st.setValue(from.fixed());
155 			IBase value = DatatypeConverter.convert(theClass, st, from.table()); 
156 			setValue(p, value);
157 		} catch (DataTypeException e) {
158 			// This will never happen.
159 		}
160 	}
161 
162 	private void setFromFieldAndComponent(Processor<Message, ? extends Structure> p, Segment segment) {
163 		Type[] f = ParserUtils.getFields(segment, from.field());
164 		if (f.length == 0) {
165 			return;
166 		}
167 		for (Type t: f) {
168 			t = getActualType(t);
169 			if (t == null) {
170 				continue;
171 			}
172 			if (theClass != null) {
173 				IBase value = DatatypeConverter.convert(theClass, t, from.table());
174 				if (value != null) {
175 					setValue(p, value);
176 				}
177 			} else if (theType != null) {
178 				// Let the parser handle the type conversion
179 				setValue(p, t);
180 			}
181 		}
182 	}
183 
184 	private Type getActualType(Type t) {
185 		t = DatatypeConverter.adjustIfVaries(t, from.type());
186 		if (t instanceof Composite comp && from.component() != 0) {
187 			// Check for component
188 			Type[] t2 = comp.getComponents();
189 			if (from.component() > t2.length) {
190 				return null;
191 			} 
192 			return t2[from.component()-1];
193 		} 
194 		if (from.component() < 2) {
195 			return t;
196 		}
197 		// A component > 1 does not exist in a primitive
198 		return null;
199 	}

201 	private IBase setValue(Processor<Message, ? extends Structure> p, IBase object) {
202 		try {
203 			method.invoke(p, object);
204 		} catch (IllegalAccessException | IllegalArgumentException | InvocationTargetException e) {
205 			// This is checked during initialization, so failure happens early if it is going to
206 			// happen at all, so it's OK to throw an error here.
207 			log.error("Cannot invoke {}: {}", method, e.getMessage(), e);
208 			throw new IllegalStateException("Cannot invoke " + method, e);
209 		}
210 		return object;
211 	}

213 	private Type setValue(Processor<Message, ? extends Structure> p, Type object) {
214 		try {
215 			method.invoke(p, object);
216 		} catch (IllegalAccessException | IllegalArgumentException e) {
217 			// This is checked during initialization, so failure happens early if it is going to
218 			// happen at all, so it's OK to throw an error here.
219 			log.error("Cannot invoke {}: {}", method, e.getMessage(), e);
220 			throw new IllegalStateException("Cannot invoke " + method, e);
221 		} catch (InvocationTargetException ex) {
222 			if (ex.getTargetException() instanceof RuntimeException rte) {
223 				throw rte;
224 			}
225 			if (ex.getTargetException() instanceof Error err) {
226 				throw new IllegalStateException("Error executing " + method, err.getCause());
227 			}
228 			throw new IllegalStateException("Exception executing " + method, ex.getCause());
229 		}
230 		return object;
231 	}

234 		return toString(from) + method.toString() ;

237 	private String toString(ComesFrom from) {
238 		StringBuilder b = new StringBuilder();
239 		b.append("\n@ComesFrom(path = \"").append(from.path()).append("\"");
240 		appendField(b, "field", from.field());
241 		appendField(b, "component", from.component());
242 		appendField(b, "fixed", from.fixed());
243 		appendField(b, "table", from.table());
244 		appendField(b, "map", from.map());
245 		appendField(b, "comment", from.comment());
246 		appendField(b, "also", from.also());
247 		appendField(b, "priority", from.priority());
248 		b.append(")\n");
249 		return b.toString();
250 	}

252 	private void appendField(StringBuilder b, String name, String[] also) {
253 		if (also.length == 0) return;
254 		b.append(", ").append(name).append(" = { ");
255 		for (String v: also) {
256 			b.append("\"").append(v).append("\", ");
257 		}
258 		b.setCharAt(b.length()-1, '}');
259 		b.setCharAt(b.length()-2, ' ');
260 	}
261 
262 	private void appendField(StringBuilder b, String name, int value) {
263 		if (value <= 0) return;
264 		b.append(", ").append(name).append(" = ").append(value);
265 	}

268 		if (StringUtils.isBlank(string)) {
269 			return;
270 		}
271 		b.append(", ").append(name).append(" = \"").append(string).append("\"");
272 	}

274 	/**
275 	 * Compare two field handlers to sort them.
276 	 * @param fh1	The first field handler to compare
277 	 * @param fh2	The second field handler to compare
278 	 * @return &lt; 0 if fh1 &lt; fh2, 0 if equal, &gt; 0 if fh1 &gt; fh2
279 	 */
280 	public static int compare(FieldHandler fh1, FieldHandler fh2) {
281 		if (fh1 == fh2)
282 			return 0;
283 		if (fh1 == null) 
284 			return -1;
285 		if (fh2 == null)
286 			return 1;
287 		// Priority comparison is reversed
288 		int comp = -Integer.compare(fh1.from.priority(), fh2.from.priority());
289 		if (comp != 0) 
290 			return comp;
291 		comp = Integer.compare(fh1.from.field(), fh2.from.field());
292 		if (comp != 0) 
293 			return comp;
294 		comp = Integer.compare(fh1.from.component(), fh2.from.component());
295 		if (comp != 0) 
296 			return comp;
297 		comp = fh1.toString().compareTo(fh2.toString());
298 		if (comp != 0) 
299 			return comp;
300 		return fh1.method.toString().compareTo(fh2.toString());
301 	}

303 	@Override
304 	public int compareTo(FieldHandler that) {
305 		return compare(this, that);

309 	public boolean equals(Object obj) {
310 		if (obj instanceof FieldHandler that) {
311 			return compareTo(that) == 0;
312 		}
313 		return false;

321 	/**
322 	 * initFieldHandlers must be called by a parser before calling parse(segment, parser)
323 	 * 
324 	 * @see AbstractStructureParser#getFieldHandlers()
325 	 * @param p	The structure parser to compute the field handlers for.
326 	 * @param fieldHandlers	The list to update with all of the field handlers 
327 	 */
328 	public static void initFieldHandlers(Processor<Message, ? extends Structure> p, List<FieldHandler> fieldHandlers) {
329 		// Synchronize on the list in case two callers are trying to to initialize it at the 
330 		// same time.  This avoids one thread from trying to use fieldHandlers while another 
331 		// potentially overwrites it.
332 		synchronized (fieldHandlers) { // NOSONAR This use of a parameter for synchronization is OK
333 			// We got the lock, recheck the entry condition.
334 			if (!fieldHandlers.isEmpty()) {
335 				// Another thread initialized the list.
336 				return;
337 			}
338 			/*
339 			 * Go through and find all methods with a ComesFrom annotation.
340 			 */
341 			for (Method method: p.getClass().getMethods()) {
342 				for (ComesFrom from: method.getAnnotationsByType(ComesFrom.class)) {
343 					fieldHandlers.add(new FieldHandler(method, from, p));
344 				}
345 			}
346 			Collections.sort(fieldHandlers);
347 		}
348 	}

350 	/**
351 	 * Get all public methods with a ComesFrom annotation.
352 	 * @param parserClass The parser to get the annotations from.
353 	 * @return all methods with a ComesFrom annotation in processing order.
354 	 */
355 	public static List<ComesFrom> getComesFrom(
356 		Class<? extends AbstractStructureParser> parserClass
357 	) {
358 		List<ComesFrom> a = new ArrayList<>();
359 		for (Method method: parserClass.getMethods()) {
360 			a.addAll(Arrays.asList(method.getAnnotationsByType(ComesFrom.class)));
361 		}
362 		Collections.sort(a, FieldHandler::compareComesFrom);
363 		return a;
364 	}

366 	/**
367 	 * Order ComesFrom annotations into processing order
368 	 * @param a The first annotation
369 	 * @param b The second annotation
370 	 * @return The annotations in processing order
371 	 */
372 	public static int compareComesFrom(ComesFrom a, ComesFrom b) {
373 		if (Objects.equals(a, b)) {
374 			return 0;
375 		}
376 		int comp = -Integer.compare(a.priority(), b.priority());
377 		if (comp != 0) return comp;
378 		comp = Integer.compare(a.field(), b.field());
379 		if (comp != 0) return comp;
380 		comp = Integer.compare(a.component(), b.component());
381 		if (comp != 0) return comp;
382 		return a.path().compareTo(b.path());
383 	}
