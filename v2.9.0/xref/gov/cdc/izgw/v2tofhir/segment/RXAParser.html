34  /**
35   * RXAParser parses any RXA segments in a message to an Immunization if the message is a VXU or an 
36   * response to an immunization query.
37   * 
38   * @author Audacious Inquiry
39   *
40   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-rxa-to-immunization.html">V2 to FHIR: RXA to Immunization</a>
41   */
42  @Produces(segment = "RXA", resource=Immunization.class, 
43  	extra = { Practitioner.class, Organization.class }
44  )
45  public class RXAParser extends AbstractSegmentParser {
46  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
47  	private IzDetail izDetail;
48  	private ImmunizationRecommendationRecommendationComponent recommendation;
49  	
50  	/**
51  	 * Create an RXA Parser for the specified MessageParser
52  	 * @param p	The message parser.
53  	 */
54  	public RXAParser(MessageParser p) {
55  		super(p, "RXA");
56  		if (fieldHandlers.isEmpty()) {
57  			FieldHandler.initFieldHandlers(this, fieldHandlers);
58  		}
59  	}
60  
61  	@Override
62  	protected List<FieldHandler> getFieldHandlers() {
63  		return fieldHandlers;
64  	}
65  
66  	@Override
67  	public IBaseResource setup() {
68  		izDetail = IzDetail.get(getMessageParser());
69  		izDetail.initializeResources(false, getSegment());
70  		if (izDetail.hasImmunization()) {
71  			return izDetail.getImmunization();
72  		} else if (izDetail.hasRecommendation()) {
73  			recommendation = izDetail.getImmunizationRecommendation().addRecommendation();
74  			return izDetail.getImmunizationRecommendation();
75  		}
76  		// Until we support RXA more generally, we don't know what to do with this segment 
77  		return null;
78  	}

83  	/**
84  	 * Set the administration date time.
85  	 * @param administrationDateTime the administration date time	
86  	 */
87  	@ComesFrom(path = "Immunization.occurrenceDateTime", field = 3, comment = "Date/Time Start of Administration")
88  	public void setAdministrationDateTime(DateTimeType administrationDateTime) {
89  		if (izDetail.hasImmunization()) {
90  			izDetail.getImmunization().setOccurrence(administrationDateTime);
91  		}
92  		// On the forecast path RXA-3 is only the time the forecast was generated.  It is not written
93  		// to a dateCriterion: that would create one with no code, and R4 requires
94  		// dateCriterion.code 1..1.
95  	}

98  	*/
99  	/**
100 	 * Set the vaccine code
101 	 * @param vaccineCode	the vaccine code
102 	 */
103 	@ComesFrom(path = "Immunization.vaccineCode", field = 5, table = "0292", comment = "Administered Code")
104 	public void setVaccineCode(CodeableConcept vaccineCode) {
105 		// CDC says RXA-5 cannot repeat, but STC example shows a repeat, handle the
106 		// repeat as additional codes.
107 		if (izDetail.hasImmunization()) {
108 			if (izDetail.getImmunization().hasVaccineCode()) {
109 				for (Coding coding: vaccineCode.getCoding()) {
110 					izDetail.getImmunization().getVaccineCode().addCoding(coding);
111 				}
112 			} else {
113 				izDetail.getImmunization().setVaccineCode(vaccineCode);
114 			}
115 		}
116 		// On the forecast path RXA-5 is the 998^no vaccine administered^CVX placeholder, so it is
117 		// not a recommendation.vaccineCode.  That comes from the OBX-3 = 30956-7 vaccine type.
118 	}

245 	/**
246 	 * Set the manufacturing organization from the MVX or Table 0227 code
247 	 * @param manufacturer	The manufacturing organization
248 	 */
249 	@ComesFrom(path = "Immunization.manufacturer.Organization", field = 17, table = "0227", comment = "Substance Manufacturer Name")
250 	public void setManufacturer(CodeableConcept manufacturer) {
251 		if (izDetail.hasImmunization()) {
252 			Organization mOrg = new Organization();
253 			// Convert a code to an Organization name and identifier
254 			for (Coding coding: manufacturer.getCoding()) {
255 				String system = coding.getSystem();
256 				if (Systems.MVX.equals(system) || TerminologyMapperFactory.get().v2TableUri("0227").equals(system)) {
257 					if (coding.hasDisplay()) {
258 						mOrg.setName(coding.getDisplay());
259 					}
260 					mOrg.addIdentifier(new Identifier().setSystem(system).setValue(coding.getCode()));
261 				}
262 			}
263 			if (mOrg.hasName() || mOrg.hasIdentifier()) {
264 				addResource(mOrg);
265 			}
266 			izDetail.getImmunization().setManufacturer(ParserUtils.toReference(mOrg, izDetail.getImmunization(), "manufacturer"));
267 		}
268 	}

302 	20	RXA-20	Completion Status	ID	0	1	IF RXA-21 NOT EQUALS "D"			Immunization.status		Immunization.code	1	1		CompletionStatus		
303 	20	RXA-20	Completion Status	ID	0	1	IF NOT VALUED AND RXA-21 NOT EQUALS "D"			Immunization.status		Immunization.code	1	1			"completed"	
304 	21	RXA-21	Action Code – RXA	ID	0	1	IF RXA-21 EQUALS "D"			Immunization.status		Immunization.code	1	1			"entered-in-error"	
305 	*/
306 	/**
307 	 * Set the completion status
308 	 * @param completionStatus the completion status
309 	 */
310 	@ComesFrom(path = "Immunization.status", field = 20, table = "0322", map = "ImmunizationStatus", comment = "Completion Status")
311 	@ComesFrom(path = "Immunization.status", field = 21, table = "0323", map = "ImmunizationStatus", comment = "Action Code – RXA")
312 	public void setCompletionStatus(Coding completionStatus) {
313 		if (izDetail.hasImmunization()) {
314 			if (!completionStatus.hasCode()) {
315 				return;
316 			}
317 			String value = StringUtils.right(completionStatus.getSystem(), 4) + "|" + completionStatus.getCode();
318 			ImmunizationStatus code = null;
319 			switch (value) {
320 			case "0322|CP", "0322|PA", "0323|A", "0323|U":
321 				code = ImmunizationStatus.COMPLETED;
322 				break;
323 			case "0322|NA", "0322|RE", "0323|D":
324 				code = ImmunizationStatus.NOTDONE;
325 				break;
326 			default:
327 				break;
328 			}
329 			// RXA-22 field in can be D, which is a delete (entered-in-error), 
330 			// that overrides first, so we don't check for conflicts
331 			izDetail.getImmunization().setStatus(code);
332 		}
333 	}
