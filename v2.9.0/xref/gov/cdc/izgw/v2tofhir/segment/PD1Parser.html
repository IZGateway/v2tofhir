27  /**
28   * The EVNParser extracts information from the EVN segment and adds it to the
29   * ProvenanceResource for the MessageHeader.
30   *
31   * @author Audacious Inquiry
32   *
33   */
34  @Produces(segment = "PD1", resource = Provenance.class)
35  @Slf4j
36  public class PD1Parser extends AbstractSegmentParser {
37  	Patient patient;
38  	PractitionerRole primaryCareProvider;
39  	Organization primaryCareOrg;
40  	Practitioner primaryCareDoc;
41  	private Reference gpRef;
42  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
43  	static {
44  		log.debug("{} loaded", PD1Parser.class.getName());
45  	}
46  
47  	/**
48  	 * Construct a new DSC Parser for the given messageParser.
49  	 * 
50  	 * @param messageParser The messageParser using this QPDParser
51  	 */
52  	public PD1Parser(MessageParser messageParser) {
53  		super(messageParser, "PD1");
54  		if (fieldHandlers.isEmpty()) {
55  			FieldHandler.initFieldHandlers(this, fieldHandlers);
56  		}
57  	}
58  
59  	@Override
60  	public List<FieldHandler> getFieldHandlers() {
61  		return fieldHandlers;
62  	}
63  
64  	public IBaseResource setup() {
65  		patient = getFirstResource(Patient.class);
66  		return patient;
67  	}
68  
69  	/**
70  	 * Set the primary care provider organization
71  	 * @param patientPrimaryFacility the primary care provider organization
72  	 */
73  	@ComesFrom(path = "Patient.generalPractitioner.Organization", field = 3, comment = "Patient Primary Facility")
74  	public void setPatientPrimaryFacility(Organization patientPrimaryFacility) {
75  		if (primaryCareProvider == null) {
76  			primaryCareProvider = createResource(PractitionerRole.class);
77  			gpRef = ParserUtils.toReference(primaryCareProvider, patient, "general-practitioner");
78  			patient.addGeneralPractitioner(gpRef);
79  		}
80  		if (primaryCareProvider.hasOrganization()) {
81  			if (patientPrimaryFacility.hasName()) {
82  				primaryCareOrg.addAlias(patientPrimaryFacility.getName());
83  			}
84  			for (Identifier ident : patientPrimaryFacility.getIdentifier()) {
85  				primaryCareOrg.addIdentifier(ident);
86  			}
87  			for (Address addr: patientPrimaryFacility.getAddress()) {
88  				primaryCareOrg.addAddress(addr);
89  			}
90  			for (ContactPoint telecom: patientPrimaryFacility.getTelecom()) {
91  				primaryCareOrg.addTelecom(telecom);
92  			}
93  		} else {
94  			addResource(primaryCareOrg = patientPrimaryFacility);
95  			primaryCareProvider.setOrganization(ParserUtils.toReference(patientPrimaryFacility, primaryCareProvider, "organization"));
96  		}
97  		ParserUtils.updateReference(primaryCareProvider, gpRef);
98  	}
99  
100 	/**
101 	 * Set the primary care provider 
102 	 * @param patientPrimaryCareProviderNameIdNo the primary care provider 
103 	 */
104 	@ComesFrom(path = "Patient.generalPractitioner.Practitioner", field = 4, comment = "Patient Primary Care Provider Name & ID No.")
105 	public void setPatientPrimaryCareProviderNameIdNo(Practitioner patientPrimaryCareProviderNameIdNo) {
106 		if (primaryCareProvider == null) {
107 			primaryCareProvider = createResource(PractitionerRole.class);
108 			gpRef = ParserUtils.toReference(primaryCareProvider, patient, "general-practitioner");
109 			patient.addGeneralPractitioner(gpRef);
110 		}
111 		if (primaryCareProvider.hasPractitioner()) {
112 			for (HumanName name: patientPrimaryCareProviderNameIdNo.getName()) {
113 				primaryCareDoc.addName(name);
114 			}
115 			for (Identifier ident : patientPrimaryCareProviderNameIdNo.getIdentifier()) {
116 				primaryCareDoc.addIdentifier(ident);
117 			}
118 			for (Address addr: patientPrimaryCareProviderNameIdNo.getAddress()) {
119 				primaryCareDoc.addAddress(addr);
120 			}
121 			for (ContactPoint telecom: patientPrimaryCareProviderNameIdNo.getTelecom()) {
122 				primaryCareDoc.addTelecom(telecom);
123 			}
124 		} else {
125 			addResource(primaryCareDoc = patientPrimaryCareProviderNameIdNo);
126 			primaryCareProvider.setPractitioner(ParserUtils.toReference(patientPrimaryCareProviderNameIdNo, primaryCareProvider, "practitioner"));
127 		}
128 		ParserUtils.updateReference(primaryCareProvider, gpRef);
129 	}

131 	/**
132 	 * Add a patient disability
133 	 * @param handicap the patient handicap
134 	 */
135 	@ComesFrom(path = "Patient.patient-disability", field = 6, comment = "Handicap")
136 	public void setHandicap(CodeableConcept handicap) {
137 		patient.addExtension(PathUtils.FHIR_EXT_PREFIX + "patient-disability", handicap);
138 	}

140 	/**
141 	 * Set the place of worship
142 	 * V2 provides a place to record the organization, FHIR only the name.  The reality is,
143 	 * if this is captured, it is likely just the name.
144 	 * @param placeOfWorship the place of worship
145 	 */
146 	@ComesFrom(path = "Patient.patient-congregation", field = 14, comment = "Place of Worship")
147 	public void setPlaceofWorship(Organization placeOfWorship) {
148 		patient.addExtension(PathUtils.FHIR_EXT_PREFIX + "patient-congregation", placeOfWorship.getNameElement());
149 	}
