44   * PIDParser handles PID segments and creates Patient, Account and Related Person resources.  It works
45   * with the PD1 parser to collect additional demographics, and the NK1 parser to collect
46   * contact information.

49   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-pid-to-account.html">V2 to FHIR PID to Account Mapping</a>
50   * 
51   * @author Audacious Inquiry

164 	 * Set a gender on a patient based on a Codeable Concept
165 	 * @param gender	A Codeable Concept representing the patient gender
166 	 */
167 	@ComesFrom(
168 		path="Patient.gender", field = 8, table="0001", 
169 		map = "Gender", 
170 		fhir = CodeableConcept.class
171 	)
172 	public void setGender(CodeableConcept gender) {
173 		for (Coding coding: gender.getCoding()) {
174 			if (coding.hasCode() && coding.hasSystem()) {
175 				String system = coding.getSystem();
176 				if (system.endsWith("0001")) {
177 					setGenderFromTable0001(patient.getGenderElement(), coding);
178 				} else if (Systems.DATA_ABSENT.equals(mapper.normalizeSystem(system))) {
179 					setDataAbsentReasonFromDataAbsent(patient.getGenderElement(), coding.getCode());
180 				} else if (Systems.NULL_FLAVOR.equals(mapper.normalizeSystem(system))) {
181 					setDataAbsentFromNullFlavor(patient.getGenderElement(), coding.getCode());
182 				}
183 			}
184 			if (patient.hasGender()) { break; }
185 		}
186 	}

188 	/**
189 	 * Set gender from HL7 Table 0001
190 	 * @param enumeration	The enumeration to set the value for 
191 	 * @param gender	The gender code
192 	 */
193 	public static void setGenderFromTable0001(Enumeration<AdministrativeGender> enumeration, Coding gender) {
194 		switch (gender.getCode()) {
195 		case "A":	enumeration.setValue(AdministrativeGender.OTHER); break;
196 		case "F":	enumeration.setValue(AdministrativeGender.FEMALE); break;
197 		case "M":	enumeration.setValue(AdministrativeGender.MALE); break;
198 		case "N":	FhirUtils.setDataAbsentReason(enumeration, "not-applicable"); break;
199 		case "O":	enumeration.setValue(AdministrativeGender.OTHER); break;
200 		case "U", "UNK":
201 					enumeration.setValue(AdministrativeGender.UNKNOWN); break;
202 		case "ASKU":
203 					FhirUtils.setDataAbsentReason(enumeration, "asked-unknown"); break;
204 		case "PHC1175":
205 					FhirUtils.setDataAbsentReason(enumeration, "asked-declined"); break;
206 		default:	FhirUtils.setDataAbsent(enumeration); break;
207 		}
208 	}

212 	 * @param gender	The gender value to update
213 	 * @param code		A code from DataAbsentReason
214 	 */
215 	public void setDataAbsentReasonFromDataAbsent(Enumeration<AdministrativeGender> gender, String code) {
216 		switch (code) {
217 		case "unsupported": gender.setValue(AdministrativeGender.OTHER); break;
218 		case "unknown": 	gender.setValue(AdministrativeGender.UNKNOWN); break;
219 		case "temp-unknown":gender.setValue(AdministrativeGender.UNKNOWN); break;
220 		default:			FhirUtils.setDataAbsentReason(gender, code); break;
221 		}
222 	}

262 		if (FhirUtils.isEmpty(race)) {
263 			return;
264 		}
265 		
266 		Coding firstRaceCoding = race.getCodingFirstRep();
267 		TerminologyMapperFactory.get()
268 			.mapRace(firstRaceCoding.getCode(), firstRaceCoding.getSystem())
269 			.ifPresent(ext -> {
270 				patient.addExtension(ext);
271 				getParser().getContext().setProperty(ext.getUrl(), ext);
272 			});
273 		
274 	}

292 		if (patient.hasAddress()) {
293 			if (!patient.getAddressFirstRep().hasDistrict()) {
294 				// Set county on first address
295 				patient.getAddressFirstRep().setDistrictElement(countyName);
296 			} else if (!patient.getAddressFirstRep().getDistrict().equals(countyName.getValue()) ) {
297 				// The county doesn't match district of first address, add a new address containing only the county
298 				patient.addAddress().setDistrictElement(countyName);
299 			} else {
300 				// They match so we can ignore this case.
301 			}
302 		}
303 	}
