58  /**
59   * OBXParser parses any OBX segments in a message to an Immunization if the message is a VXU or an 
60   * response to an immunization query.
61   * 
62   * It merges OBX information into the Immunization or ImmunizationRecommendation resource where 
63   * appropriate, otherwise it creates new Observation resources.
64   * 
65   * @author Audacious Inquiry
66   *
67   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-rxr-to-immunization.html">V2 to FHIR: RXR to Immunization</a>
68   */
69  @Produces(segment = "OBX", resource=Observation.class,
70  		  extra = { Immunization.class, ImmunizationRecommendation.class })
71  public class OBXParser extends AbstractSegmentParser {

100 	public IBaseResource setup() {
101 		izDetail = IzDetail.get(getMessageParser());
102 		if (izDetail.hasRecommendation()) {
103 			recommendation = izDetail.getRecommendation();
104 		}
105 
106 		observation = createResource(Observation.class);
107 		// The V2-to-FHIR IG maps every OBX to Observation.subject = the message's Patient
108 		// ("Observation[2].subject.reference=Patient[1].id" on the VXU_V04 to Bundle map).  Without
109 		// it an observation cannot be attributed to anyone -- a forecast observation has no partOf
110 		// either, so subject is its only link to the patient.
111 		Patient patient = getLastResource(Patient.class);
112 		if (patient != null) {
113 			observation.setSubject(ParserUtils.toReference(patient, observation, "patient", "subject"));
114 		}
115 		return observation;
116 	}

118 	private enum VisCode {
119 		// Used with Immunization
120 		VACCINE_ELIGIBILITY_CODE("64994-7"),
121 		VIS_DOCUMENT_TYPE_CODE("69764-9"),
122 		VIS_VERSION_DATE_CODE("29768-9"),
123 		VIS_DELIVERY_DATE_CODE("29769-7"),
124 		// Used with ImmunizationRecommendation
125 		FORECAST_SCHEDULE("59779-9"),
126 		/**
127 		 * Vaccine type.  In a Z42 (evaluated history and forecast) message this is the vaccine a
128 		 * dose was evaluated against. So renamed to work for both. Otherwise, works how it always has.
129 		 * Replaces FORECAST_VACCINE_CODE, which declared this same LOINC code later in the enum and
130 		 * so was never matched because VisCode.match() returns the first constant in declaration order.
131 		 */
132 		VACCINE_TYPE("30956-7"),
133 		FORECAST_SERIES_NAME("59780-7"),
134 		FORECAST_DOSE_NUMBER("30973-2"),
135 		DOSE_VALIDITY("59781-5"),
136 		FORECAST_NUMBER_DOSES("59782-3"),
137 		SERIES_STATUS("59783-1"),
138 		NEXT_DOSE_EARLIEST("30981-5", "Earliest date to give"),
139 		NEXT_DOSE_RECOMMENDED("30980-7", "Date vaccine due"),
140 		NEXT_DOSE_LATEST("59777-3", "Latest date to give immunization"),
141 		NEXT_DOSE_OVERDUE("59778-1", "Date when overdue for immunization"),
142 		WHY_INVALID("30982-3"),
143 		// None of the above
144 		OTHER("");
145 		private final String loincCode;
146 		private final String display;
147 		private VisCode(String loincCode) {
148 			this.loincCode = loincCode;
149 			this.display = null;
150 		}
151 		private VisCode(String loincCode, String display) {
152 			this.loincCode = loincCode;
153 			this.display = display;
154 		}
155 		public String getCode() {
156 			return loincCode;
157 		}
158 		private static VisCode match(CodeableConcept cc) {
159 			for (Coding coding: cc.getCoding()) {
160 				VisCode v = match(coding);
161 				if (!VisCode.OTHER.equals(v)) {
162 					return v;
163 				}
164 			}
165 			return OTHER;
166 		}
167 		private static VisCode match(Coding coding) {
168 			if (Systems.LOINC.equals(coding.getSystem())) {
169 				for (VisCode v: VisCode.values()) {
170 					if (coding.hasCode() && coding.getCode().equals(v.loincCode)) {
171 						return v;
172 					}
173 				}
174 			}
175 			return OTHER;
176 		}
177 		boolean isPresent(ImmunizationEducationComponent education) {
178 			switch (this) {
179 			case VIS_DELIVERY_DATE_CODE:
180 				return education.hasPresentationDate();
181 			case VIS_DOCUMENT_TYPE_CODE:
182 				return education.getDocumentTypeElement().hasValue();
183 			case VACCINE_TYPE:
184 				return education.getDocumentTypeElement().hasExtension();
185 			case VIS_VERSION_DATE_CODE:
186 				return education.hasPublicationDateElement();
187 			default:
188 				return false;
189 			}
190 		}
191 		String getDisplay() {
192 			return display;
193 		}

207 	 * Figure out where to redirect OBX-5 information for VIS OBX types 
208 	 * @param observationCode the observation code in OBX-3
209 	 */
210 	@ComesFrom(path = "Observation.code", field = 3, comment = "Observation Identifier") 
211 	public void redirectTo(CodeableConcept observationCode) {
212 		observation.setCode(observationCode);
213 		if (izDetail.hasImmunization() || izDetail.hasRecommendation()) {
214 			redirect = VisCode.match(observationCode);
215 		} else {
216 			redirect = null;
217 		}
218 		if (izDetail.hasRecommendation() && VisCode.VACCINE_TYPE.equals(redirect)) {
219 			// A vaccine type starts a new forecast. OBX-3 is processed before OBX-5, so this
220 			// component exists before its vaccine code is set.
221 			ImmunizationRecommendationRecommendationComponent block = izDetail.startForecastBlock();
222 			if (block != null) {
223 				recommendation = block;
224 			}
225 		}

229 	 * Set the value
230 	 * @param v2Type	The v2 data type to convert.
231 	 */
232 	@ComesFrom(path = "Observation.value[x]", field = 5, comment = "Observation Value")
233 	public void setValue(Type v2Type) {  // NOSONAR -- Yes, it's a big switch statement
234 		v2Type = DatatypeConverter.adjustIfVaries(v2Type);
235 		if (type == null) {
236 			getParser().warn("Type is unknown for OBX-5");
237 			return;
238 		}
239 		Class<? extends IBase> target = null;
240 		switch (type) {
241 		case "AD":	target = Address.class; break;	//	Address	
242 		case "CE", 
243 			 "CF",  		
244 			 "CNE",		
245 			 "CWE":	
246 				 // Coded Element (With Formatted Values)
247 				 //	Coded with No Exceptions
248 				 // Coded Entry
249 				 target = CodeableConcept.class; break;		
250 		case "CK":	target = Identifier.class; break;	//	Composite ID With Check Digit	
251 		case "CX":	target = Identifier.class; break;	//	Extended Composite ID With Check Digit	
252 		case "DT":	target = DateTimeType.class; break;	//	Observation doesn't like DateType	
253 		case "DTM":	target = DateTimeType.class; break;	//	Time Stamp (Date & Time)
254 		case "ED":  target = Attachment.class; break;	//	Encapsulated Data	
255 		case "FT":	target = StringType.class; break; //	Formatted Text (Display)	
256 		case "HD":	target = Identifier.class; break; //	Hierarchic Designator
257 		case "ID":	target = CodeType.class; break; //	Coded Value for HL7 Defined Tables	
258 		case "IS":	target = CodeType.class; break; //	Coded Value for User-Defined Tables	
259 		case "NM":	target = Quantity.class; break; //	Numeric	(Observation only accepts Quantity)
260 		case "PL":	target = Location.class; break; //	Person Location
261 		case "PN":	target = HumanName.class; break; //	Person Name
262 		case "RP":	target = Attachment.class; break; //	Reference Pointer
263 		case "SN":	
264 			if (v2Type instanceof Composite comp && comp.getComponents().length > 3) {
265 				target = Range.class; 	
266 			} else {
267 				target = Quantity.class;
268 			}
269 			break;
270 		case "ST":	target = StringType.class; break; //	String Data.	
271 		case "TM":	target = TimeType.class; break; //	Time	
272 		case "TN":	target = ContactPoint.class; break; //	Telephone Number	
273 		case "TS":	target = DateTimeType.class; break; //	TimeStamp	 (OBX requires DateTimeType)
274 		case "TX":	target = StringType.class; break; //	Text Data (Display)	
275 		case "XAD":	target = Address.class; break; //	Extended Address	
276 		case "CN":	target = RelatedPerson.class; break;	//	Composite ID And Name	
277 		case "XCN":	target = RelatedPerson.class; break; //	Extended Composite Name And Number For Persons	
278 		case "XON":	target = Organization.class; break; //	Extended Composite Name And Number For Organizations	
279 		case "XPN":	target = HumanName.class; break; //	Extended Person Name	
280 		case "XTN":	target = ContactPoint.class; break; //	Extended Telecommunications Number
281 		case "CP",	//	Composite Price	
282 			 "DR",	//	Date/Time Range	
283 			 "MA",	//	Multiplexed Array	
284 			 "MO",	//	Money	
285 			 "NA":	//	Reference Pointer	
286 		default:
287 			break;
288 		}
289 		
290 		if (target == null) {
291 			getParser().warn("Cannot convert V2 {}", type);
292 			return;
293 		}
294 		
295 		IBase converted = DatatypeConverter.convert(target, v2Type, null);
296 		if (converted instanceof Organization org) {
297 			observation.setValue(org.getNameElement());
298 		} else if (converted instanceof RelatedPerson rp) {
299 			observation.setValue(new StringType(TextUtils.toString(rp.getNameFirstRep())));
300 		} else if (converted instanceof Practitioner pr) {
301 			observation.setValue(new StringType(TextUtils.toString(pr.getNameFirstRep())));
302 		} else if (converted instanceof Location loc) {
303 			observation.setValue(new StringType(TextUtils.toString(loc.getNameElement())));
304 		} else if (converted instanceof Address addr) {
305 			observation.setValue(new StringType(TextUtils.toString(addr)));
306 		} else if (converted instanceof Identifier identifier) {
307 			// You cannot set an identifier value directly into Observation.value
308 			if (identifier.hasSystem() && identifier.hasValue()) {
309 				observation.setValue(new StringType(identifier.getSystem() + "#" + identifier.getValue()));
310 			} else if (identifier.hasValue()) {
311 				observation.setValue(identifier.getValueElement());
312 			} else if (identifier.hasSystem()) {
313 				observation.setValue(new StringType(identifier.getSystem()));
314 			}
315 			// Save the identifier as an external identifier for the observation
316 			observation.addIdentifier(identifier);
317 		} else if (converted instanceof Attachment attachment) {
318 			docRef = createDocRef(attachment);
319 			observation.getDerivedFrom().add(ParserUtils.toReference(docRef, observation, "derivedFrom"));
320 		} else if (converted instanceof org.hl7.fhir.r4.model.Type t){
321 			observation.setValue(t);
322 		}
323 		
324 		if (redirect == null || VisCode.OTHER.equals(redirect)) {
325 			return;
326 		}
327 		
328 		// This observation applies to either a created Immunization or an ImmunizationRecommendation,
329 		// link it via Observation.partOf.
330 		linkObservation();
331 
332 		if (izDetail.hasImmunization()) {
333 			handleVisObservations(converted);
334 		} else if (izDetail.hasRecommendation()) {
335 			handleRecommendationObservations(converted);
336 		}
337 	}

339 	private DocumentReference createDocRef(Attachment attachment) {
340 		docRef = createResource(DocumentReference.class);
341 		docRef.setStatus(DocumentReferenceStatus.CURRENT);
342 		docRef.addContent().setAttachment(attachment);
343 		// Backlink the document reference to the observation
344 		docRef.getContext().getRelated().add(ParserUtils.toReference(observation, docRef, "relatedContext"));
345 		return docRef;
346 	}

348 	@Override
349 	/**
350 	 * Finish populating the docRef resource if present from the observation 
351 	 */
352 	public void finish() {
353 		if (docRef == null || docRef.isEmpty()) {
354 			return;
355 		}
356 		docRef.setDate(observation.getEffectiveDateTimeType().getValue());
357 		// The authors of the document reference are the performers of the observation.
358 		for (Reference r: observation.getPerformer()) {
359 			Resource res = (Resource) r.getResource();
360 			if (res != null) {
361 				docRef.getAuthor().add(ParserUtils.toReference(res, docRef, "author"));
362 			}
363 		}
364 		
365 		// Update status and docStatus based on observation status
366 		ObservationStatus status = observation.getStatus();
367 		if (status != null) {
368 			switch (status) {
369 			case AMENDED, CORRECTED:
370 				docRef.setStatus(DocumentReferenceStatus.CURRENT);
371 				docRef.setDocStatus(ReferredDocumentStatus.AMENDED);
372 				break;
373 			case CANCELLED:
374 				docRef.setStatus(DocumentReferenceStatus.SUPERSEDED);
375 				docRef.setDocStatus(ReferredDocumentStatus.ENTEREDINERROR);
376 				break;
377 			case ENTEREDINERROR:
378 				docRef.setStatus(DocumentReferenceStatus.ENTEREDINERROR);
379 				docRef.setDocStatus(ReferredDocumentStatus.ENTEREDINERROR);
380 				break;
381 			case FINAL:
382 				docRef.setStatus(DocumentReferenceStatus.CURRENT);
383 				docRef.setDocStatus(ReferredDocumentStatus.FINAL);
384 				break;
385 			case NULL:
386 				docRef.setStatus(DocumentReferenceStatus.NULL);
387 				docRef.setDocStatus(ReferredDocumentStatus.NULL);
388 				break;
389 			case PRELIMINARY, REGISTERED:
390 				docRef.setStatus(DocumentReferenceStatus.CURRENT);
391 				docRef.setDocStatus(ReferredDocumentStatus.PRELIMINARY);
392 				break;
393 			case UNKNOWN:
394 				docRef.setStatus(null);
395 				docRef.setDocStatus(null);
396 				break;
397 			default:
398 				break;
399 			}
400 		}
401 		// The subject (and sourcePatientInfo) of the DocumentReference is the subject of the observation.
402 		Resource subject = getResource(observation.getSubject());
403 		if (subject != null) {
404 			docRef.setSubject(ParserUtils.toReference(subject, docRef, "subject"));
405 			if (subject instanceof Patient) {
406 				docRef.getContext().setSourcePatientInfo(ParserUtils.toReference(subject, docRef, "sourcePatientInfo"));
407 			}
408 		}
409 		
410 		// The encounter of the DocumentReference is the encounter of the observation.
411 		Resource encounter = getResource(observation.getEncounter());
412 		if (encounter != null) {
413 			docRef.getContext().getEncounter().add(ParserUtils.toReference(encounter, docRef, "encounter"));
414 		}
415 		
416 	}

445 			);
446 		}

449 	/**
450 	 * Add one observation from a history group to the Immunization.
451 	 * <p>
452 	 * The observation code in OBX-3 selects the target element:
453 	 * <ul>
454 	 * <li>64994-7 Vaccine funding program eligibility goes to programEligibility.</li>
455 	 * <li>59779-9 Immunization schedule used goes to protocolApplied.authority.</li>
456 	 * <li>30973-2 Dose number in series goes to protocolApplied.doseNumber.</li>
457 	 * <li>59782-3 Number of doses in series goes to protocolApplied.seriesDoses.</li>
458 	 * <li>69764-9, 29768-9 and 29769-7 are Vaccine Information Statement codes.
459 	 *     They go to education.</li>
460 	 * <li>30956-7 Vaccine type goes to education only if the message is not a Z42.</li>
461 	 * </ul>
462 	 * <p>
463 	 * The method returns immediately for all other codes. This is necessary because
464 	 * getEducationFirstRep() adds an element. An early return prevents an empty education element.
465 	 *
466 	 * @param converted The converted OBX-5 value
467 	 */
468 	private void handleVisObservations(IBase converted) {
469 		Immunization immunization = izDetail.getImmunization();
470 		switch (redirect) {
471 		case VACCINE_ELIGIBILITY_CODE:
472 			// linkObservation() has already set Observation.partOf for this observation.
473 			immunization.addProgramEligibility((CodeableConcept) converted);
474 			return;
475 		case FORECAST_SCHEDULE:
476 			// The schedule the dose was evaluated against names its publishing authority.
477 			immunization.getProtocolAppliedFirstRep()
478 				.setAuthority(ParserUtils.toReference(getScheduleAuthority(converted), immunization, "authority"));
479 			return;
480 		case FORECAST_DOSE_NUMBER:
481 			// R4 requires protocolApplied.doseNumber[x] 1..1, and this is the only thing that
482 			// supplies it on the history path.
483 			toPositiveInt(converted).ifPresent(immunization.getProtocolAppliedFirstRep()::setDoseNumber);
484 			return;
485 		case FORECAST_NUMBER_DOSES:
486 			toPositiveInt(converted).ifPresent(immunization.getProtocolAppliedFirstRep()::setSeriesDoses);
487 			return;
488 		case VACCINE_TYPE:
489 			if (izDetail.isEvaluatedHistoryAndForecast()) {
490 				// Z42: the antigen this dose was evaluated against, not a VIS.  See VisCode.VACCINE_TYPE.
491 				return;
492 			}
493 			// Any other profile keeps the original reading, so nothing this library produced for a
494 			// VXU or a history-only response changes.
495 			break;
496 		case VIS_DELIVERY_DATE_CODE, VIS_DOCUMENT_TYPE_CODE, VIS_VERSION_DATE_CODE:
497 			break;
498 		default:
499 			// Nothing else belongs in Immunization.education -- 59781-5 dose validity in particular.
500 			// Return before getLastEducation(), which would otherwise create an empty element.
501 			return;
502 		}
503 		ImmunizationEducationComponent education = getLastEducation(immunization);
504 		// If the observation is already present in education
505 		if (redirect.isPresent(education)) {
506 			// Create a new education element.
507 			education = izDetail.getImmunization().addEducation();
508 		}
509 		switch (redirect) {
510 		case VIS_DELIVERY_DATE_CODE:
511 			if (converted instanceof BaseDateTimeType bt) {
512 				education.setPresentationDateElement(DatatypeConverter.castInto(bt, new DateTimeType()));
513 				return;
514 			} 
515 			getParser().warn("Cannot convert {} to DateTimeType for VIS Delivery Date", converted.fhirType());
516 			return;
517 		case VIS_DOCUMENT_TYPE_CODE:
518 			if (converted instanceof StringType sv) {
519 				education.setDocumentTypeElement(sv);
520 				return;
521 			} 
522 			if (converted instanceof CodeableConcept cc) {
523 				education.setDocumentType(TextUtils.toString(cc));
524 				return;
525 			} 
526 			getParser().warn("Cannot convert {} to StringType for VIS Document Type", converted.fhirType());
527 			return;
528 		case VACCINE_TYPE:
529 			// Only reached outside a Z42 -- the vaccine the VIS covers.
530 			if (converted instanceof CodeableConcept cc) {
531 				education.getDocumentTypeElement()
532 					.addExtension()
533 						.setUrl("http://hl7.org/fhir/StructureDefinition/iso21090-SC-coding")
534 						.setValue(cc.getCodingFirstRep());
535 				return;
536 			}
537 			getParser().warn("Cannot convert {} to CodeableConcept for VIS Vaccine Type", converted.fhirType());
538 			return;
539 		case VIS_VERSION_DATE_CODE:
540 			if (converted instanceof BaseDateTimeType bt) {
541 				education.setPublicationDateElement(DatatypeConverter.castInto(bt, new DateTimeType()));
542 				return;
543 			} 
544 			getParser().warn("Cannot convert {} to DateTimeType for VIS Version Date", converted.fhirType());
545 			return;
546 		default:
547 			return;
548 		}
549 	}

552 	 * Get or create the Organization that published the schedule named by a 59779-9 observation
553 	 * (e.g. VXC16^ACIP^CDCPHINVS).  One Organization is shared by the whole message.
554 	 * @param converted	The converted OBX-5 value
555 	 * @return	The publishing authority
556 	 */
557 	private Organization getScheduleAuthority(IBase converted) {
558 		Organization authority = izDetail.getScheduleAuthority();
559 		if (authority == null) {
560 			authority = createResource(Organization.class);
561 			izDetail.setScheduleAuthority(authority);
562 		}
563 		if (!authority.hasName() && converted instanceof CodeableConcept cc) {
564 			Coding coding = cc.getCodingFirstRep();
565 			// The schedule name is the display (ACIP), falling back to the code (VXC16).
566 			authority.setName(coding.hasDisplay() ? coding.getDisplay() : coding.getCode());
567 			authority.addIdentifier(new Identifier().setSystem(coding.getSystem()).setValue(coding.getCode()));
568 		}
569 		return authority;
570 	}

579 	/**
580 	 * Link the Observation to the Immunization it applies to.
581 	 *
582 	 * A forecast observation gets no link at all, deliberately.  R4 types Observation.partOf as
583 	 * Reference(MedicationAdministration | MedicationDispense | MedicationStatement | Procedure |
584 	 * Immunization | ImagingStudy), so an ImmunizationRecommendation cannot go there.  The element
585 	 * that would fit, recommendation.supportingPatientInformation, is not used either: every
586 	 * forecast observation's content is already carried by the recommendation component itself
587 	 * (vaccineCode, forecastStatus, dateCriterion, doseNumber, seriesDoses, forecastReason,
588 	 * authority), so the reference would only point at a duplicate.  Consumers that apply a
589 	 * searchset filter drop the Observations, which would leave those references unresolvable.
590 	 *
591 	 * TODO: if a forecast observation ever carries something the component cannot hold, link it
592 	 * with ParserUtils.toReference(observation, immunizationRecommendation, "information") — the
593 	 * search name matters, because downstream _include resolves on registered names, not on fields.
594 	 */
595 	private void linkObservation() {
596 		if (VisCode.OTHER.equals(redirect) || !izDetail.hasImmunization()) {
597 			return;
598 		}
599 		// Both spellings are registered: "part-of" is the FHIR search parameter name, "partof" is
600 		// what this library has always used and what existing callers pass to _revinclude.
601 		Reference ref = ParserUtils.toReference(izDetail.getImmunization(), observation, "partof", "part-of");
602 		if (!observation.getPartOf().contains(ref)) {
603 			observation.addPartOf(ref);
604 		}
605 	}

607 	private void handleRecommendationObservations(IBase converted) {
608 		if (redirect == null || recommendation == null) {
609 			return;
610 		}
611 
612 		ImmunizationRecommendationRecommendationDateCriterionComponent criterion = null;
613 		int value = 0;
614 		switch (redirect) {
615 		case DOSE_VALIDITY:
616 			break;
617 		case FORECAST_DOSE_NUMBER:
618 			value = ((Quantity)converted).getValue().intValue(); 
619 			recommendation.setDoseNumber(new PositiveIntType(value));
620 			break;
621 		case FORECAST_NUMBER_DOSES:
622 			value = ((Quantity)converted).getValue().intValue(); 
623 			recommendation.setSeriesDoses(new PositiveIntType(value));
624 			break;
625 		case FORECAST_SCHEDULE:
626 			// The schedule the forecast was computed against names its publishing authority.
627 			// R4: "Indicates the authority who published the protocol (e.g. ACIP)."
628 			ImmunizationRecommendation forecast = izDetail.getImmunizationRecommendation();
629 			if (!forecast.hasAuthority()) {
630 				forecast.setAuthority(
631 					ParserUtils.toReference(getScheduleAuthority(converted), forecast, "authority")
632 				);
633 			}
634 			break;
635 		case FORECAST_SERIES_NAME:
636 			recommendation.setSeriesElement((StringType)converted);
637 			break;
638 		case VACCINE_TYPE:
639 			recommendation.addVaccineCode((CodeableConcept)converted);
640 			break;
641 		case NEXT_DOSE_OVERDUE,
642 			 NEXT_DOSE_EARLIEST,
643 			 NEXT_DOSE_LATEST,
644 			 NEXT_DOSE_RECOMMENDED:
645 			criterion = recommendation.addDateCriterion();
646 			criterion.setCode(new CodeableConcept().addCoding(new Coding(Systems.LOINC, redirect.getCode(), redirect.getDisplay())));
647 			criterion.setValueElement(DatatypeConverter.castInto((BaseDateTimeType)converted, new DateTimeType()));
648 			break;
649 		case SERIES_STATUS:
650 			recommendation.setForecastStatus((CodeableConcept)converted);
651 			break;
652 		case WHY_INVALID:
653 			// The reason the forecast has the status it has.  IIS send this as free text
654 			// ("Patient has exceeded the maximum age"), and forecastReason's binding is
655 			// example-strength, so text with no coding is the faithful representation.
656 			if (converted instanceof CodeableConcept cc) {
657 				recommendation.addForecastReason(cc);
658 			} else if (converted instanceof org.hl7.fhir.r4.model.PrimitiveType<?> text && text.hasValue()) {
659 				recommendation.addForecastReason(new CodeableConcept().setText(text.asStringValue()));
660 			} else {
661 				getParser().warn("Cannot convert {} to a forecast reason", converted.fhirType());
662 			}
663 			break;
664 		default:
665 			break;
666 		}
667 	}

721 	/**

723 	 * @param observationResultStatus a code from table 0085 
724 	 * @return the converted ObservationStatus value
725 	 */
726 	private ObservationStatus toObservationStatus(CodeType observationResultStatus) {
727 		if (!observationResultStatus.hasCode()) {
728 			return null;
729 		}
730 		switch (observationResultStatus.getCode()) {
731 		case "C":	return ObservationStatus.CORRECTED;
732 		case "D":	return ObservationStatus.ENTEREDINERROR;
733 		case "F":	return ObservationStatus.FINAL;

736 		case "R":	return ObservationStatus.PRELIMINARY;
737 		case "S":	return ObservationStatus.PRELIMINARY;
738 		case "U":	return ObservationStatus.FINAL;
739 		case "W":	return ObservationStatus.ENTEREDINERROR;
740 		case "X":	return ObservationStatus.CANCELLED;
741 		default:	return null;
742 		}
