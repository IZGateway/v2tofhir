25  /**
26   * The MSH Parser creates a MessageHeader Resource for MSH segments.
27   * 
28   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-msh-to-bundle.html">V2-to-FHIR: MSH to Bundle</a>
29   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-msh-to-messageheader.html">V2-to-FHIR: MSH to MessageHeader</a>
30   * 
31   * @author Audacious Inquiry
32   */
33  @Produces(segment="MSH", resource=MessageHeader.class, extra = { Organization.class, OperationOutcome.class, Bundle.class })
34  @Slf4j
35  public class MSHParser extends AbstractSegmentParser {
36  	private static final String SENDER = "sender";
37  
38  	private static final String RECEIVER = "receiver";
39  
40  	private MessageHeader mh;
41  	
42  	private static final String SENDING_ORGANIZATION = "sendingOrganization";
43  	private static final String DESTINATION_ORGANIZATION = "destinationOrganization";
44  
45  	static {
46  		log.debug("{} loaded", MSHParser.class.getName());

48  	private static final List<FieldHandler> fieldHandlers = new ArrayList<>();
49  	/**
50  	 * Construct a new MSHParser for the specified messageParser instance.
51  	 * @param messageParser	The messageParser this parser is associated with.
52  	 */
53  	public MSHParser(MessageParser messageParser) {
54  		super(messageParser, "MSG");
55  		if (fieldHandlers.isEmpty()) {
56  			FieldHandler.initFieldHandlers(this, fieldHandlers);
57  		}
58  	}

60  	public List<FieldHandler> getFieldHandlers() {
61  		return fieldHandlers;
62  	}

64  	@Override
65  	public IBaseResource setup() {
66  		Provenance provenance;
67  		mh = createResource(MessageHeader.class);
68  		provenance = (Provenance) mh.getUserData(Provenance.class.getName());
69  		if (provenance != null) {
70  			provenance.getActivity().addCoding(new Coding(null, "v2-FHIR transformation", "HL7 V2 to FHIR transformation"));
71  		}
72  		return mh;
73  	}

75  	/**
76  	 * Create the sending organization (or update the existing one) and add a 
77  	 * reference to it to the MessageHeader.sender field.
78  	 *  
79  	 * @param ident	The identifier of the sending organization.
80  	 */
81  	@ComesFrom(path="MessageHeader.responsible", field = 22, type="XON", comment="Sending Responsible Organization")
82  	public void setSourceSender(Identifier ident) {
83  		Organization sendingOrganization = getOrganization(SENDING_ORGANIZATION);
84  		sendingOrganization.addIdentifier(ident);
85  		mh.setResponsible(ParserUtils.toReference(sendingOrganization, mh, "responsible"));
86  	}

88  	/**
89  	 * Create the sending organization (or update the existing one) and add a 
90  	 * reference to it to the MessageHeader.sender field.
91  	 *  
92  	 * @param senderEndpoint	The identifier of the sending application.
93  	 */
94  	@ComesFrom(path="MessageHeader.source.endpoint", field = 3, 
95  			   also={ "MessageHeader.sender", "Organization.endpoint.identifier", "Endpoint.name", "Endpoint.identifier" }, 
96  			   comment="Sending Application")
97  	public void setSourceSenderEndpoint(Identifier senderEndpoint) {
98  		mh.getSource().setEndpoint(senderEndpoint.getSystem());
99  		Organization sendingOrganization = getOrganization(SENDING_ORGANIZATION); 
100 		Endpoint ep = createResource(Endpoint.class);
101 		ep.setName(senderEndpoint.getSystem());
102 		ep.addIdentifier().setValue(senderEndpoint.getSystem());
103 		sendingOrganization.getEndpoint().add(ParserUtils.toReference(ep, sendingOrganization, "endpoint"));
104 		mh.setSender(ParserUtils.toReference(sendingOrganization, mh, SENDER));
105 	}

107 	/**
108 	 * Set the sending facility
109 	 * @param sendingFacility The sending facility
110 	 */
111 	@ComesFrom(path="MessageHeader.source.name", field = 4, comment="Sending Facility",
112 			also="MessageHeader.sender")
113 	public void setSourceSenderName(Identifier sendingFacility) {
114 		mh.getSource().setName(sendingFacility.getSystem());
115 		Organization sendingOrganization = getOrganization(SENDING_ORGANIZATION);
116 		sendingOrganization.setName(sendingFacility.getSystem());
117 		mh.setSender(ParserUtils.toReference(sendingOrganization, mh, SENDER));
118 	}

120 	private Organization getOrganization(String organizationType) {
121 		Organization organization = (Organization) mh.getUserData(organizationType);
122 		if (organization == null) {
123 			if (SENDING_ORGANIZATION.equals(organizationType)) {
124 				organization = createResource(Organization.class);
125 				mh.setSender(ParserUtils.toReference(organization, mh, SENDER));
126 				mh.setUserData(organizationType, organization);
127 			} else if (DESTINATION_ORGANIZATION.equals(organizationType)) {
128 				organization = createResource(Organization.class);
129 				mh.getDestinationFirstRep().setReceiver(ParserUtils.toReference(organization, mh, RECEIVER));
130 				mh.setUserData(organizationType, organization);
131 			} else {
132 				throw new IllegalArgumentException("Unrecognized organization type: " + organizationType);
133 			}
134 		}
135 		return organization;
136 	}

138 	/**
139 	 * Copy information from the reciever organization to the destination organization
140 	 * @param receiver The receiving organization from an XON
141 	 */
142 	@ComesFrom(path="MessageHeader.destination.receiver", field = 23, type="XON", comment="Receiving Responsible Organization")
143 	public void setDestinationReceiver(Organization receiver) {
144 		Organization receivingOrganization = getOrganization(DESTINATION_ORGANIZATION);
145 		if (receiver.hasName()) {
146 			receivingOrganization.setName(receiver.getName());
147 		}
148 		if (receiver.hasIdentifier()) {
149 			for (Identifier ident: receiver.getIdentifier()) {
150 				receivingOrganization.addIdentifier(ident);
151 			}
152 		}
153 		if (receiver.hasEndpoint()) {
154 			for (Reference endpoint: receiver.getEndpoint()) {
155 				receivingOrganization.addEndpoint(endpoint);
156 			}
157 		}
158 		mh.getDestinationFirstRep().setReceiver(ParserUtils.toReference(receivingOrganization, mh, RECEIVER));
159 	}

163 	 * Set the receiver endpoint from MSH-5
164 	 * @param receiverEndpoint	The receiver endpoint
165 	 */
166 	@ComesFrom(path="MessageHeader.destination.endpoint", field = 5, comment="Receiving Application",
167 			   also="MessageHeader.destination.receiver.Organization.endpoint")
168 	public void setDestinationReceiverEndpoint(Identifier receiverEndpoint) {
169 		mh.getDestinationFirstRep().setEndpoint(receiverEndpoint.getSystem());
170 		
171 		Organization organization = getOrganization(DESTINATION_ORGANIZATION);
172 		Endpoint endpoint = createResource(Endpoint.class);
173 		endpoint.addIdentifier(receiverEndpoint);
174 		organization.addEndpoint(ParserUtils.toReference(endpoint, organization, "endpoint"));
175 		mh.getDestinationFirstRep().setReceiver(ParserUtils.toReference(organization, mh, RECEIVER));
176 	}

227 		getBundle().setImplicitRules(rules);

242 	 * Set the Message Profile from MSH-21
243 	 * @param ident	The message profile id
244 	 */
245 	@ComesFrom(path="Bundle.meta.profile", field = 21)
246 	public void setMetaProfile(Identifier ident) {
247 		String rules = "";
248 		if (ident.hasSystem()) {
249 			rules = ident.getSystem();
250 		}
251 		if (ident.hasValue()) {
252 			rules = rules + "#" + ident.getValue();
253 		}
254 		getBundle().getMeta().addProfile(rules);
255 	}
