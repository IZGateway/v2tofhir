1   package gov.cdc.izgw.v2tofhir.segment;
2   
3   import java.util.List;
4   
5   import org.hl7.fhir.instance.model.api.IBase;
6   import org.hl7.fhir.r4.model.Base;
7   import org.hl7.fhir.r4.model.Bundle;
8   import org.hl7.fhir.r4.model.OperationOutcome;
9   import org.hl7.fhir.r4.model.OperationOutcome.OperationOutcomeIssueComponent;
10  import gov.cdc.izgw.v2tofhir.annotation.Produces;
11  import gov.cdc.izgw.v2tofhir.converter.Parser;
12  
13  /**
14   * 
15   * A processor for a structure to convert to FHIR.
16   * 
17   * @author Audacious Inquiry
18   * @param <U>	The unit the structure lives in. 
19   * @param <S>	The structure to convert.
20   */
21  public interface Processor<U, S> {
22  	/**
23  	 * This is a do-nothing marker class to indicate Parsers that cannot be found
24  	 * in the cache of processors.
25  	 * 
26  	 * @author Audacious Inquiry
27  	 *
28  	 */
29  	static class NullProcessor implements Processor<Object, Object> {
30  		/** Create a processor that does nothing. */
31  		public NullProcessor() {
32  			// Nothing to initialize
33  		}
34  
35  		@Override
36  		public Base parse(Object structure) {
37  			// Do Nothing
38  			return null;
39  		}
40  
41  		@Override
42  		public boolean isEmpty(Object structure) {
43  			return true;
44  		}
45  
46  		@Override
47  		public String structure() {
48  			return "(null)";
49  		}
50  
51  		@Override
52  		public Parser<Object, Object> getParser() {
53  			return null;
54  		}
55  	}
56  
57  	/** A do-nothing processor.  */
58  	static final Processor<Object, Object> NULL_PROCESSOR = new NullProcessor();
59  	/** The class of the do-nothing processor.  */
60  	static final Class<NullProcessor> NULL_PROCESSOR_CLASS = NullProcessor.class;
61  
62  	/**
63  	 * Parses an structure into FHIR Resources.
64  	 * @param structure	The structure to parse
65  	 * @return  the constructed FHIR Resource or data type
66  	 * @throws Exception A structure to parse.
67  	 */
68  	IBase parse(S structure) throws Exception;  // NOSONAR
69  	
70  	/**
71  	 * Returns true if the structure is effectively empty (e.g., no conversion to be performed).
72  	 * @param structure	The structure to check
73  	 * @return true if the structure is effectively empty
74  	 */
75  	boolean isEmpty(S structure);
76  
77  	/**
78  	 * Return what this parser produces.
79  	 * @return what this parser produces
80  	 */
81  	default Produces getProduces() {
82  		return this.getClass().getAnnotation(Produces.class);
83  	}
84  
85  	/**
86  	 * The name of the segment this parser works on.
87  	 * @return The name of the segment this parser works on
88  	 */
89  	String structure();
90  
91  	/**
92  	 * Get the parser that uses this processor.
93  	 * @return the current parser
94  	 */
95  	Parser<U, S> getParser();
96  
97  	/**
98  	 * Get the bundle that is being prepared.
99  	 * This method can be used by parsers to get additional information from the Bundle
100 	 * @return The bundle that is being prepared during the parse.
101 	 */
102 	default Bundle getBundle() {
103 		return getParser().getContext().getBundle();
104 	}
105 
106 	/**
107 	 * Get the last issue in an OperationOutcome, adding one if there are none
108 	 * @param oo	The OperationOutcome
109 	 * @return	The issue
110 	 */
111 	public static OperationOutcomeIssueComponent getLastIssue(OperationOutcome oo) {
112 		List<OperationOutcomeIssueComponent> l = oo.getIssue();
113 		if (l.isEmpty()) {
114 			return oo.getIssueFirstRep();
115 		}
116 		return l.get(l.size() - 1);
117 	}
118 	
119 	
120 	/**
121 	 * Finish up any processing after all units have been processed.
122 	 */
123 	default void finish() {
124 		// do nothing by default
125 	}
126 
127 	/**
128 	 * Handle any post-processing of the structure after it and all 
129 	 * sub-structures have been processed.
130 	 * 
131 	 * @param structure	The structure that was processed.
132 	 */
133 	default void postProcess(S structure) {
134 		// do nothing by default
135 	}
136 }
