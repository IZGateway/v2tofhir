13  import lombok.extern.slf4j.Slf4j;

16  /**
17   * A SegmentParser parses segments of the specified type.
18   * 
19   * @author Audacious Inquiry
20   */
21  @Slf4j
22  public abstract class AbstractSegmentParser extends AbstractStructureParser implements Processor<Message, Segment> {
23  	private final Parser<Message, Structure> parser;
24  	@Getter
25  	private final MessageParser messageParser;
26  	/**
27  	 * Construct a segment parser for the specified message parser and segment type
28  	 * @param p	The message parser
29  	 * @param s	The segment type name
30  	 */
31  	protected AbstractSegmentParser(MessageParser p, String s) {
32  		super(p, s);
33  		parser = messageParser = p;
34  	}
35  	
36  	@Override
37  	public boolean isEmpty(Segment segment) { // NOSONAR (Necessary for Inheritance)
38  		return super.isEmpty(segment);
39  	}
40  	
41  	/**
42  	 * Annotation driven parsing.  Call this method to use parsing driven
43  	 * by ComesFrom and Produces annotations in the parser.
44  	 * 
45  	 * This method will be called by MessageParser for each segment of the given type that
46  	 * appears within the method.  It will create the primary resource produced by the
47  	 * parser (by calling the setup() method) and then parses individual fields of the 
48  	 * segment and passes them to parser methods to add them to the primary resource or 
49  	 * to create any extra resources.
50  	 * 
51  	 * @param segment The segment to be parsed
52  	 */
53  	@Override
54  	public IBase parse(Segment segment) {
55  		if (isEmpty(segment)) {
56  			return null;
57  		}
58  
59  		super.segment = segment;
60  		IBaseResource r = setup();
61  		if (r == null) {
62  			// setup() returned nothing, there must be nothing to do
63  			return null;
64  		}
65  		List<FieldHandler> handlers = getFieldHandlers();
66  		for (FieldHandler fieldHandler : handlers) {
67  			fieldHandler.handle(this, segment, r);
68  		}
69  		return r;
70  	}
71  	
72  
73  	@SuppressWarnings("unchecked")
74  	@Override
75  	public Parser<Message, Segment> getParser() {
76  		@SuppressWarnings("rawtypes")
77  		Parser p = parser;
78  		return (Parser<Message, Segment>)p;
79  	}
80  }
