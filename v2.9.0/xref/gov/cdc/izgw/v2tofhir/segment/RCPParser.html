19  /**
20   * The RCP Parser updates the Parameters resource generated by the QPDParser to include the
21   * count of records requested.
22   *
23   * @author Audacious Inquiry
24   *
25   */
26  @Produces(segment="RCP", resource=Parameters.class)
27  @Slf4j
28  public class RCPParser extends AbstractSegmentParser {
29  	private Parameters params;
30  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
31  	static {
32  		log.debug("{} loaded", RCPParser.class.getName());
33  	}
34  	/**
35  	 * Construct a new QPD Parser for the given messageParser.
36  	 * 
37  	 * @param messageParser The messageParser using this QPDParser
38  	 */
39  	public RCPParser(MessageParser messageParser) {
40  		super(messageParser, "RCP");
41  		if (fieldHandlers.isEmpty()) {
42  			FieldHandler.initFieldHandlers(this, fieldHandlers);
43  		}
44  	}
45  	
46  	@Override
47  	public List<FieldHandler> getFieldHandlers() {
48  		return fieldHandlers;
49  	}
50  	
51  	public IBaseResource setup() {
52  		params = getFirstResource(Parameters.class);
53  		return params;
54  	}
55  	
56  	/*
57  		FIELD	LENGTH	DATA TYPE	OPTIONALITY	REPEATABILITY	TABLE	
58  		RCP.1 - Query Priority	1	ID	O	-	0091	
59  		RCP.2 - Quantity Limited Request	0	CQ	O	-		
60  		RCP.3 - Response Modality	0	CNE	O	-	0394	
61  		RCP.4 - Execution And Delivery Time	0	DTM	C	-		
62  		RCP.5 - Modify Indicator	1	ID	O	-	0395	
63  		RCP.6 - Sort-by Field	0	SRT	O	∞		
64  		RCP.7 - Segment Group Inclusion	256	ID	O	∞	0391	 
65  	 */
66  	/**
67  	 * Set the count parameter on the search.
68  	 * Update the search on params and the Bundle.entry.request component with the count.
69  	 * @param quantityLimitedRequest	The count.
70  	 */
71  	@ComesFrom(path = "Parameters._count", field = 2, component = 1, comment = "Quantity Limited Request")
72  	public void setQuantityLimitedRequest(PositiveIntType quantityLimitedRequest) {
73  		params.addParameter().setName("_count").setValue(quantityLimitedRequest);
74  		ParametersParameterComponent search = params.getParameter("_search");
75  		String countQueryParam = "&_count=" + quantityLimitedRequest.getValueAsString();
76  		// Update anyplace the _search was stored with the count.
77  		if (search != null && search.getValue() instanceof StringType s) {
78  			s.setValue(s.getValue() + countQueryParam);
79  		}
80  		BundleEntryRequestComponent req = (BundleEntryRequestComponent) params.getUserData(BundleEntryRequestComponent.class.getName());
81  		if (req != null && req.hasUrl()) {
82  			req.setUrl(req.getUrl() + countQueryParam);
83  		}
84  		BundleEntryResponseComponent resp = (BundleEntryResponseComponent) params.getUserData(BundleEntryResponseComponent.class.getName());
85  		if (resp != null && resp.hasLocation()) {
86  			resp.setLocation(resp.getLocation() + countQueryParam);
87  		}
88  	}
89  }
