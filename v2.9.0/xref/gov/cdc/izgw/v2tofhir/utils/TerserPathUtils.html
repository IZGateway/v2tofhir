12  /**
13   * This utility class contains methods for manipulating and converting between
14   * HAPI V2 Terser and FhirPath paths (as applied over V2 models). 
15   */
16  public class TerserPathUtils {
17  	private static final String TERSER_PATH_SEP = "/";
18  
19  	private TerserPathUtils() {}
20  	/**
21  	 * Convert a FhirPath to a TerserPath
22  	 * @param path	The FhirPath expression to convert
23  	 * @return	A Terser path representation
24  	 */
25  	public static String fhirPathToTerserPath(String path) {
26  		path = TERSER_PATH_SEP + path;
27  		// Replace . with /
28  		path = path.replace(".", TERSER_PATH_SEP);
29  		while (path.contains("[")) {
30  			String before = StringUtils.substringBefore(path, "[").trim();
31  			String inside = StringUtils.substringBetween(path, "[", "]").trim();
32  			String after = StringUtils.substringAfter("]", 0).trim();
33  			path = before + "[" + (StringUtils.isNumeric(inside) ? Integer.parseInt(inside)  + 1 : inside) + "]" + after; 
34  		}
35  		return path;
36  	}
37  
38  	/**
39  	 * Given a structure, get a Terser path for it
40  	 * @param s The structure to get a Terser path for
41  	 * @return The terser path to the specified structure within a message.
42  	 * @throws HL7Exception If a path to the structure cannot be found in its message.
43  	 */
44  	public static String getTerserPath(Structure s) throws HL7Exception {
45  		if (s instanceof Message) {
46  			return "";
47  		}
48  		String myName = s.getName();
49  		int myRep = 1;
50  		Group g = s.getParent();
51  		if (g == null) {
52  			return s.getName();
53  		}
54  		Structure[] children = g.getAll(myName);
55  		for (myRep = 0; myRep < children.length; myRep++) {
56  			String path = getTerserPath(g) + TERSER_PATH_SEP + myName;
57  			if (children[myRep] == s) {
58  				if (myRep == 0) {
59  					return path;
60  				} else {
61  					return path + "(" + (myRep+1) + ")";
62  				}
63  			}
64  		}
65  		if (s instanceof Segment) {
66  			return g.getName() + TERSER_PATH_SEP + myName;
67  		}
68  		throw new HL7Exception("Cannot find " + s.getName() + " in " + g.getName());
69  	}

71  	/**
72  	 * Convert a Terser path to a FhirPath
73  	 * @param path	The Terser path expression to convert
74  	 * @return	A FhirPath representation
75  	 */
76  	public static String terserPathToFhirPath(String path) {
77  		// Remove initial / 
78  		if (path.startsWith(TERSER_PATH_SEP)) {
79  			path = path.substring(1);
80  		}
81  		// Replace / with .
82  		path = path.replace(TERSER_PATH_SEP, ".");
83  		// Replace ( and ) with [ and ]
84  		while (path.contains("(")) {
85  			String before = StringUtils.substringBefore(path, "(").trim();
86  			String inside = StringUtils.substringBetween(path, "(", ")").trim();
87  			String after = StringUtils.substringAfter(")", 0).trim();
88  			path = before + "[" + (StringUtils.isNumeric(inside) ? Integer.parseInt(inside)  - 1 : inside) + "]" + after; 
89  		}
90  		return path;
91  	}

93  	/**
94  	 * Convert a Path in V2 encoding of an ERL data type to a FhirPath
95  	 * @param path	The V2 encoding of an ERL data type.
96  	 * @return	The converted FhirPath
97  	 */
98  	public static String v2ToFHIRPath(String path) {
99  		
100 		String[] parts = path.split("^");
101 		int[] numbers = new int[parts.length];
102 		Type type = null;
103 		StringBuilder fhirPath = new StringBuilder();
104 		for (int i = 0; i < parts.length; i++) {
105 			if (i > 0 && StringUtils.isEmpty(parts[i])) {
106 				parts[i] = "1";
107 				numbers[i] = 1;
108 			} else if (StringUtils.isNotEmpty(parts[i])) {
109 				numbers[i] = StringUtils.isNumeric(parts[i]) ? Integer.parseInt(parts[i]): -1;
110 			} 
111 			updateFhirPath(parts, numbers, type, fhirPath, i);
112 		}
113 		return fhirPath.toString();
114 	}
115 	private static void updateFhirPath(String[] parts, int[] numbers, Type type, StringBuilder fhirPath, int i) {
116 		switch (i) {
117 		case 0: // Segment name
118 			fhirPath.append(parts[0].toUpperCase());
119 			break;
120 		case 1, 3: // Repetition number
121 			if (numbers[i] > 0) {
122 				fhirPath.append("[").append(numbers[i]).append("]");
123 			}
124 			break;
125 		case 2: // Field position
126 			fhirPath.append(".").append(parts[2]);
127 			break;
128 		case 4: // Component or subcomponent number
129 			// Load segment from v2.8.1 structures.  If that fails, load from 2.5.1
130 			type = ParserUtils.getFieldType(parts[0], numbers[2]);
131 			fhirPath.append(type == null ? "UNKNOWN" : type.getName()).append(numbers[i]);
132 			break;
133 		case 5:
134 			type = ParserUtils.getComponent(type, numbers[i]);
135 			fhirPath.append(type == null ? "UNKNOWN" : type.getName()).append(numbers[i]);
136 			break;
137 		default:
138 			break;
139 		}
140 	}
