91  	 * Create a new Query for an Immunization or ImmunizationRecommendation
92  	 * @param qpd	The QPD to create
93  	 * @throws HL7Exception If a field cannot be read from the QPD segment
94  	 */
95  	public IzQuery(QPD qpd) throws HL7Exception {
96  		this.qpd = qpd;
97  		name = QBPUtils.getField(qpd, 4, XPN.class);
98  		maiden = QBPUtils.getField(qpd, 5, XPN.class);
99  		birthDate = QBPUtils.getField(qpd, 6, TS.class);
100 		gender = QBPUtils.getField(qpd, 7, IS.class);
101 		address = QBPUtils.getField(qpd, 8, XAD.class);
102 		telephone = QBPUtils.getField(qpd, 9, XTN.class).getTelephoneNumber();
103 	}

105 	/**
106 	 * Add a given search parameter to the QPD segment.
107 	 * @param fhirParamName	The FHIR parameter name
108 	 * @param params	The list of parameters.
109 	 * @throws HL7Exception	If an error occurs.
110 	 */
111 	public void addParameter(	// NOSONAR: Giant switch is acceptable
112 		String fhirParamName, List<String> params
113 	) throws HL7Exception {
114 		if (fhirParamName == null || fhirParamName.isEmpty()) {
115 			return;
116 		}
117 		for (String param: params) {
118 			if (fhirParamName.length() > 0 && fhirParamName.charAt(0) == '_' && !COUNT.equals(fhirParamName)) {
119 				// Ignore FHIR special search parameters.
120 				continue;
121 			}
122 			switch (fhirParamName) {
123 			case COUNT:
124 				setCount(param);
125 				break;
126 			
127 			case Immunization.SP_PATIENT:
128 				ReferenceParam refParam = new ReferenceParam();
129 				refParam.setValueAsQueryToken(null, Immunization.SP_PATIENT, null, param);
130 				String ident = FhirIdCodec.decode(refParam.getIdPart());
131 				setPatientList(fhirParamName, ident);
132 				break;
133 			case PATIENT_LIST: //	QPD-3	PatientList (can repeat)
134 				setPatientList(fhirParamName, param);
135 				break;
136 				
137 			case PATIENT_NAME_FAMILY: //	QPD-4.1	PatientName
138 				setPatientFamilyName(param);
139 				break;
140 			case PATIENT_NAME_GIVEN: //	QPD-4.2	and QPD-4.3	PatientName
141 				setPatientGivenName(param);
142 				break;
143 			case PATIENT_NAME_SUFFIX: //	QPD-4.4	PatientName
144 				setPatientNameSuffix(param);
145 				break;
146 
147 			case PATIENT_MOTHERS_MAIDEN_NAME: //	QPD-5	PatientMotherMaidenName
148 				setMothersMaidenName(param);
149 				break;
150 			// Given and Suffix won't be used in V2 searches on mother's maiden name, but it allows
151 			// the v2tofhir components to round trip.
152 			case PATIENT_MOTHERS_MAIDEN_NAME + "-given":
153 				setMothersMaidenGivenName(param);
154 				break;
155 			case PATIENT_MOTHERS_MAIDEN_NAME + "-suffix":
156 				setMothersMaidenNameSuffix(param);
157 				break;
158 			case PATIENT_BIRTHDATE: //	QPD-6	PatientDateOfBirth
159 				setPatientBirthDate(fhirParamName, param);
160 				break;
161 				
162 			case PATIENT_GENDER: //	QPD-7	PatientSex
163 				setPatientGender(fhirParamName, param);
164 				break;
165 			
166 			case PATIENT_ADDRESS: //	QPD-8-1 and QPD-8-2	PatientAddress
167 				setPatientAddress(param);
168 				break;
169 			case PATIENT_ADDRESS_CITY: //	QPD-8-3	PatientAddress
170 				setPatientSuffixCity(param);
171 				break;
172 			case PATIENT_ADDRESS_STATE: //	QPD-8-4	PatientAddress
173 				setPatientAddressState(param);
174 				break;
175 			case PATIENT_ADDRESS_POSTAL: //	QPD-8-5	PatientAddress
176 				setPatientAddressPostal(param);
177 				break;
178 			case PATIENT_ADDRESS_COUNTRY: //	QPD-8-5	PatientAddress
179 				setPatientAddressCountry(param);
180 				break;
181 			
182 			case PATIENT_HOMEPHONE: //	QPD-9	PatientHomePhone
183 				setPatientPhone(param);
184 				break;
185 			
186 			case PATIENT_MULTIPLE_BIRTH_INDICATOR: //	QPD-10	PatientMultipleBirthIndicator
187 				setMultipleBirthIndicator(param);
188 				break;
189 			
190 			case PATIENT_MULTIPLE_BIRTH_ORDER: //	QPD-11	PatientBirthOrder
191 				setMultipleBirthOrder(param);
192 				break;
193 			
194 			default:
195 				throw new IllegalArgumentException("Invalid query parameter: " + fhirParamName);
196 			}
197 		}
198 	}

200 	/**
201 	 * Finalize some the parameters and then check for a valid query
202 	 * @throws HL7Exception Should an error occur during parsing of HL7 content
203 	 */
204 	public void update() throws HL7Exception {
205 		if (!getName().isEmpty()) {
206 			getName().getNameTypeCode().setValue("L");
207 		}
208 		if (!getMaiden().isEmpty()) {
209 			getMaiden().getNameTypeCode().setValue("M");
210 		}
211 		if (!getAddress().isEmpty()) {
212 			getAddress().getAddressType().setValue("L");
213 		}
214 		RCP rcp = (RCP) getQpd().getMessage().get("RCP");
215 		rcp.getQuantityLimitedRequest().getQuantity().setValue(Integer.toString(getCount()));
216 		rcp.getQuantityLimitedRequest().getUnits().getIdentifier().setValue("RD");
217 		rcp.getQuantityLimitedRequest().getUnits().getText().setValue("records");
218 		rcp.getQuantityLimitedRequest().getUnits().getNameOfCodingSystem().setValue("HL70126");
219 		
220 		if (!isHasId() && 
221 			(getName().isEmpty() || 
222 			 getBirthDate().isEmpty()) ) {
223 			throw new IllegalArgumentException("A query must contain either a "
224 					+ "patient.identifier or the patient name and birthDate");
225 		}
226 	}

228 	private void setCount(String param) {
229 		NumberParam number = new NumberParam();
230 		number.setValueAsQueryToken(QBPUtils.R4, COUNT, null, param);
231 		setCount(number.getValue().intValue());
232 		if (getCount() > 10) {
233 			throw new IllegalArgumentException("_count must be <= 10");
234 		} else if (getCount() <= 0) {
235 			throw new IllegalArgumentException("_count must be > 0");
236 		}
237 	}
238 	private void setCount(int count) {
239 		this.count = count;
240 	}

242 	private void setMothersMaidenGivenName(String param) throws HL7Exception {
243 		if (getMaiden().getGivenName().isEmpty()) {
244 			getMaiden().getGivenName().setValue(param);
245 		} else if (getMaiden().getSecondAndFurtherGivenNamesOrInitialsThereof().isEmpty()) {
246 			getMaiden().getSecondAndFurtherGivenNamesOrInitialsThereof().setValue(param);
247 		} else {
248 			throw new IllegalArgumentException("Cannot repeat query parameter more than one: " + param + "-given");
249 		}
250 	}

253 		if (getMaiden().getFamilyName().getSurname().isEmpty()) {
254 			getMaiden().getFamilyName().getSurname().setValue(param);
255 			getMaiden().getNameTypeCode().setValue("M");

258 		}

268 	private void setMultipleBirthIndicator(String param) throws HL7Exception{
269 		TokenParam token;
270 		token = new TokenParam();
271 		token.setValueAsQueryToken(QBPUtils.R4, PATIENT_MULTIPLE_BIRTH_INDICATOR, null, param);
272 		String mbi = token.getValueNotNull();
273 		ID mbid = QBPUtils.getField(qpd, 10, ID.class);
274 		if (!mbid.isEmpty()) {
275 			throw new IllegalArgumentException(IzQuery.CANNOT_REPEAT + param);
276 		}
277 		if (mbi.length() != 0) {
278 			switch (mbi.toLowerCase().charAt(0)) {
279 			case 'y', 't':
280 				mbid.setValue("Y"); break;
281 			case 'n', 'f':
282 				mbid.setValue("N"); break;
283 			default:
284 				break;
285 			}
286 		}
287 	}

347 	private void setPatientGender(String fhirParamName, String param) throws HL7Exception{
348 		TokenParam token;
349 		if (!getGender().isEmpty()) {
350 			throw new IllegalArgumentException(IzQuery.CANNOT_REPEAT + param);
351 		}
352 		token = new TokenParam();
353 		token.setValueAsQueryToken(QBPUtils.R4, fhirParamName, null, param);
354 		if (token.getValueNotNull().length() == 0) {
355 			token.setValue("U");	// Per CDC Guide
356 		}
357 		// HL7 codes are same as first character of FHIR codes.
358 		getGender().setValue(token.getValue().substring(0, 1).toUpperCase());
359 	}

371 	private void setPatientList(String fhirParamName, String param) throws HL7Exception{
372 		TokenParam token;
373 		token = new TokenParam();
374 		token.setValueAsQueryToken(QBPUtils.R4, fhirParamName, null, param);
375 
376 		hasId = true;
377 		// get QPD-3
378 		CX cx = QBPUtils.addRepetition(qpd, 3, CX.class);
379 		cx.getIDNumber().setValue(token.getValue());
380 		cx.getAssigningAuthority().getNamespaceID().setValue(token.getSystem());
381 		cx.getIdentifierTypeCode().setValue("MR");  // Per CDC Guide
382 	}
