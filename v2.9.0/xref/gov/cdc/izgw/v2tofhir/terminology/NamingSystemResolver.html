1   package gov.cdc.izgw.v2tofhir.terminology;

3   import java.util.Optional;
4   
5   import org.hl7.fhir.r4.model.NamingSystem;
6   
7   /**
8    * Resolves OIDs to canonical URIs and vice-versa, and provides URI lookups for HL7 V2 tables.
9    *
10   * <p>All default implementations return empty results; override to provide real resolution.</p>
11   *
12   * @author Audacious Inquiry
13   * @see TerminologyMapper
14   */
15  public interface NamingSystemResolver {
16  
17      /**
18       * Returns the canonical URI for the given OID.
19       *
20       * @param oid the OID to look up (e.g., {@code "2.16.840.1.113883.6.1"})
21       * @return the canonical URI, or {@link Optional#empty()} if not found
22       */
23      default Optional<String> toUri(String oid) {
24          return Optional.empty();
25      }
26  
27      /**
28       * Returns the OID for the given canonical URI.
29       *
30       * @param uri the URI to look up (e.g., {@code "http://loinc.org"})
31       * @return the OID, or {@link Optional#empty()} if not found
32       */
33      default Optional<String> toOid(String uri) {
34          return Optional.empty();
35      }
36  
37      /**
38       * Returns the full {@link NamingSystem} resource for the given OID or URI.
39       *
40       * @param oidOrUri the OID or URI to look up
41       * @return the {@link NamingSystem}, or {@link Optional#empty()} if not found
42       */
43      default Optional<NamingSystem> getNamingSystem(String oidOrUri) {
44          return Optional.empty();
45      }
46  
47      /**
48       * Returns the canonical FHIR URI for an HL7 V2 table.
49       *
50       * <p>Input may be a bare number ({@code "0001"}), a prefixed table name ({@code "HL70001"}),
51       * or the full URI (returned unchanged).
52       * Example: {@code "0001"} → {@code "http://terminology.hl7.org/CodeSystem/v2-0001"}</p>
53       *
54       * @param tableNameOrNumber the V2 table name or number
55       * @return the canonical FHIR CodeSystem URI for the table
56       */
57      default String v2TableUri(String tableNameOrNumber) {
58          if (tableNameOrNumber == null) {
59              return null;
60          }
61          if (tableNameOrNumber.startsWith("http://") || tableNameOrNumber.startsWith("urn:")) {
62              return tableNameOrNumber;
63          }
64          String number = tableNameOrNumber.replaceFirst("(?i)^HL7", "");
65          return "http://terminology.hl7.org/CodeSystem/v2-" + number;
66      }
67  
68      /**
69       * Returns the preferred system URI for an identifier's assigning authority or system name.
70       *
71       * <p>Converts V2 assigning authority names and OID-based system identifiers to their
72       * preferred FHIR Identifier system URIs. Returns the input unchanged if no mapping is known.</p>
73       *
74       * @param system the raw system string from a V2 identifier field
75       * @return the preferred system URI; never {@code null} if input is non-null
76       */
77      default String normalizeSystem(String system) {
78          return system;
79      }
80  }
