3   /**
4    * Composite terminology service interface extending all eight focused sub-interfaces.
5    *
6    * <p>Implementations provide a unified point of access for all terminology operations.
7    * Callers that only need a subset of capabilities should accept the narrowest applicable
8    * sub-interface rather than {@code TerminologyMapper} directly.</p>
9    *
10   * <p>Sub-interfaces and their responsibilities:</p>
11   * <ul>
12   *   <li>{@link NamingSystemResolver} — OID/URI bidirectional lookup and V2 table URI resolution</li>
13   *   <li>{@link ConceptTranslator} — concept map translation by target system or named mapping</li>
14   *   <li>{@link DisplayNameResolver} — preferred display text for coded concepts</li>
15   *   <li>{@link CodeValidator} — value set membership validation</li>
16   *   <li>{@link TerminologyLoader} — load and retrieve FHIR terminology resources</li>
17   *   <li>{@link UnitMapper} — V2 unit string to UCUM coding conversion</li>
18   *   <li>{@link DemographicsMapper} — race and ethnicity to US Core extension mapping</li>
19   *   <li>{@link GeoMapper} — ISO 3166 country code resolution and normalization</li>
20   * </ul>
21   *
22   * <p>All sub-interface methods provide {@code default} implementations so partial implementors
23   * are not required to stub unrelated operations. See each sub-interface for default behaviour.</p>
24   *
25   * @author Audacious Inquiry
26   * @see NamingSystemResolver
27   * @see ConceptTranslator
28   * @see DisplayNameResolver
29   * @see CodeValidator
30   * @see TerminologyLoader
31   * @see UnitMapper
32   * @see DemographicsMapper
33   * @see GeoMapper
34   */
35  public interface TerminologyMapper extends
36          NamingSystemResolver,
37          ConceptTranslator,
38          DisplayNameResolver,
39          CodeValidator,
40          TerminologyLoader,
41          UnitMapper,
42          DemographicsMapper,
43          GeoMapper {
44  
45      /**
46       * Resolves the diamond default: both {@link NamingSystemResolver} and
47       * {@link TerminologyLoader} declare {@code getNamingSystem(String)}.
48       *
49       * <p>Implementations should check both the OID/URI registry and the loaded-resource
50       * registry and return whichever matches. The default returns empty.</p>
51       *
52       * @param oidOrUriOrId an OID, canonical URI, or resource id
53       * @return the matching {@link org.hl7.fhir.r4.model.NamingSystem}, or empty if not found
54       */
55      @Override
56      default java.util.Optional<org.hl7.fhir.r4.model.NamingSystem> getNamingSystem(String oidOrUriOrId) {
57          return java.util.Optional.empty();
58      }
59  }
