1   package gov.cdc.izgw.v2tofhir.terminology;
2   
3   import org.hl7.fhir.r4.model.Coding;
4   
5   /**
6    * Immutable value object carrying the result of a concept translation.
7    *
8    * <p>Use {@link #exact(Coding, Coding, String)} when the mapping is an exact code match,
9    * or {@link #approximate(Coding, Coding, String)} when the match is equivalent or approximate.</p>
10   *
11   * @param sourceCoding  the coding that was translated
12   * @param resultCoding  the translated coding (system + code + display)
13   * @param mappingSource the name of the concept map or mapping table used
14   * @param exact         {@code true} if the match was exact; {@code false} if approximate
15   *
16   * @author Audacious Inquiry
17   */
18  public record TranslationResult(
19          Coding sourceCoding,
20          Coding resultCoding,
21          String mappingSource,
22          boolean exact) {
23  
24      /**
25       * Creates a {@code TranslationResult} representing an exact code match.
26       *
27       * @param sourceCoding  the coding that was translated
28       * @param resultCoding  the translated coding
29       * @param mappingSource the name of the concept map or mapping table used
30       * @return a new {@code TranslationResult} with {@code exact == true}
31       */
32      public static TranslationResult exact(Coding sourceCoding, Coding resultCoding, String mappingSource) {
33          return new TranslationResult(sourceCoding, resultCoding, mappingSource, true);
34      }
35  
36      /**
37       * Creates a {@code TranslationResult} representing an approximate or equivalent match.
38       *
39       * @param sourceCoding  the coding that was translated
40       * @param resultCoding  the translated coding
41       * @param mappingSource the name of the concept map or mapping table used
42       * @return a new {@code TranslationResult} with {@code exact == false}
43       */
44      public static TranslationResult approximate(Coding sourceCoding, Coding resultCoding, String mappingSource) {
45          return new TranslationResult(sourceCoding, resultCoding, mappingSource, false);
46      }
47  
48      /**
49       * Returns {@code true} if this result represents an exact code match.
50       *
51       * @return {@code true} for exact match; {@code false} for approximate
52       */
53      public boolean isExact() {
54          return exact;
55      }
56  }
