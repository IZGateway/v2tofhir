20  /**
21   * Default implementation of {@link TerminologyMapper} that delegates every method to the
22   * existing static utility classes ({@link Systems}, {@link Mapping}, {@link Units},
23   * {@link ISO3166}, {@link RaceAndEthnicity}).
24   *
25   * <p>The result of every call through {@code DefaultTerminologyMapper} is identical to calling
26   * the equivalent static method directly. This class acts as the bridge between the new
27   * injectable {@link TerminologyMapper} interface and the legacy static utilities.</p>
28   *
29   * <p>Dynamically loaded resources (via {@link #load(Resource)}) are stored in per-instance
30   * maps and do not affect the shared static caches.</p>
31   *
32   * @author Audacious Inquiry
33   * @see TerminologyMapper
34   * @see TerminologyMapperFactory
35   */
36  @Slf4j
37  public class DefaultTerminologyMapper implements TerminologyMapper {
38  
39      /** Create a mapper that uses the built-in terminology tables. */
40      public DefaultTerminologyMapper() {
41          // Nothing to initialize
42      }

45      private final Map<String, CodeSystem> loadedCodeSystems = new LinkedHashMap<>();
46      private final Map<String, ValueSet> loadedValueSets = new LinkedHashMap<>();

79      /**
80       * {@inheritDoc}
81       * <p>Delegates to {@link Mapping#mapTableNameToSystem(String)}, which resolves V2 table
82       * names, numeric table numbers, OIDs, and aliased URIs to their canonical FHIR system
83       * URI. Already-canonical URIs are returned unchanged. Returns {@code null} for blank

85       */

88          return Mapping.mapTableNameToSystem(tableNameOrNumber);

91      /**
92       * {@inheritDoc}
93       *
94       * <p>Delegates to {@link Mapping#mapSystem(org.hl7.fhir.r4.model.Identifier)} via a
95       * temporary {@link org.hl7.fhir.r4.model.Identifier}, which resolves V2 assigning
96       * authority names and OID-based system identifiers to their preferred FHIR Identifier
97       * system URIs. Returns the input unchanged if no mapping is known.</p>
98       */
99      @Override
100     public String normalizeSystem(String system) {
101         if (system == null) {
102             return null;
103         }
104         org.hl7.fhir.r4.model.Identifier temp = new org.hl7.fhir.r4.model.Identifier().setSystem(system);
105         Mapping.mapSystem(temp);
106         return temp.getSystem();
107     }

113     /**
114      * {@inheritDoc}
115      *
116      * <p>Delegates to {@link Mapping#getMapping(String)} then {@link Mapping#mapCode(Coding, boolean)}.
117      * Returns {@link Optional#empty()} if the mapping name is unknown or no mapping exists
118      * for the source code.</p>
119      */
120     @Override
121     public Optional<TranslationResult> mapCode(String mappingName, Coding source) {
122         Mapping m = Mapping.getMapping(mappingName);
123         if (m == null) {
124             return Optional.empty();
125         }
126         Coding result = m.mapCode(source, true);
127         return Optional.ofNullable(result)
128                 .map(c -> TranslationResult.exact(source, c, mappingName));
129     }

207     /**
208      * {@inheritDoc}
209      *
210      * <p>Constructs a {@link CodeableConcept} from the supplied code and system, delegates
211      * to {@link RaceAndEthnicity#setRaceCode(CodeableConcept, Extension)}, and returns the
212      * populated extension. Returns {@link Optional#empty()} if the code is not recognised.</p>
213      */
214     @Override
215     public Optional<Extension> mapRace(String code, String codeSystem) {
216         CodeableConcept race = new CodeableConcept().addCoding(new Coding(codeSystem, code, null));
217         Extension ext = new Extension().setUrl(RaceAndEthnicity.US_CORE_RACE);
218         RaceAndEthnicity.setRaceCode(race, ext);
219         return ext.hasExtension() ? Optional.of(ext) : Optional.empty();
220     }

241     /**
242      * {@inheritDoc}
243      *
244      * <p>Delegates to {@link ISO3166#twoLetterCode(String)} and {@link ISO3166#name(String)}.
245      * Returns {@link Optional#empty()} if the code is not recognised.</p>
246      */
247     @Override
248     public Optional<Coding> lookupCountry(String code, String codeSystem) {
249         String name = ISO3166.name(code);
250         if (name == null) {
251             return Optional.empty();
252         }
253         String alpha2 = ISO3166.twoLetterCode(code);
254         return Optional.of(new Coding(codeSystem, alpha2 != null ? alpha2 : code, name));
255     }

270     /**
271      * {@inheritDoc}
272      *
273      * <p>Supported resource types and their handling:</p>
274      * <ul>
275      *   <li>{@link NamingSystem} — registered in the shared {@link Systems} naming system registry</li>
276      *   <li>{@link ConceptMap} — stored in a per-instance map; retrievable via {@link #getConceptMap(String)}</li>
277      *   <li>{@link CodeSystem} — stored in a per-instance map; retrievable via {@link #getCodeSystem(String)}</li>
278      *   <li>{@link ValueSet} — stored in a per-instance map; retrievable via {@link #getValueSet(String)}</li>
279      * </ul>
280      *
281      * @throws TerminologyException if {@code resource} is {@code null} or an unsupported type
282      */
283     @Override
284     public void load(Resource resource) {
285         if (resource == null) {
286             throw new TerminologyException("Cannot load a null resource");
287         }
288         switch (resource) {
289         case NamingSystem ns -> {
290             List<String> identifiers = new ArrayList<>();
291             ns.getUniqueId().forEach(uid -> identifiers.add(uid.getValue()));
292             if (!identifiers.isEmpty()) {
293                 Systems.addNamingSystem(ns.getName(), identifiers.toArray(new String[0]));
294             } else {
295                 log.warn("NamingSystem '{}' has no unique identifiers; skipping registration", ns.getName());
296             }
297         }
298         case ConceptMap cm -> loadedConceptMaps.put(cm.getIdElement().getIdPart(), cm);
299         case CodeSystem cs -> loadedCodeSystems.put(cs.getIdElement().getIdPart(), cs);
300         case ValueSet vs -> loadedValueSets.put(vs.getIdElement().getIdPart(), vs);
301         default -> throw new TerminologyException(
302 				"Unsupported resource type for load(): " + resource.getResourceType());
303         }
304     }

306     /**
307      * {@inheritDoc}
308      *
309      * <p>Checks the per-instance loaded map first, then falls back to the
310      * {@link Units#CONCEPT_MAP_ID} resource owned by {@link Units} (if the id matches),
311      * and finally to the static {@link Mapping} registry via {@link Mapping#getMapping(String)}
312      * and {@link Mapping#asConceptMap()}.</p>
313      */
314     @Override
315     public Optional<ConceptMap> getConceptMap(String id) {
316         ConceptMap cm = loadedConceptMaps.get(id);
317         if (cm != null) {
318             return Optional.of(cm);
319         }
320         if (Units.CONCEPT_MAP_ID.equals(id)) {
321             return Units.getConceptMap();
322         }
323         return Optional.ofNullable(Mapping.getMapping(id))
324                 .map(Mapping::asConceptMap);
325     }
326 
327     /**
328      * {@inheritDoc}
329      *
330      * <p>Checks the per-instance loaded map first, then attempts to retrieve via the
331      * {@link NamingSystem} user-data cache in {@link Systems}.</p>
332      */
333     @Override
334     public Optional<CodeSystem> getCodeSystem(String id) {
335         CodeSystem cs = loadedCodeSystems.get(id);
336         if (cs != null) {
337             return Optional.of(cs);
338         }
339         NamingSystem ns = Systems.getNamingSystem(id);
340         if (ns != null) {
341             Object userData = ns.getUserData(CodeSystem.class.getName());
342             if (userData instanceof CodeSystem cachedCs) {
343                 return Optional.of(cachedCs);
344             }
345         }
346         return Optional.empty();
347     }

349     /**
350      * {@inheritDoc}
351      *
352      * <p>Checks the per-instance loaded map first, then falls back to the
353      * {@link Units#VALUE_SET_ID} resource owned by {@link Units} (if the id matches).</p>
354      */
355     @Override
356     public Optional<ValueSet> getValueSet(String id) {
357         ValueSet vs = loadedValueSets.get(id);
358         if (vs != null) {
359             return Optional.of(vs);
360         }
361         if (Units.VALUE_SET_ID.equals(id)) {
362             return Units.getValueSet();
363         }
364         return Optional.empty();
365     }
