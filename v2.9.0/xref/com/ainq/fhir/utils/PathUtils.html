23  /**
24   * Parse a path from a FHIR Object and return the object.
25   * 
26   * Paths in this class use the Restricted FHIRPath syntax with a few minor simplifications:
27   * 
28   * extension('extension-url') can be written without the full url when the extension is
29   * a FHIR standard extension (comes from  http://hl7.org/fhir/StructureDefinition/), or from
30   * FHIR US Core (http://hl7.org/fhir/us/core/StructureDefinition/). Also, the quotes in the 
31   * extension name aren't necessary.
32   * Also, one can use the extension short name directly:
33   * 	Patient.us-core-race works as a path expression.
34   *  
35   * @author Audacious Inquiry
36   * @see <a href="https://hl7.org/fhir/r4/fhirpath.html#simple">FHIRPath Restricted Subset</a>
37   */
38  @Slf4j
39  public class PathUtils {
40  	/** The extension prefix for standard FHIR extensions */
41  	public static final String FHIR_EXT_PREFIX = "http://hl7.org/fhir/StructureDefinition/";
42  	/** The extension prefix for standard US Core extensions */
43  	public static final String US_CORE_EXT_PREFIX = "http://hl7.org/fhir/us/core/StructureDefinition/";

46  		log.debug("{} loaded", PathUtils.class.getName());

48  	/**
49  	 * Controls behavior for get method, enabling new objects to be created.
50  	 */
51  	public enum Action {
52  		/** Get any existing property at the path, don't create intermediates */
53  		GET_FIRST,
54  		/** Get the last property at the path */
55  		GET_LAST,
56  		/** Create the property at the path if it doesn't exist (implies GET_LAST at intermediate stages) */
57  		CREATE_IF_MISSING
58  	}
59  	
60  	/**
61  	 * Get the values at the specified restricted FHIRPath 
62  	 * @param b	The base object to get the 
63  	 * @param path The FHIRPath
64  	 * @return	The values
65  	 */
66  	public static  IBase get(IBase b, String path) {
67  		return get(b, path, Action.GET_LAST);
68  	}

70  	static final String[] FHIR_TYPES = {			
71  			/* Primitives */
72  			"Boolean", "Code", "Id", "Instant", "Integer", "Date", "DateTime", "Decimal", 
73  			"PositiveInt", "String", "Time", "UnsignedInt", "Uri", "Url",
74  			/* General Purpose */
75  			"Address", "Age", "Annotation", "Attachment", "Coding", "CodeableConcept", "ContactPoint",
76  			"Count", "Distance", "Duration", "Identifier", "Money", "MoneyQuantity", "Period", 
77  			"Quantity", "Range", "Ratio", "RatioRange", "Signature", "SimpleQuantity", "Timing",
78  			/* Special Cases */
79  			"Expression", "Reference"
80  		};

82  	/** 
83  	 * Endings on a property name that refine it to a specific type.  These names
84  	 * are generally the name of a FHIR Type.
85  	 * 
86  	 * NOTE: .*? must use a reluctant qualifier to find the longest value for
87  	 * type because DateTime ends with Time
88  	 */
89  	private static final Pattern VARIES_PATTERN = Pattern.compile(
90  		"^.*?(" + StringUtils.joinWith("|", (Object[])FHIR_TYPES) + ")$"
91  	);
92  	/**
93  	 * Exceptions to the above pattern.  These are few.  So far only
94  	 * the following are false matches: 
95  	 * Patient.birthDate, 
96  	 * Immunization.doseQuantity,
97  	 * Immunization.expirationDate,
98  	 * ServiceRequest.locationCode,
99  	 * Immunization.reasonCode 
100 	 * Observation.referenceRange, 
101 	 * Immunization.vaccineCode 
102 	 */
103 	private static final Set<String> VARIES_EXCEPTIONS = 
104 		new LinkedHashSet<>(Arrays.asList(
105 			"birthDate", "doseQuantity", "expirationDate",
106 			"locationCode", "reasonCode", "referenceRange", "vaccineCode")
107 		);
108 	/**
109 	 * Get all elements at the specified path.
110 	 * @param b	The base
111 	 * @param path	The path
112 	 * @return	The list of elements at that location.
113 	 */
114 	public static List<IBase> getAll(IBase b, String path) {
115 		
116 		List<IBase> l = null;
117 		
118 		// Split a restricted FHIRPath into its constituent parts.
119 		// FUTURE: Handle cases like extension("http://hl7.org/ ...
120 		String propertyName = StringUtils.substringBefore(path, ".");
121 		String rest = StringUtils.substringAfter(path, ".");
122 		do {
123 			// Pass a reference to propertyName for functions which need to adjust it.
124 			String[] theName = { propertyName };
125 			int index = getIndex(theName);
126 			
127 			String extension = getExtension(theName);
128 			String type = getType(b, theName);
129 			
130 			try {
131 				l = getValues(b, theName[0], extension);
132 			} catch (Exception ex) {
133 				log.error("{} extracting {}.{}",
134 					ex.getClass().getSimpleName(),
135 					b.fhirType(),
136 					theName[0] + (StringUtils.isNotEmpty(extension) ? "." + extension : ""),
137 					ex
138 				);
139 				throw ex;
140 			}
141 			filterListByType(type, l);
142 			
143 			adjustListForIndex(l, index, null);
144 			
145 			propertyName = StringUtils.substringBefore(rest, ".");
146 			rest = StringUtils.substringAfter(rest, ".");
147 			
148 			if (l.isEmpty()) {
149 				return Collections.emptyList();
150 			}
151 			b = l.get(l.size() - 1); 
152 		} while (StringUtils.isNotEmpty(propertyName));
153 		
154 		return l;
155 	}

157 	/**
158 	 * Get the values at the specified restricted FHIRPath 
159 	 * @param b		The context for the path
160 	 * @param path	The path
161 	 * @param action Which value to get or add
162 	 * @return	The requested class.
163 	 * @throws FHIRException when the property does not exist
164 	 * @throws NumberFormatException when an index is not valid (&lt; 0 or not a valid number)
165 	 */
166 	public static IBase get(IBase b, String path, Action action) {
167 		if (action == null) {
168 			action = Action.GET_LAST;
169 		}
170 		
171 		// Split a restricted FHIRPath into its constituent parts.
172 		// FUTURE: Handle cases like extension("http://hl7.org/ ...
173 		String propertyName = StringUtils.substringBefore(path, ".");
174 		String rest = StringUtils.substringAfter(path, ".");
175 		IBase result = b;
176 		do {
177 			// Pass a reference to propertyName for functions which need to adjust it.
178 			String[] theName = { propertyName };
179 			int index = getIndex(theName);
180 			
181 			String extension = getExtension(theName);
182 			String type = getType(b, theName);
183 			
184 			List<IBase> l = getValues(result, theName[0], extension);
185 	
186 			filterListByType(type, l);
187 			
188 			IBase result2 = result;
189 			Supplier<IBase> adder = () -> Property.makeProperty(result2, theName[0]);
190 			if (theName[0].length() == 0 && type != null) {
191 				// When type is non-null and the property name is empty
192 				// we just need to create a new object of the specified type.
193 				// NOTE: This object won't be attached to ANYTHING b/c we don't
194 				// know what list to attach it to.
195 				adder = () -> createElement(result2, type);
196 			}
197 				
198 			adjustListForIndex(l, index, 
199 					Action.CREATE_IF_MISSING.equals(action) ? adder : null
200 				);
201 			propertyName = StringUtils.substringBefore(rest, ".");
202 			rest = StringUtils.substringAfter(rest, ".");
203 			
204 			if (l.isEmpty()) {
205 				return null;
206 			}
207 			if (Action.CREATE_IF_MISSING.equals(action) || Action.GET_FIRST.equals(action)) {
208 				result = l.get(0);
209 			} else { // Action == GET_LAST
210 				result = l.get(l.size() - 1); 
211 			}
212 		} while (StringUtils.isNotEmpty(propertyName));
213 		
214 		return result;
215 	}

217 	private static IBase createElement(IBase b, String type) {
218 		@SuppressWarnings("unchecked")
219 		Class<IBase> clazz = (Class<IBase>) b.getClass();
220 		String theClass = clazz.getPackageName() + "." + type;
221 		try {
222 			@SuppressWarnings("unchecked")
223 			Class<IBase> class2 = (Class<IBase>)clazz.getClassLoader().loadClass(theClass);
224 			return class2.getDeclaredConstructor().newInstance();
225 		} catch (ClassNotFoundException | InstantiationException | IllegalAccessException | IllegalArgumentException | InvocationTargetException
226 				| NoSuchMethodException | SecurityException e) {
227 			String msg = "Cannot create " + theClass;
228 			throw new ServiceConfigurationError(msg, e);
229 		} 
230 	}

232 	private static List<IBase> getValues(IBase b, String propertyName, String extension) {
233 		List<IBase> l = new ArrayList<>();
234 		if (extension != null && b instanceof IBaseHasExtensions elem) {
235 			for (IBaseExtension<?, ?> ext : elem.getExtension()) {
236 				if (extension.equals(ext.getUrl())) {
237 					l.add(ext);
238 				}
239 			}
240 		} else if (propertyName.length() != 0) {
241 			l = Property.getValues(b, propertyName);
242 		} else {
243 			// If propertName.length is zero, it's a type match on b
244 			l.add(b);
245 		}
246 		return l;
247 	}

249 	private static IBase adjustListForIndex(List<IBase> l, int index, Supplier<IBase> adder) {
250 		IBase result = null;
251 		if (index < 0) {
252 			// There was no index
253 			if (adder != null && l.isEmpty()) {
254 				// a value should be created, but there are none, so create one
255 				result = adder.get();
256 				l.add(result);
257 				return result;
258 			}
259 			return null;
260 		}
261 
262 
263 		// Item exists
264 		if (index < l.size()) {
265 			// Get the index-th element.
266 			result = l.get(index);
267 		} 
268 		
269 		if (adder != null) {
270 			// Create the index-th element.
271 			while (index >= l.size()) {
272 				result = adder.get();
273 				l.add(result);
274 			}
275 		}
276 		
277 		// Adjust the list to contain ONLY the indexed item
278 		l.clear();
279 		if (result != null) {
280 			l.add(result);
281 		}
282 		return result;
283 	}

285 	private static void filterListByType(String type, List<IBase> l) {
286 		if (type != null) {
287 			Iterator<IBase> it = l.iterator();
288 			while (it.hasNext()) {
289 				IBase b = it.next();
290 				if (!isOfType(b, type)) {
291 					it.remove();
292 				}
293 			}
294 		}
295 	}

297 	private static List<String> bases = Arrays.asList("DomainResource", "Resource", "Base", "Object");
298 	private static boolean isOfType(IBase b, String type) {
299 		Class<?> theClass = b.getClass();
300 		do {
301 			String className = theClass.getSimpleName();
302 			if (className.equals(type) || className.equals(type + "Type")) {
303 				return true;
304 			}
305 			theClass = theClass.getSuperclass();
306 		} while (!bases.contains(theClass.getSimpleName()));
307 
308 		return false;
309 	}

311 	/**
312 	 * Get information about a Property of a FHIR resource or element.
313 	 * @param clazz	The class of the resource or element to get a property for.
314 	 * @param originalPath	The path to the property
315 	 * @return	The property
316 	 */
317 	public static Property getProperty(Class<? extends IBase> clazz, String originalPath) {
318 		String path = originalPath;
319 		String propertyName = StringUtils.substringAfterLast(originalPath, ".");
320 		if (propertyName == null || propertyName.length() == 0) {
321 			throw new IllegalArgumentException("Path invalid " + originalPath);
322 		}
323 		path = path.substring(0, path.length() - propertyName.length() - 1);
324 		IBase r = null;
325 		IBase base = null;
326 		try {
327 			r = clazz.getDeclaredConstructor().newInstance();
328 			base = get(r, path, Action.CREATE_IF_MISSING);
329 			return Property.getProperty(base, propertyName);
330 		} catch (Exception e) {
331 			throw new IllegalArgumentException("Error accessing propery " + originalPath, e);
332 		}
333 	}

335 	private static int getIndex(String[] propertyName) {
336 		String index = StringUtils.substringBetween(propertyName[0], "[", "]");
337 		// Handle values with no index and special cases like value[x] 
338 		if (index == null || "x".equals(index)) {
339 			return -1;
340 		}
341 		propertyName[0] = StringUtils.substringBefore(propertyName[0], "[");
342 		
343 		if ("$last".equals(index)) {
344 			return Integer.MAX_VALUE;
345 		}
346 		
347 		int value = Integer.parseInt(index);
348 		if (value < 0) {
349 			throw new IndexOutOfBoundsException(value + " is not a legal index value in " + propertyName);
350 		}
351 		return value;
352 	}

354 	private static String getExtension(String[] propertyName) {
355 		String extension = null;
356 		if (propertyName[0].startsWith("extension(")) {
357 			extension = StringUtils.substringBetween(propertyName[0], "(", ")");
358 			extension = StringUtils.replaceChars(extension, "\"'", "");
359 			if (extension == null) {
360 				throw new FHIRException("Missing extension name: " + propertyName[0]);
361 			} else if (StringUtils.substringAfter(extension, ")").length() > 0) {
362 				throw new FHIRException("Invalid extension function: " + propertyName[0]);
363 			}
364 			
365 		} else if (propertyName[0].contains("-")) {
366 			// save extension name
367 			extension = propertyName[0];
368 			// Translate commonly used extension names.
369 			if (extension.startsWith("us-core")) {
370 				return US_CORE_EXT_PREFIX + extension;
371 			} 
372 			
373 			if (!extension.startsWith("http:")) {
374 				return FHIR_EXT_PREFIX + extension;
375 			}
376 		} else {
377 			return null;
378 		}
379 		
380 		return extension;
381 	}

383 	/**
384 	 * Get the type associated with the specified property.  
385 	 * This is used to adjust property names by removing terminal type specifiers 
386 	 * to correct names like eventCoding (which is actually named event[x] in 
387 	 * MessageHeader).  The challenge is that some property names (e.g., Patient.birthDate)
388 	 * also end in a type name, but don't follow that pattern.
389 	 * 
390 	 * @param b	The base fhir object having the property
391 	 * @param propertyName	The name of the property.
392 	 * @return
393 	 */
394 	static String getType(IBase b, String[] propertyName) {
395 		// Deal with special cases
396 		if ("Reference".equals(b.fhirType()) && Character.isUpperCase(propertyName[0].charAt(0))) {
397 			// allow shortcut for Reference.TypeName
398 			return null;
399 		}
400 		
401 		String type = null;
402 		if (Character.isUpperCase(propertyName[0].charAt(0))) {
403 			// A property name such as Bundle, IntegerType, etc, is assumed to be a type.
404 			type = propertyName[0];
405 			propertyName[0] = "";
406 			return type;
407 		} 
408 		if (propertyName[0].startsWith("ofType(")) {
409 			type = StringUtils.substringBetween("ofType(", ")");
410 			if (StringUtils.substringAfter(propertyName[0], ")").length() != 0 || type == null) {
411 				throw new FHIRException("Invalid ofType call: " + propertyName);
412 			}
413 			propertyName[0] = propertyName[0].substring(0, propertyName[0].length() - type.length());
414 		} else if (VARIES_EXCEPTIONS.contains(propertyName[0])) {
415 			// This is an exception to the rule for property names ending
416 			// in a type.
417 			return null;
418 		} else {
419 			Matcher m = VARIES_PATTERN.matcher(propertyName[0]);
420 			if (m.matches()) {
421 				type = m.group(1);
422 				propertyName[0] = 
423 					StringUtils.left(
424 						propertyName[0], propertyName[0].length() - type.length()
425 					) + "[x]";
426 			}
427 		}
428 		return type;
429 	}
