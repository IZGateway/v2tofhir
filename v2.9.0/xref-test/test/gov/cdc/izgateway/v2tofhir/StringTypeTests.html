32  	protected <P, T extends PrimitiveType<P>> void testStrings(String inputString, UnaryOperator<String> normalize, Predicate<String> rangeTest, Class<T> clazz, Function<Type, T> creator) throws DataTypeException {
33  		T expected = null;
34  		String expectedString = normalize == null ? inputString : normalize.apply(inputString);
35  		String input = inputString;
36  		
37  		if (rangeTest == null || rangeTest.test(input)) {
38  			// Input is within range.
39  			// Convert the plain string with FHIR
40  			try {
41  				expected = clazz.getDeclaredConstructor(String.class).newInstance(expectedString);
42  			} catch (InstantiationException | IllegalAccessException | IllegalArgumentException
43  					| InvocationTargetException | NoSuchMethodException | SecurityException e) {
44  				throw new ServiceConfigurationError("Could not create expected value", e);
45  			} catch (Exception e) {
46  				// Ignore FHIR creation failures, these are normal with invalid inputs, we'll check if expected == null later.
47  				expectedString = null;  // If FHIR create fails, expected string is null;
48  			}
49  			if ((expected instanceof IdType) && StringUtils.isBlank(inputString)) {
50  				// IdType is normalized from blanks to null
51  				expectedString = null;
52  			}
53  			assertEquals(expectedString, expected.asStringValue()); // Verify FHIR String matches with expected String.
54  
55  			MyPrimitive st = new MyPrimitive(inputString);  // Create a V2 Numeric using the base for all V2 NM types
56  			T actual = creator.apply(st);
57  			testFormattedInput(actual, expected, expectedString);
58  		}
59  			
60  	}

86  	protected <P, T extends PrimitiveType<P>> void testFormattedInput(T actual, T expected, String expectedString) throws DataTypeException {
87  		if (actual != null) {
88  			actual.setValue(actual.getValue()); // Force renormalization of the String Value in FHIR
89  		}
90  		String actualString = actual == null ? null : actual.getValueAsString();
91  		if (expected != null) {  // FHIR Created something
92  			assertEquals(expected.getValue(), actual.getValue());
93  		}
94  		assertEquals(expectedString, actualString);
95  	}
