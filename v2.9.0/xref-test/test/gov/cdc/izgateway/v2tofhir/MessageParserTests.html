110 	void writeTheTests(TestData testData, Message msg) {
111 		if (bw == null) {
112 			try {
113 				bw = new BufferedWriter(new FileWriter("./target/messages.txt"));
114 				Runtime.getRuntime().addShutdownHook(new Thread(this::shutdown));
115 			} catch (IOException e) {
116 				// If we cannot write, just exit.
117 				e.printStackTrace();  // NOSONAR
118 				System.exit(1);
119 			}
120 		}
121 		for (String segment: testData.getTestData().split("\r")) {
122 			String segmentName = StringUtils.left(segment, 3);
123 			Class<Processor<Message, Structure>> parser = MP.loadParser(segmentName);
124 			if (parser == null) {
125 				// Skip segments there is no parser for.
126 				continue;
127 			}
128 			LinkedHashMap<String, String> assertions = getExistingAssertions(testData, segmentName);
129 			
130 			for (ComesFrom ann: parser.getAnnotationsByType(ComesFrom.class)) {
131 				String annotation = getAssertion(ann, msg);
132 				String key = StringUtils.substringBetween(annotation, "@", ".exists(");
133 				String exists = assertions.get(key);
134 				if (exists != null) {
135 					log.warn("Assertion ({}) exists for ({})", exists, annotation);
136 				} else {
137 					assertions.put(key, annotation);
138 				}
139 			}
140 			try {
141 				writeSegment(segment);
142 				writeAssertions(assertions.values());
143 			} catch (IOException e) {
144 				log.error("IO Error writing assertions, aborting test");
145 				// If there was an IO Error, subsequent tests will also
146 				// fail. Just give up.
147 				System.exit(1);
148 			}
149 		}

151 	private static LinkedHashMap<String, String> getExistingAssertions(TestData testData, String segmentName) {
152 		LinkedHashMap<String, String> assertions = new LinkedHashMap<>();
153 		for (int i = 0; i < testData.getNames().size(); i++) {
154 			if (testData.getName(i).startsWith(segmentName)) {
155 				String alreadyAsserted = testData.getFhirPath(i);
156 				if (alreadyAsserted.contains(".exists(")) {
157 					String key = StringUtils.substringBefore(alreadyAsserted, ".exists(");
158 					if (!assertions.containsKey(key)) {
159 						assertions.put(key, "  @" + alreadyAsserted + "\n");
160 					}
161 					assertions.put(alreadyAsserted, "  @" + alreadyAsserted + "\n");
162 				}
163 			}
164 		}
165 		return assertions;
166 	}
167 	private static String getAssertion(ComesFrom ann, Message msg) {
168 		StringBuilder b = new StringBuilder();
169 		addComesFrom(ann, b);
170 		Terser t = new Terser(msg);
171 		b.append("  @").append(ann.path()).append(".exists( \\\n");
172 		for (String source: ann.source()) {
173 			if (source.isEmpty()) {
174 				continue;
175 			}
176 			// Source is a Terser path in the form SEG-#-#
177 			try {
178 				String data = t.get(source);
179 				b.append("    /* ").append(source).append(" = ").append(data).append(" */ \\\n");
180 			} catch (HL7Exception e) {
181 				// Shouldn't happen, the message was already parsed.
182 				log.warn("Error parsing message: {}", e.getMessage());
183 			}
184 		}
185 		if (!ann.fixed().isEmpty()) {
186 			b.append("    /* fixed = \"").append(ann.fixed()).append("\" */ \\\n");
187 		}
188 		b.append("  ) \\\n");
189 		if (!ann.comment().isEmpty()) {
190 			b.append("  // ").append(ann.comment()).append("\n");
191 		}
192 		return b.toString();
193 	}
194 	private static void addComesFrom(ComesFrom ann, StringBuilder b) {
195 		b.append("  // ComesFrom(path=\"").append(ann.path()).append("\", ");
196 		if (ann.source().length == 1) {
197 			b.append("source = \"").append(ann.source()[0]).append("\")");
198 		} else if (ann.source().length > 1) {
199 			b.append("source = { ");
200 			for (int i = 0; i < ann.source().length; i++) {
201 				if (i != 0) b.append(", ");
202 				b.append("\"").append(ann.source()[i]).append("\"");
203 			}
204 			b.append("})");
205 		} else if (!ann.fixed().isEmpty()) {
206 			b.append("fixed = \"").append(ann.fixed()).append("\")");
207 		}
208 		b.append(" \\\n");
209 	}

220 	void writeSegment(String segment) throws IOException {
221 		int max = 80;
222 		int pos = StringUtils.lastIndexOf(StringUtils.left(segment, max), '|');
223 		do {
224 			String left = StringUtils.left(segment, pos >= 0 ? pos + 1 : max);
225 			bw.write(left);
226 			segment = StringUtils.substring(segment, left.length());
227 			if (StringUtils.isNotBlank(segment)) {
228 				bw.write("\\\n  ");
229 				max = 78;
230 			}
231 		} while (StringUtils.isNotBlank(segment));
232 		bw.write("\n");
233 	}
234 	void writeAssertions(Iterable<String> assertions) throws IOException {
235 		String last = null;
236 		
237 		for (String assertion: assertions) {
238 			if (assertion.equals(last)) {
239 				// Skip values in sequence that are duplicates
240 				continue;
241 			}
242 			bw.write(last = assertion);
243 		}
244 	}

401 	private String verify(Segment segment, Resource res, ComesFrom from) {
402 		// So now we have a path and a segment and field.
403 		// We should be able to compare these using TextUtils.toString() methods.
404 		getFields(segment, from);
405 		List<IBase> l = getProducedItems(res, from);
406 		for (Type type: fieldValues) {
407 			type = DatatypeConverter.adjustIfVaries(type, from.type());
408 			String value = getMappedValue(from, type).replace("#", " ");
409 			if (StringUtils.isEmpty(value)) {
410 				continue;
411 			}
412 			String produced = getProducedValue(l, value, type.getName());
413 			
414 			// For invalid datetime, check to see if value is actually a valid
415 			// date time, and let it be ok if that nothing was produced if 
416 			// value is invalid.
417 			if (produced == null && DatatypeConverter.DATETIME_TYPES.contains(type.getName())) {
418 				produced = checkDatetime(value);
419 			}
420 			
421 			String error = null;
422 			if (produced == null) {
423 				error = "Did not find " + fieldName + " = '" + value + " in " + l + "' at " + from.path();
424 				log.info(error);
425 				return error;
426 			} else {
427 				log.info("Found " + fieldName + " = '" + value + "' in '" + produced + "' at " + from.path());
428 			}
429 		}
430 		return null;
431 	}

433 	private String checkDatetime(String value) {
434 		String[] parts = value.split("[\\.\\-+]");
435 		if (!StringUtils.isNumeric(				// All Digits must be numeric
436 				Strings.CS.replace(value, ".+-", "")) ||
437 			(parts[0].length() % 2) == 1 ||		// First part Must have YYYY[MM[DD[HH[MM[SS]]]]] and even length
438 			parts[0].length() < 4 || 			// at least 4, 
439 			parts[0].length() > 14 ||			// and at most 14
440 			(value.contains(".") && 			// decimal part can have 1-4 digits
441 				(parts[1].isEmpty() || parts[1].length() > 4)) ||
442 			(!value.contains(".") && parts.length > 1 && parts[1].length() != 4) ||	// TZ must be 4 digits if present
443 			(value.contains(".") && 
444 				(parts.length < 2 || parts[2].length() != 4))   // TZ must be 4 digits if present
445 		) {
446 			return "null due to invalid input";
447 		}
448 		FastDateFormat ft = FastDateFormat.getInstance("yyyyMMddhhmmss".substring(0, parts[0].length()));
449 		try {
450 			Date d = ft.parse(parts[0]);
451 			if (!ft.format(d).equals(value)) {
452 				return "invalid date time value";
453 			}
454 		} catch (ParseException e) {
455 			return "null due to incorrect time/date input";
456 		}
457 		
458 		String tz = (parts.length > 2) || (!value.contains(".") && parts.length > 1) ? StringUtils.right(value,  5) : null;
459 		if (tz != null) {
460 			try {
461 				int tzi = Integer.parseInt(tz);
462 				if (tzi < -1400 || tzi > 1500 || (tzi % 15 != 0)) {
463 					return "invalid timezone";
464 				}
465 			} catch (NumberFormatException ex) {
466 				return "non-numeric timezone";
467 			}
468 		}
469 		return null;
470 	}

472 	private String getProducedValue(List<IBase> l, String value, String v2Type) {
473 		String produced = null;
474 		
475 		for (IBase base: l) {
476 			// "#" for " " replacement fixes identifier and reference types.
477 			produced = TextUtils.toString(base).replace("#", " ");
478 			// produced can be CodeableConcept or Coding, and value 
479 			// can be from a simpler entity, such as an ST or ID.
480 			if (Strings.CI.contains(produced, value) ||
481 				Strings.CI.contains(value, produced)) {
482 				return produced;
483 			} else if (base instanceof CodeableConcept cc) {
484 				for (Coding coding: cc.getCoding()) {
485 					// Substring.between doesn't work because display can have ().
486 					produced = "(" +
487 						StringUtils.substringAfterLast(TextUtils.toString(coding), "(");
488 					if (Strings.CI.contains(value, produced)) {
489 						return produced;
490 					}
491 				}
492 				log.info("{} <> {}", value, produced);
493 			} else if (base instanceof InstantType) {
494 				// Deal with missing data in V2 timestamps.
495 				String[] parts = value.split("[\\-+]");
496 				if (parts.length > 1) {
497 					parts[1] = StringUtils.right(value, parts[1].length() + 1);
498 				}
499 				if (produced.startsWith(parts[0]) && parts.length == 1 || produced.endsWith(parts[1])) {
500 					return produced;
501 				}
502 			} else if (base instanceof Organization org && "CE".equals(v2Type)) {
503 				// Deal with special case of CE -> Organization for MVX Codes
504 				produced = TextUtils.toString(org.getIdentifierFirstRep()).replace("#", " ");
505 				if (Strings.CI.contains(value, produced)) {
506 					return produced;
507 				}
508 			}
509 		}
510 		log.info("Produced {} <> value {}", produced, value);
511 		return null;
512 	}

535 	private String getMappedValue(ComesFrom from, Type type) {
536 		String value;
537 		if (!from.map().isEmpty()) {
538 			value = ParserUtils.toString(type);
539 			String oldValue = value;
540 			org.hl7.fhir.r4.model.Coding sourceCoding = new org.hl7.fhir.r4.model.Coding().setCode(value);
541 			org.hl7.fhir.r4.model.Coding coding = TerminologyMapperFactory.get()
542 					.mapCode(from.map(), sourceCoding)
543 					.map(gov.cdc.izgw.v2tofhir.terminology.TranslationResult::resultCoding)
544 					.orElse(null);
545 			if (coding != null) {
546 				value = coding.hasCode() ? coding.getCode() : null;
547 			}
548 			fieldName += " mapped from " + oldValue + " by " + from.map();
549 		} else {
550 			value = TextUtils.toString(type);
551 		}
552 		return value;
553 	}
