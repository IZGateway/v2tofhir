20  /**
21   * Tests for the conversion of the HL7 V2 person-location composites (PL, LA1 and LA2) to
22   * Location, covering which component names each Location, which location-physical-type code it
23   * carries, how several populated components nest through Location.partOf, and which element
24   * components 9 and beyond supply for each of the three datatypes.
25   *
26   * The physical-type codes come from the HL7 V2-to-FHIR IG ConceptMap datatype-pl-to-location:
27   * PL-3 Bed = bd, PL-2 Room = ro, PL-1 Point of Care = no code, PL-8 Floor = lvl,
28   * PL-7 Building = bu, PL-4 Facility = si.
29   *
30   * @author Audacious Inquiry
31   */
32  class LocationCompositeTests extends TestBase {
33  	private static final String PHYSICAL_TYPE = "http://terminology.hl7.org/CodeSystem/location-physical-type";
34  	/** The RXA-11 from the eHealth Exchange pilot response the conformance review was written against */
35  	private static final String REVIEWED_RXA_11 =
36  		"IZGATEWAYART^^^IZGATEWAY-AART^^^IZ GATEWAY AART TEST^^330 C ST SW UNIT 7^^UNKNOWN^VA^20201";
37  
38  	/**
39  	 * The reviewed RXA-11: Point of Care, Facility and Building are populated, so each names its own
40  	 * Location and they nest from the most specific component to the least.  The review reported
41  	 * bd/Bed on the Point of Care and lvl/Level on the Facility, neither of which the message sent.
42  	 */
43  	@Test
44  	void testReviewedRxa11IsLabelledFromItsOwnComponents() {
45  		List<Location> chain = chainOf(toLa2(REVIEWED_RXA_11));
46  		assertEquals(3, chain.size(), "One Location per populated component");
47  
48  		assertEquals(Arrays.asList("IZGATEWAYART", "IZ GATEWAY AART TEST", "IZGATEWAY-AART"), namesOf(chain),
49  				"Point of Care, then Building, then Facility");
50  		assertEquals(Arrays.asList(null, "bu", "si"), codesOf(chain));
51  		assertFalse(codesOf(chain).contains("bd"), "Nothing sent a Bed");
52  		assertFalse(codesOf(chain).contains("lvl"), "Nothing sent a Floor");
53  		for (Location location : chain) {
54  			assertEquals(LocationMode.INSTANCE, location.getMode());
55  		}
56  	}

59  	@Test

62  		assertEquals(Arrays.asList("IZGATEWAYART", "IZ GATEWAY AART TEST", "IZGATEWAY-AART"), namesOf(chain));
63  		assertEquals(Arrays.asList(null, "bu", "si"), codesOf(chain));
64  	}

88  		List<String> codes = codesOf(chain);

94  	/**
95  	 * PL-5 Location Status and PL-6 Person Location Type describe the location rather than naming a
96  	 * place, so each is carried once on the Location and neither produces one of its own.
97  	 */
98  	@Test
99  	void testLocationStatusAndPersonLocationTypeAreNotLocations() {
100 		Location location = toLocation(toPl("IZGATEWAYART^^^^A^E"));
101 		assertEquals("A", location.getOperationalStatus().getCode(), "From PL-5");
102 		assertEquals(1, location.getType().size(), "PL-6 once");
103 		assertEquals("E", location.getTypeFirstRep().getCodingFirstRep().getCode(), "From PL-6");
104 
105 		List<Location> chain = chainOf(location);
106 		assertEquals(1, chain.size(), "Only Point of Care names a Location");
107 		assertFalse(codesOf(chain).contains("A"));
108 		assertFalse(codesOf(chain).contains("E"));

127 	/**
128 	 * LA2-9 through LA2-16 are the address, which is where they go.  An LA2 has no Location
129 	 * Description component and no Comprehensive Location Identifier component, so it produces
130 	 * neither -- the review saw the street address of the reviewed RXA-11 as a description.
131 	 */
132 	@Test
133 	void testLa2CarriesAnAddressAndNoDescription() {
134 		Location location = toLocation(toLa2(REVIEWED_RXA_11));
135 		assertEquals("330 C ST SW UNIT 7", location.getAddress().getLine().get(0).getValue());
136 		assertEquals("UNKNOWN", location.getAddress().getCity());
137 		assertEquals("VA", location.getAddress().getState());
138 		assertEquals("20201", location.getAddress().getPostalCode());
139 		for (Location inChain : chainOf(location)) {
140 			assertFalse(inChain.hasDescription(), "LA2-9 is a street address, not a description");
141 			assertFalse(inChain.hasIdentifier(), "LA2-10 is another designation, not an identifier");
142 		}
143 	}

175 	private static Location toLocation(Type type) {
176 		Location location = DatatypeConverter.toLocation(type);
177 		assertNotNull(location, "The composite converts to a Location");
178 		return location;
179 	}

181 	/** PL is PV1-3 Assigned Patient Location */
182 	private static Type toPl(String pl) {
183 		return fieldOf("PV1|1|I|" + pl, 3);
184 	}
185 
186 	/** LA2 is RXA-11 Administered-at Location */
187 	private static Type toLa2(String la2) {
188 		return fieldOf("RXA|0|1|20220810|20220810|09^Td (adult), adsorbed^CVX|0.5|mL^MilliLiter^UCUM|||"
189 			+ "01^Historical Information - Source Unspecified^NIP001|" + la2, 11);
190 	}
191 
192 	/** LA1 is RXE-8 Deliver-To Location */
193 	private static Type toLa1(String la1) {
194 		return fieldOf("RXE||||||||" + la1, 8);
195 	}

197 	private static Type fieldOf(String segment, int field) {
198 		try {
199 			Segment seg = parseSegment(segment);
200 			Type type = seg.getField(field, 0);
201 			assertNotNull(type);
202 			return type;
203 		} catch (Exception e) {
204 			throw new AssertionError("Cannot parse " + segment, e);
205 		}
206 	}
207 
208 	/** The Location and everything it is partOf, in order */
209 	private static List<Location> chainOf(Type type) {
210 		return chainOf(toLocation(type));
211 	}
212 
213 	private static List<Location> chainOf(Location location) {
214 		List<Location> chain = new ArrayList<>();
215 		Location current = location;
216 		while (current != null) {
217 			chain.add(current);
218 			current = current.hasPartOf() ? (Location) current.getPartOf().getResource() : null;
219 		}
220 		return chain;
221 	}
222 
223 	private static List<String> namesOf(List<Location> chain) {
224 		return chain.stream().map(Location::getName).toList();
225 	}
226 
227 	/** The physicalType code of each Location in the chain, null where it carries none */
228 	private static List<String> codesOf(List<Location> chain) {
229 		List<String> codes = new ArrayList<>();
230 		for (Location location : chain) {
231 			codes.add(location.hasPhysicalType() ? location.getPhysicalType().getCodingFirstRep().getCode() : null);
232 		}
233 		return codes;
234 	}
