14  /**
15   * Unit tests for {@link ConceptTranslator#mapCodeableConcept(String, CodeableConcept)}.
16   *
17   * <p>All tests use a simple anonymous-inner-class stub of {@link ConceptTranslator} that
18   * returns controlled {@link TranslationResult} values so the two-pass logic in the
19   * {@code default} method can be verified in isolation.</p>
20   */
21  class ConceptTranslatorTests {
22  
23      private static final String MAPPING = "testMap";
24  
25      /** Stub that always returns {@link Optional#empty()} for every code. */
26      private static final ConceptTranslator EMPTY_TRANSLATOR = new ConceptTranslator() {
27          @Override
28          public Optional<TranslationResult> mapCode(String mappingName, Coding source) {
29              return Optional.empty();
30          }
31      };
32  
33      // -------------------------------------------------------------------------
34      // (a) No recognized coding → Pass 3 returns first code unchanged
35      // -------------------------------------------------------------------------
36  
37      @Test
38      void mapCodeableConcept_noRecognizedCoding_returnsFirstCodeUnchanged() {
39          CodeableConcept cc = new CodeableConcept()
40                  .addCoding(new Coding("http://example.com", "A", "Alpha"))
41                  .addCoding(new Coding("http://example.com", "B", "Beta"));
42  
43          CodeType result = EMPTY_TRANSLATOR.mapCodeableConcept(MAPPING, cc);
44  
45          assertNotNull(result);
46          assertEquals("A", result.getValue(), "Pass 3 must return the first coding's code");
47      }
48  
49      // -------------------------------------------------------------------------
50      // (b) One pass-through coding and one explicit match → explicit match wins
51      // -------------------------------------------------------------------------
52  
53      @Test
54      void mapCodeableConcept_explicitMatchWinsOverPassThrough() {
55          // Coding "A" → pass-through (exact=false), Coding "B" → explicit match (exact=true)
56          ConceptTranslator translator = new ConceptTranslator() {
57              @Override
58              public Optional<TranslationResult> mapCode(String mappingName, Coding source) {
59                  if ("A".equals(source.getCode())) {
60                      Coding passThrough = new Coding(source.getSystem(), "A-PT", "Alpha PassThrough");
61                      return Optional.of(TranslationResult.approximate(source, passThrough, mappingName));
62                  }
63                  if ("B".equals(source.getCode())) {
64                      Coding mapped = new Coding("http://mapped.com", "B-MAPPED", "Beta Mapped");
65                      return Optional.of(TranslationResult.exact(source, mapped, mappingName));
66                  }
67                  return Optional.empty();
68              }
69          };
70  
71          // "A" is first in the list but only pass-through; "B" has an exact match
72          CodeableConcept cc = new CodeableConcept()
73                  .addCoding(new Coding("http://example.com", "A", "Alpha"))
74                  .addCoding(new Coding("http://example.com", "B", "Beta"));
75  
76          CodeType result = translator.mapCodeableConcept(MAPPING, cc);
77  
78          assertNotNull(result);
79          assertEquals("B-MAPPED", result.getValue(),
80                  "Pass 1 must return the explicit match even when it is not the first coding");
81      }
82  
83      // -------------------------------------------------------------------------
84      // (c) Only pass-through codings → first pass-through returned (Pass 2)
85      // -------------------------------------------------------------------------
86  
87      @Test
88      void mapCodeableConcept_onlyPassThroughCodings_returnsFirstPassThrough() {
89          // Both codings map to approximate results; Pass 2 must return the first one
90          ConceptTranslator translator = new ConceptTranslator() {
91              @Override
92              public Optional<TranslationResult> mapCode(String mappingName, Coding source) {
93                  Coding passThrough = new Coding(source.getSystem(), source.getCode() + "-PT", null);
94                  return Optional.of(TranslationResult.approximate(source, passThrough, mappingName));
95              }
96          };
97  
98          CodeableConcept cc = new CodeableConcept()
99                  .addCoding(new Coding("http://example.com", "X", "X Display"))
100                 .addCoding(new Coding("http://example.com", "Y", "Y Display"));
101 
102         CodeType result = translator.mapCodeableConcept(MAPPING, cc);
103 
104         assertNotNull(result);
105         assertEquals("X-PT", result.getValue(),
106                 "Pass 2 must return the first pass-through result");
107     }
108
