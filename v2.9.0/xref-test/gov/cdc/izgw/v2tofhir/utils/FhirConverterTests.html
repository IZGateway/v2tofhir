16  /**
17   * Regression tests for {@link FhirConverter#read}, notably parsing of
18   * {@code Content-Type} headers that carry parameters such as
19   * {@code ;charset=UTF-8} (IGDD-3164).
20   */
21  class FhirConverterTests {
22  
23      private static final String PATIENT_JSON = "{\"resourceType\":\"Patient\",\"id\":\"example\"}";
24      private static final String PATIENT_XML =
25              "<Patient xmlns=\"http://hl7.org/fhir\"><id value=\"example\"/></Patient>";
26      private static final String PATIENT_YAML = "resourceType: Patient\nid: example\n";
27  
28      private final FhirConverter converter = new FhirConverter();
29  
30      private Patient read(String body, String contentType) {
31          MockHttpInputMessage message = new MockHttpInputMessage(body.getBytes(StandardCharsets.UTF_8));
32          if (contentType != null) {
33              message.getHeaders().set(HttpHeaders.CONTENT_TYPE, contentType);
34          }
35          Resource resource = assertDoesNotThrow(() -> converter.read(Patient.class, message));
36          Patient patient = assertInstanceOf(Patient.class, resource);
37          assertEquals("example", patient.getIdElement().getIdPart());
38          return patient;
39      }
40  
41      @Test
42      void readsJsonWithCharsetParameter() {
43          read(PATIENT_JSON, "application/fhir+json;charset=UTF-8");
44      }

46      @Test

80      // IGDD-3168: Spring passes the declared @RequestBody type to read(). A handler
81      // declaring an abstract type such as org.hl7.fhir.r4.model.Resource previously
82      // failed with "HAPI-1682: Can not scan abstract or interface class".
83  
84      private static final String MATCH_PARAMETERS_JSON =
85              "{\"resourceType\":\"Parameters\",\"parameter\":[{\"name\":\"resource\",\"resource\":"
86              + "{\"resourceType\":\"Patient\",\"name\":[{\"family\":\"Smith\",\"given\":[\"John\"]}],"
87              + "\"birthDate\":\"1970-01-01\"}}]}";
88      private static final String MATCH_PARAMETERS_XML =
89              "<Parameters xmlns=\"http://hl7.org/fhir\"><parameter><name value=\"resource\"/>"
90              + "<resource><Patient><name><family value=\"Smith\"/><given value=\"John\"/></name>"
91              + "<birthDate value=\"1970-01-01\"/></Patient></resource></parameter></Parameters>";
92      private static final String NAMED_PATIENT_JSON =
93              "{\"resourceType\":\"Patient\",\"name\":[{\"family\":\"Smith\",\"given\":[\"John\"]}],"
94              + "\"birthDate\":\"1970-01-01\"}";
95  
96      private Resource readAs(Class<? extends Resource> clazz, String body, String contentType) {
97          MockHttpInputMessage message = new MockHttpInputMessage(body.getBytes(StandardCharsets.UTF_8));
98          if (contentType != null) {
99              message.getHeaders().set(HttpHeaders.CONTENT_TYPE, contentType);
100         }
101         return assertDoesNotThrow(() -> converter.read(clazz, message));
102     }
103 
104     @Test
105     void readsParametersWhenRequestedTypeIsAbstractResource() {
106         // Case A: the $match bug - declared @RequestBody type is the abstract base Resource.
107         Resource resource = readAs(Resource.class, MATCH_PARAMETERS_JSON, "application/fhir+json");
108         Parameters parameters = assertInstanceOf(Parameters.class, resource);
109         assertEquals("Parameters", parameters.fhirType());
110     }
111 
112     @Test
113     void readsBarePatientWhenRequestedTypeIsAbstractResource() {
114         // Case B: a bare Patient payload resolves to the concrete Patient type.
115         Resource resource = readAs(Resource.class, NAMED_PATIENT_JSON, "application/fhir+json");
116         Patient patient = assertInstanceOf(Patient.class, resource);
117         assertEquals("Patient", patient.fhirType());
118     }
119 
120     @Test
121     void readsPatientViaConcreteTypePath() {
122         // Case C: concrete requested type still uses the existing typed parseResource path.
123         Resource resource = readAs(Patient.class, NAMED_PATIENT_JSON, "application/fhir+json");
124         Patient patient = assertInstanceOf(Patient.class, resource);
125         assertEquals("Smith", patient.getNameFirstRep().getFamily());
126     }
127 
128     @Test
129     void readsParametersXmlWhenRequestedTypeIsAbstractResource() {
130         // Case D: the XML parser path also works through the abstract branch.
131         Resource resource = readAs(Resource.class, MATCH_PARAMETERS_XML, "application/fhir+xml");
132         Parameters parameters = assertInstanceOf(Parameters.class, resource);
133         assertEquals("Parameters", parameters.fhirType());
134     }
