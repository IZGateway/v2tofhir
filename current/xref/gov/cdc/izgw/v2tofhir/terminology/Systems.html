252 	/**
253 	 * Add a naming system to the map of known systems.
254 	 * @param name The name of the system
255 	 * @param list	A list of strings containing aliases for a coding or identifier system.  
256 	 * The first entry must be the URI.
257 	 */
258 	public static void addNamingSystem(String name, String... list) {
259 		NamingSystem ns = null;
260 		String uri = list[0];
261 		ns = namingSystems.get(uri);
262 		if (ns == null) {
263 			// Create a new one
264 			ns = createNamingSystem(uri, name);
265 			namingSystems.put(uri, ns);
266 		}
267 		if (name != null) {
268 			if (!ns.hasName()) {
269 				ns.setName(name);
270 			}
271 			namingSystems.put(name, ns);
272 		}
273 		for (String s: list) {
274 			if (!uri.equals(s)) {
275 				updateNamingSystem(ns, s);
276 			}
277 			namingSystems.put(s, ns);
278 		}
279 	}

281 	/**
282 	 * Create a naming system with the given URI and name.
283 	 * @param uri	The URI for the naming system
284 	 * @param name	A common name for the naming system
285 	 * @return	The created naming system
286 	 */
287 	public static NamingSystem createNamingSystem(String uri, String name) {
288 		NamingSystem ns = new NamingSystem();
289 		ns.setUrl(uri);
290 		NamingSystemUniqueIdComponent uid = ns.addUniqueId();
291 		uid.setType(NamingSystemIdentifierType.URI);
292 		uid.setValue(uri);
293 		uid.setPreferred(true);
294 		uid = ns.addUniqueId();
295 		if (StringUtils.isNotBlank(name)) {
296 			uid.setType(NamingSystemIdentifierType.OTHER);
297 			uid.setValue(name.trim());
298 			uid.setPreferred(false);
299 		}
300 		return ns;
301 	}

303 	/**
304 	 * Add a unique id to a naming system if it doesn't already exist.
305 	 * @param ns	The naming system to update
306 	 * @param uid The unique id to add
307 	 */
308 	public static void updateNamingSystem(NamingSystem ns, String uid) {
309 		for (NamingSystemUniqueIdComponent uniqueId : ns.getUniqueId()) {
310 			if (uid.equals(uniqueId.getValue())) {
311 				return;
312 			}
313 		}
314 		NamingSystemUniqueIdComponent uniqueId = ns.addUniqueId();
315 		uniqueId.setValue(uid);
316 		uniqueId.setPreferred(false);
317 		if (uid.matches("^([\\-a-z0-9]++):.*$")) {
318 			uniqueId.setType(NamingSystemIdentifierType.URI);
319 		} else if (uid.matches("^\\d++(.\\d++)++$")) {
320 			uniqueId.setType(NamingSystemIdentifierType.OID);
321 		} else if (uid.matches("^\\p{XDigit}++(-\\p{XDigit}++)++$")) {
322 			uniqueId.setType(NamingSystemIdentifierType.UUID);
323 		} else {
324 			uniqueId.setType(NamingSystemIdentifierType.OTHER);
325 		}
326 		namingSystems.put(uid, ns);
327 	}

329 	/**
330 	 * Given a URI, get the associated OID
331 	 * @param uri	The URI to get the OID for
332 	 * @return the associated OID or null if none is known (without the urn:oid: prefix).
333 	 */
334 	public static String toOid(String uri) {
335 		if (StringUtils.isEmpty(uri)) {
336 			return null;
337 		}
338 		NamingSystem ns = namingSystems.get(uri);
339 		if (ns == null) {
340 			return null;
341 		}
342 		for (NamingSystemUniqueIdComponent uid : ns.getUniqueId()) {
343 			if (NamingSystemIdentifierType.OID.equals(uid.getType())) {
344 				return uid.getValue();
345 			}
346 		}
347 		return null;
348 	}

350 	/** 
351 	 * Get the commonly used V2 name for a coding or identifier system.
352 	 *  
353 	 * @param uri	The system name
354 	 * @return	The commonly used V2 name, or null if name is unknown. 
355 	 */
356 	public static String toTextName(String uri) {
357 		if (StringUtils.isEmpty(uri)) {
358 			return "";
359 		}
360 		NamingSystem ns = namingSystems.get(uri);
361 		if (ns == null) {
362 			return null;
363 		}
364 		for (NamingSystemUniqueIdComponent uid : ns.getUniqueId()) {
365 			if (!uid.getPreferred()) {  // First non-preferred uinque id is used as name 
366 				return uid.getValue();
367 			}
368 		}
369 		if (ns.hasUniqueId()) {
370 			return ns.getUniqueIdFirstRep().getValue();
371 		}
372 		return null;
373 		
374 	}
375 	/**
376 	 * Given a OID, get the associated URI
377 	 * @param oid The oid to look up, with or without urn:oid: prefix
378 	 * @return the associated URI or null if none is known
379 	 */
380 	public static String toUri(String oid) {
381 		if (oid.startsWith("urn:oid:")) {
382 			oid = oid.substring(8);
383 		}
384 		NamingSystem ns = namingSystems.get(oid);
385 		if (ns == null) {
386 			return null;
387 		}
388 		return ns.getUrl();
389 	}

391 	/**
392 	 * Get the aliases for any given system.
393 	 * @param system	The system
394 	 * @return	A list of names the system is known by.
395 	 */
396 	public static List<String> getSystemNames(String system) {
397 		if (StringUtils.isBlank(system)) {
398 			return Collections.emptyList();
399 		}
400 		system = StringUtils.trim(system);
401 		String original = system;
402 		NamingSystem ns = getNamingSystem(system);
403 		List<String> found = null;
404 		if (ns == null) {
405 			found = new ArrayList<>();
406 		} else {
407 			found = ns.getUniqueId().stream().map(u -> u.getValue()).collect(Collectors.toCollection(ArrayList::new));
408 		}
409 		// Return a name for things that are obviously HL7 systems.
410 		if ("HL7".equalsIgnoreCase(system)) {
411 			// Used in QPD-1 is some queries.
412 			if (found.isEmpty()) {
413 				found.add("http://terminology.hl7.org/CodeSystem/v2-0003");
414 			}
415 		} else if (
416 			system.matches("^(?i)(HL7-?|http://terminology.hl7.org/CodeSystem/v2-|)\\d{4}$") || 
417 			system.startsWith("2.16.840.1.113883.12.")	// V2 Table OID
418 		) {
419 			String name = "HL7" + StringUtils.right("000" + system, 4);
420 			if (found.isEmpty()) {
421 				found.add("http://terminology.hl7.org/CodeSystem/v2-" + StringUtils.right(name, 4));
422 			}
423 			found.add(name);
424 		}
425 		// Special case for table name as HL7 in IHE PIX/PDQ profile
426 		if (found.contains("HL70003") && !found.contains("HL7")) {
427 			found.add("HL7");
428 		}
429 		if (!found.contains(original)) {
430 			found.add(original);
431 		}
432 		return found;
433 	}

436 	 * Get the NamingSystem for a given name, system uri, or other alias.
437 	 * @param system	The search value
438 	 * @return	The NamingSystem, or null if not found
439 	 */
440 	public static NamingSystem getNamingSystem(String system) {
441 		if (StringUtils.isBlank(system)) {
442 			return null;
443 		}
444 		NamingSystem ns = namingSystems.get(system);
445 		if (ns == null) {
446 			// Normalize value by replacing any whitespace, hypens or underscores, and uppercasing the text.
447 			String normalizedValue = system.replace("[-_\\s]+", "").toUpperCase();
448 			ns = namingSystems.get(normalizedValue);
449 		}
450 		return ns;
