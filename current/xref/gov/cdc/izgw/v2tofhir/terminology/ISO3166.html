261 	private static Map<String, String[]> lookup = new LinkedHashMap<>();
262 	static {
263 		for (String[] array: data) {
264 			for (String name: array) {
265 				name = normalize(name);
266 				lookup.put(name, array);
267 			}
268 		}
269 	}
270 	private static String normalize(String name) {
271 		return name.replace(" and ", " ").replace(" of ", "").replace("the ", "").replace("', ", "").toUpperCase();
272 	}
273 	/**
274 	 * Get the ISO-3166 two letter code for the given country name or code
275 	 * @param name	The country name or ISO code
276 	 * @return	the ISO-3166 two letter code for the given country 
277 	 */
278 	public static String twoLetterCode(String name) {
279 		String[] array = lookup.get(normalize(name));
280 		return array == null ? null : array[0];
281 	}
282 	/**
283 	 * Get the ISO-3166 three letter code for the given country name or code
284 	 * @param name	The country name or ISO code
285 	 * @return	the ISO-3166 three letter code for the given country 
286 	 */
287 	public static String threeLetterCode(String name) {
288 		String[] array = lookup.get(normalize(name));
289 		return array == null ? null : array[1];
290 	}
291 	/**
292 	 * Get the ISO-3166 name for the given country name or code

295 	 */
296 	public static String name(String name) {
297 		String[] array = lookup.get(normalize(name));
