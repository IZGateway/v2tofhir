16  /**
17   * The QID Parser extracts the query tag and query name from the QID segment
18   * and attaches them to the Parameters resource.
19   *
20   * @author Audacious Inquiry
21   *
22   */
23  @Produces(segment="DSC", resource=Parameters.class)
24  @Slf4j
25  public class DSCParser extends AbstractSegmentParser {
26  	private Parameters params;
27  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
28  	static {
29  		log.debug("{} loaded", RCPParser.class.getName());
30  	}
31  	/**
32  	 * Construct a new DSC Parser for the given messageParser.
33  	 * 
34  	 * @param messageParser The messageParser using this QPDParser
35  	 */
36  	public DSCParser(MessageParser messageParser) {
37  		super(messageParser, "DSC");
38  		if (fieldHandlers.isEmpty()) {
39  			FieldHandler.initFieldHandlers(this, fieldHandlers);
40  		}
41  	}
42  	
43  	@Override
44  	public List<FieldHandler> getFieldHandlers() {
45  		return fieldHandlers;
46  	}
47  	
48  	public IBaseResource setup() {
49  		params = createResource(Parameters.class);
50  		return params;
51  	}
52  	
53  	@Override
54  	public IBase parse(Segment qid) {
55  		setup();
56  		params.addParameter("Pointer", ParserUtils.toString(qid, 1));
57  		params.addParameter("Style", ParserUtils.toString(qid, 2));
58  		return params;
59  	}
60  }
