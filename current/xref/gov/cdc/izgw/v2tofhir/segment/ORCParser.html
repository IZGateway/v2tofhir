32  /**
33   * ORCParser parses any ORC segments in a message to a ServiceRequest
34   * If the message is a VXU or an response to an immunization query, it also produces an Immunization resource.
35   * 
36   * @author Audacious Inquiry
37   *
38   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-orc-to-servicerequest.html">V2 to FHIR: ORC to ServiceRequest</a>
39   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-orc-to-immunization.html">V2 to FHIR: ORC to Immunization</a>
40   */
41  @Produces(segment = "ORC", resource=ServiceRequest.class, 
42  	extra = { Practitioner.class, Organization.class, Immunization.class }
43  )
44  public class ORCParser extends AbstractSegmentParser {
45  	private static final String REQUESTER = "requester";
46  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
47  	private ServiceRequest order;
48  	private PractitionerRole requester;
49  	private Reference requesterRef;
50  	private IzDetail izDetail;

52  	/**
53  	 * Create an ORC Parser for the specified MessageParser
54  	 * @param p	The message parser.
55  	 */
56  	public ORCParser(MessageParser p) {
57  		super(p, "ORC");
58  		if (fieldHandlers.isEmpty()) {
59  			FieldHandler.initFieldHandlers(this, fieldHandlers);
60  		}
61  	}

63  	@Override
64  	protected List<FieldHandler> getFieldHandlers() {
65  		return fieldHandlers;
66  	}

68  	@Override
69  	public IBaseResource setup() {
70  		izDetail = IzDetail.get(getMessageParser());
71  		izDetail.initializeResources(true, getSegment());
72  		order = createResource(ServiceRequest.class);
73  		if (izDetail.hasImmunization()) {
74  			// Reports an Immunization performed, and reported by filler, so this is a filler order 
75  			order.setIntent(ServiceRequestIntent.FILLERORDER);
76  		}
77  		Patient patient = this.getLastResource(Patient.class);
78  		if (patient != null) {
79  			order.setSubject(ParserUtils.toReference(patient, order, "patient", "subject"));
80  		}
81  		Encounter encounter = this.getLastResource(Encounter.class);
82  		if (encounter != null) {
83  			order.setEncounter(ParserUtils.toReference(encounter, order, "encounter"));
84  		}
85  		order.setIntent(ServiceRequestIntent.ORDER);
86  		return order;
87  	}

89  	/**
90  	 * Set Order control code from ORC-1
91  	 * 
92  	 * Also determines if this ORC is part of a VXU, or RSP_K11 response to an Immunization Query, and if so,
93  	 * creates the Immunization resource.
94  	 * 
95  	 * @param orderControl	The order control code from Table HL7-0119
96  	 */
97  	@ComesFrom(path = "ServiceRequest.status", field = 1, table = "0119", map = "OrderControlCode[ServiceRequest.status]",
98  			   also = { "ServiceRequest.intent = 'order'",
99  					    "ServiceRequest.subject = Patient",
100 					    "ServiceRequest.encounter = Encounter"
101 			   }, priority = 10)
102 	public void setOrderControl(Coding orderControl) {
103 		// ORC-1 is required, so this method should always be called
104 		
105 		if (order.hasStatus()) {
106 			// Status was already set by ORC-5
107 			return;
108 		}
109 		if (orderControl.hasCode()) {
110 			switch (orderControl.getCode()) {
111 			// For immunization messages, we are only expecting RE.
112 			case "RE", "CN", "OE", "OF", "OR":
113 				order.setStatus(ServiceRequestStatus.COMPLETED);
114 				break;
115 			case "AF", "CH", "FU", "NW", "OK", "PA", "RL", "RO", "RP", 
116 				 "RQ", "RR", "RU":
117 				order.setStatus(ServiceRequestStatus.ACTIVE);
118 				break;
119 			case "CA", "CR", "DC", "DF", "DR", "OC", "OD", "UA":
120 				order.setStatus(ServiceRequestStatus.REVOKED);
121 				break;
122 			case "DE":
123 				order.setStatus(ServiceRequestStatus.ENTEREDINERROR);
124 				break;
125 			case "HD", "HR", "OH":
126 				order.setStatus(ServiceRequestStatus.ONHOLD);
127 				break;
128 			default:
129 				order.setStatus(ServiceRequestStatus.UNKNOWN);
130 				break;
131 			}
132 		}
133 	}

135 	private void addOrderIdentifier(Identifier ident, CodeableConcept type) {
136 		if (ident.getType() != type) {
137 			ident = ident.copy();
138 			ident.setType(type);
139 		}
140 		order.addIdentifier(ident);
141 		if (izDetail.hasImmunization()) {
142 			izDetail.getImmunization().addIdentifier(ident);
143 		} else if (izDetail.hasRecommendation() && Codes.FILLER_ORDER_IDENTIFIER_TYPE == type) {
144 			// Use ORC-3 as the ImmunizationRecommendation identifier
145 			ImmunizationRecommendation recommendation = izDetail.getImmunizationRecommendation();
146 			if (recommendation != null && !recommendation.hasIdentifier()) {
147 				recommendation.addIdentifier(ident);
148 			}
149 		}
150 	}

196 				break;

198 				order.setStatus(ServiceRequestStatus.REVOKED);
199 				break;

201 				order.setStatus(ServiceRequestStatus.COMPLETED);
202 				break;
203 			case "HD":
204 				order.setStatus(ServiceRequestStatus.ONHOLD);
205 				break;
206 			case "ER", "RP":
207 			default:
208 				break;
209 			}
210 		}

254 	/**
255 	 * Get the ordering PractitionerRole or create one if it
256 	 * does not already exist.
257 	 * @return the ordering PractitionerRole 
258 	 */
259 	private PractitionerRole getRequester() {
260 		if (requester == null) {
261 			requester = createResource(PractitionerRole.class);
262 			requesterRef = ParserUtils.toReference(requester, order, REQUESTER);
263 			order.setRequester(requesterRef);
264 			if (izDetail.hasImmunization()) {
265 				requesterRef = ParserUtils.toReference(requester, izDetail.getImmunization(), "performer", "practitioner");
266 				ImmunizationPerformerComponent perf = izDetail.getImmunization().addPerformer().setActor(requesterRef);
267 				perf.setFunction(Codes.ORDERING_PROVIDER_FUNCTION_CODE);
268 			}
269 		}
270 		return requester;
271 	}

273 	private void updateRequesterName() {
274 		// NOTE: This violates the rule about first name in wins
275 		// for PractitionerRole, but that's because their name
276 		// has to dynamically change to account for the fact that
277 		// they are built in pieces.
278 		StringBuilder b = new StringBuilder();
279 		if (requester.hasPractitioner()) {
280 			b.append(requester.getPractitioner().getDisplay());
281 			if (requester.hasOrganization()) {
282 				b.append(" @ ");
283 			}
284 		}
285 		if (requester.hasOrganization()) {
286 			b.append(requester.getOrganization().getDisplay());
287 		}
288 		requesterRef.setDisplay(b.toString());
289 	}

304 	@ComesFrom(path = "ServiceRequest.requester.PractitionerRole.organization", field = 21) 
305 	public void setOrderingOrganization(Organization organization) {
306 		if (izDetail.getRequestingOrganization() != null) {
307 			// If organization already exists, just copy name and identifier from XON
308 			izDetail.getRequestingOrganization().setNameElement(organization.getNameElement());
309 			izDetail.getRequestingOrganization().setIdentifier(organization.getIdentifier());
310 		} else {
311 			izDetail.setRequestingOrganization(addResource(organization));
312 		}
313 		getRequester();
314 		Reference orgReference = ParserUtils.toReference(izDetail.getRequestingOrganization(), requester, REQUESTER);
315 		requester.setOrganization(orgReference);
316 		updateRequesterName();
317 	}
