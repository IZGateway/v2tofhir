5   import org.hl7.fhir.instance.model.api.IBaseResource;
6   
7   import java.util.ArrayList;
8   import org.hl7.fhir.r4.model.CodeableConcept;
9   import org.hl7.fhir.r4.model.Immunization;
10  import gov.cdc.izgw.v2tofhir.annotation.ComesFrom;
11  import gov.cdc.izgw.v2tofhir.annotation.Produces;
12  import gov.cdc.izgw.v2tofhir.converter.MessageParser;

14  /**
15   * RXRParser parses any RXR segments in a message to an Immunization if the message is a VXU or an 
16   * response to an immunization query.
17   * 
18   * @author Audacious Inquiry
19   *
20   * @see <a href="https://hl7.org/fhir/uv/v2mappings/2024Jan/ConceptMap-segment-rxr-to-immunization.html">V2 to FHIR: RXR to Immunization</a>
21   */
22  @Produces(segment = "RXR", resource=Immunization.class)
23  public class RXRParser extends AbstractSegmentParser {
24  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
25  	private IzDetail izDetail;
26  	
27  	/**
28  	 * Create an RXA Parser for the specified MessageParser
29  	 * @param p	The message parser.
30  	 */
31  	public RXRParser(MessageParser p) {
32  		super(p, "RXR");
33  		if (fieldHandlers.isEmpty()) {
34  			FieldHandler.initFieldHandlers(this, fieldHandlers);
35  		}
36  	}
37  
38  	@Override
39  	protected List<FieldHandler> getFieldHandlers() {
40  		return fieldHandlers;
41  	}
42  
43  	@Override
44  	public IBaseResource setup() {
45  		izDetail = IzDetail.get(getMessageParser());
46  		return izDetail.getImmunization();
47  	}
48  	
49  	/*
50  	 * 1	RXR-1	Route	CWE	1	1				Immunization.route		Immunization.CodeableConcept	0	1	CWE[CodeableConcept]	RouteOfAdministration
51  	 */
52  	/**
53  	 * Set the route of administration
54  	 * @param route the route of administration
55  	 */
56  	@ComesFrom(path = "Immunization.route", field = 1, table = "0162", comment = "Route")
57  	public void setRoute(CodeableConcept route) {
58  		if (izDetail.hasImmunization()) {
59  			izDetail.getImmunization().setRoute(route);
60  		}
61  	}
62  	/*
63  	 * 2	RXR-2	Administration Site	CWE	0	1				Immunization.site		Immunization.CodeableConcept	0	1	CWE[CodeableConcept]	AdministrationSite		
64  	 */
65  	/**
66  	 * Set the site of administration
67  	 * @param administrationSite the route of administration
68  	 */
69  	@ComesFrom(path = "Immunization.site", field = 2, table = "0163", comment = "Administration Site")
70  	public void setAdministrationSite(CodeableConcept administrationSite) {
71  		if (izDetail.hasImmunization()) {
72  			izDetail.getImmunization().setSite(administrationSite);
73  		}
74  	}
75  }
