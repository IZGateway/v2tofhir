46  /**
47   * The QPD Parser generates a Parameters resource documenting the QPD parameters found in QPD-3 to QPD-*
48   * 
49   * For queries (message type of QBP):
50   * 
51   * The Bundle.entry containing this Parameters resource in Bundle.entry.resource will contain a 
52   * Bundle.entry.request element with the method set to GET and the url set to the FHIR _search URL which would perform the query.
53   * 
54   * For responses to queries (message type of RSP):
55   * 
56   * The Bundle.entry containing this Parameters resource in Bundle.entry.resource will contain a 
57   * Bundle.entry.response element with the location set to the FHIR _search URL which would perform the query.
58   *
59   * @author Audacious Inquiry
60   *
61   */
62  @Produces(segment="QPD", resource=Parameters.class)
63  @Slf4j

140 	@Override
141 	public IBase parse(Segment seg) {
142 		Type query = V2Utils.getField(seg, 1);
143 		CodeableConcept queryName = DatatypeConverter.toCodeableConcept(query);
144 		if (queryName == null) {
145 			try {
146 				getParser().warn("Cannot parse unnamed query for: " + seg.encode());
147 			} catch (HL7Exception e) {
148 				// Ignore it.
149 			}
150 			return null;
151 		}
152 		IBase b = setup();
153 		
154 		params.getMeta().addTag(queryName.getCodingFirstRep());
155 		
156 		String queryTag = ParserUtils.toString(V2Utils.getField(seg, 2));
157 		String[] parameters = getQueryParameters(queryName);
158 		params.getMeta().addTag("QueryTag", queryTag, null);
159 
160 		if (parameters.length == 0) {
161 			getParser().warn("Cannot parse {} query", query);
162 			return b;
163 		}
164 		String request = getSearchRequest(seg, parameters);
165 		
166 		BundleEntryComponent entry = (BundleEntryComponent) params.getUserData(BundleEntryComponent.class.getName());
167 		Boolean isResponse = isResponse();
168 		if (entry != null && Boolean.TRUE.equals(isResponse)) {
169 			// Was this QPD in response to a query message, or was it the query itself.
170 			BundleEntryResponseComponent resp = entry.getResponse();
171 			params.setUserData(BundleEntryResponseComponent.class.getName(), resp);
172 			resp.setLocation(request);
173 			resp.setLastModified(getBundle().getTimestamp());
174 		} else if (entry != null && Boolean.FALSE.equals(isResponse)) {
175 			BundleEntryRequestComponent req = entry.getRequest();
176 			params.setUserData(BundleEntryRequestComponent.class.getName(), req);
177 			req.setMethod(HTTPVerb.GET);
178 			req.setUrl(request);
179 		}
180 		// Always add the search translation to params.
181 		params.addParameter("_search", request);
182 		return b;
183 	}

185 	private String getSearchRequest(Segment seg, String[] parameters) {
186 		StringBuilder request = null;
187 		request = new StringBuilder("/fhir/");
188 		request.append(parameters[3]).append("?");
189 		for (int i = 3, offset = 4; i <= seg.numFields() && offset < parameters.length; i++, offset += 4) {
190 			Type[] types = V2Utils.getFields(seg, i); // Get the fields
191 			for (Type t : types) {
192 				String fhirType = parameters[offset + 3];
193 				t = adjustIfVaries(t, parameters[offset + 1]);
194 				org.hl7.fhir.r4.model.Type converted = DatatypeConverter.convert(fhirType, t, null);
195 				addQueryParameter(request, params, parameters[offset + 2], converted);
196 			}
197 		}
198 		// Remove any trailing ? or &
199 		request.setLength(request.length() - 1);
200 		return request.toString();
201 	}

203 	private Boolean isResponse() {
204 		if (mh != null && mh.getMeta().hasTag()) {
205 			if (mh.getMeta().getTag().stream().anyMatch(c -> c.hasCode() && "QBP".equals(c.getCode()))) {
206 				return true;
207 			} else if (mh.getMeta().getTag().stream().anyMatch(c -> c.hasCode() && "RSP".equals(c.getCode()))) {
208 				return false;
209 			}
210 		}
211 		return null; // NOSONAR: null = unknown
212 	}

214 	/**
215 	 * Some segments can have varying field types, for example, QPD.  This method
216 	 * allows the field values to be adjusted to the appropriate V2 type.
217 	 *  
218 	 * @param t	The type to adjust.
219 	 * @param typeName The name of the field type to adjust to.
220 	 * @return	The adjusted type if adjustment is needed, otherwise the original type.
221 	 */
222 	private Type adjustIfVaries(Type t, String typeName) {
223 		if (t instanceof Varies v) {
224 			t = v.getData();
225 		}
226 		if (t instanceof GenericComposite) {
227 			// Well, this is unfortunate, we need to do some classloading shenanigans.
228 			t = convertTo(t, typeName);
229 		}
230 		return t;
231 	}

233 	private Type convertTo(Type t, String typeName) {
234 		Class<? extends Type> target = resolve("ca.uhn.hl7v2.model.v281.datatype", typeName);
235 		if (target == null) {
236 			target = resolve("ca.uhn.hl7v2.model.v251.datatype", typeName);
237 		}
238 		try {
239 			Type newType = target.getDeclaredConstructor(Message.class).newInstance(t.getMessage());
240 			newType.parse(t.encode());
241 			return newType;
242 		} catch (InstantiationException | IllegalAccessException | IllegalArgumentException | InvocationTargetException
243 				| NoSuchMethodException | SecurityException e) {
244 			log.warn("Cannot contruct new " + target.getClass().getName());
245 			return t;
246 		} catch (HL7Exception e) {
247 			log.warn("Cannot parse " + t + " into " + target.getClass().getName());
248 			return t;
249 		}
250 	}

252 	Class<? extends Type> resolve(String packageName, String typeName) {
253 		String className = packageName + "." + typeName;
254 		try {
255 			@SuppressWarnings("unchecked")
256 			Class<? extends Type> clazz = (Class<? extends Type>) Type.class.getClassLoader().loadClass(className);
257 			return clazz;
258 		} catch (ClassNotFoundException e1) {
259 			log.warn("Cannot resolve " + className);
260 			return null;
261 		}
262 	}

264 	private void addQueryParameter(  // NOSONAR: Big switch OK
265 		StringBuilder request, 
266 		Parameters params, 
267 		String name, 
268 		org.hl7.fhir.r4.model.Type converted
269 	) {
270 		if (converted == null) {
271 			return;
272 		}
273 		boolean used = false;
274 		switch (converted.getClass().getSimpleName()) {
275 		case "Address":
276 			Address addr = (Address) converted;
277 			for (StringType line: addr.getLine()) {
278 				appendParameter(request, line, name);
279 			}
280 			used = addParam(request, addr.getCity(), IzQuery.PATIENT_ADDRESS_CITY);
281 			used |= addParam(request, addr.getState(), IzQuery.PATIENT_ADDRESS_STATE);
282 			used |= addParam(request, addr.getPostalCode(), IzQuery.PATIENT_ADDRESS_POSTAL);
283 			used |= addParam(request, addr.getCountry(), IzQuery.PATIENT_ADDRESS_COUNTRY);
284 			break;
285 		case "HumanName":
286 			HumanName hn = (HumanName) converted;
287 			if (name.equals(IzQuery.PATIENT_MOTHERS_MAIDEN_NAME)) {
288 				used = addParam(request, hn.getFamily(), IzQuery.PATIENT_MOTHERS_MAIDEN_NAME); 
289 				// support multiple values for given
290 				for (StringType given : hn.getGiven()) {
291 					used |= appendParameter(request, given, IzQuery.PATIENT_MOTHERS_MAIDEN_NAME + "-given");
292 				}
293 				used |= addParam(request, hn.getSuffixAsSingleString(), IzQuery.PATIENT_MOTHERS_MAIDEN_NAME + "-suffix");
294 			} else {
295 				used = addParam(request, hn.getFamily(), IzQuery.PATIENT_NAME_FAMILY); 
296 				// support multiple values for given
297 				for (StringType given : hn.getGiven()) {
298 					used |= appendParameter(request, given, IzQuery.PATIENT_NAME_GIVEN);
299 				}
300 				used |= addParam(request, hn.getSuffixAsSingleString(), IzQuery.PATIENT_NAME_SUFFIX);
301 			}
302 			break;
303 		case "Identifier":
304 			Identifier ident = (Identifier) converted;
305 			String value = ident.hasSystem() ? ident.getSystem() + "|" : "";
306 			if (ident.hasValue()) {
307 				value += ident.getValue();
308 			}
309 			used = addParam(request, value, name);
310 			break;
311 		case "DateType":
312 			DateType dt = new DateType(((BaseDateTimeType)converted).getValue());
313 			used = addParam(request, dt.asStringValue(), name);
314 			break;
315 		case "ContactPoint":
316 			ContactPoint cp = (ContactPoint)converted;
317 			if (!cp.hasSystem() || cp.getSystem().toString().equals(name)) {
318 				used = addParam(request, cp.getValue(), name);
319 			}
320 			break;
321 		case "StringType":
322 			StringType string = (StringType)converted;
323 			used = addParam(request, string.asStringValue(), name);
324 			break;
325 		case "CodeType":
326 			CodeType code = (CodeType)converted;
327 			used = addParam(request, code.asStringValue(), name);
328 			break;
329 		default:
330 			break;
331 		}
332 		if (used) {
333 			params.addParameter().setName(name).setValue(converted);
334 		}
335 	}

337 	private boolean appendParameter(StringBuilder request, StringType value, String name) {
338 		return addParam(request, value.asStringValue(), name);
339 	}
340 	private boolean addParam(StringBuilder request, String value, String name) {
341 		if (!StringUtils.isBlank(value)) {
342 			request
343 				.append(name)
344 				.append("=")
345 				.append(URLEncoder.encode(value.trim(), StandardCharsets.UTF_8))
346 				.append("&");
347 			return true;
348 		}
349 		return false;
350 	}

352 	private String[] getQueryParameters(CodeableConcept queryName) {
353 		for (String[] names: v2QueryNames) {
354 			if (queryName.getCoding().stream().anyMatch(coding -> codingMatches(coding, names))) {
355 				return names;
356 			}
357 		}
358 		return new String[0];
359 	}
360 	
361 	boolean codingMatches(Coding coding, String[] names) {
362 		return StringUtils.isBlank(names[0]) || names[0].equals(coding.getCode());  
363 	}
