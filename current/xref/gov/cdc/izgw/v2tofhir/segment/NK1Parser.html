32   * The PV1 segment populates a related person resource
33   * 
34   * @see <a href=
35   *      "https://build.fhir.org/ig/HL7/v2-to-fhir/ConceptMap-segment-nk1-to-relatedperson.html">V2
36   *      to FHIR: PV1 to Related Person</a>
37   * @author Audacious Inquiry
38   */
39  @Produces(segment = "NK1", resource = RelatedPerson.class, extra = { Patient.class })
40  @Slf4j
41  public class NK1Parser extends AbstractSegmentParser {
42  	private RelatedPerson relatedPerson;
43  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
44

46  		log.debug("{} loaded", MRGParser.class.getName());

49  	/**
50  	 * Construct a new PV1Parser
51  	 * 
52  	 * @param messageParser The messageParser to construct this parser for.
53  	 */
54  	public NK1Parser(MessageParser messageParser) {
55  		super(messageParser, "NK1");
56  		if (fieldHandlers.isEmpty()) {
57  			FieldHandler.initFieldHandlers(this, fieldHandlers);
58  		}
59  	}
60  
61  	@Override
62  	public List<FieldHandler> getFieldHandlers() {
63  		return fieldHandlers;
64  	}
65  
66  	@Override
67  	public IBaseResource setup() {
68  		relatedPerson = createResource(RelatedPerson.class);
69  		Patient patient = getLastResource(Patient.class);
70  		if (patient == null) {
71  			// patient may not exist if PID somehow failed or wasn't parsed.
72  			// so create one so that it can be referenced.
73  			patient = createResource(Patient.class);
74  		}
75  		relatedPerson.setPatient(ParserUtils.toReference(patient, relatedPerson, "patient"));
76  		return relatedPerson;
77  	}

89  	/**
90  	 * Set the relationship
91  	 * @param relationship the relationship
92  	 */
93  	@ComesFrom(path = "RelatedPerson.relationship", field = 3, table = "0063", map = "Relationship", comment = "Relationship")
94  	@ComesFrom(path = "RelatedPerson.relationship", field = 7, table = "0131", map = "ContactRole", comment = "Contact Role")
95  	public void setRelationship(CodeableConcept relationship) {
96  		CodeableConcept mappedConcept = new CodeableConcept(); 
97  		TerminologyMapper mapper = TerminologyMapperFactory.get();
98  		for (Coding coding: relationship.getCoding()) {
99  			if (!coding.hasSystem() || !coding.hasCode()) {
100 				continue;
101 			}
102 			String mappingName;
103 			if (coding.getSystem().endsWith("0063")) {
104 				mappingName = "Relationship";
105 			} else if (coding.getSystem().endsWith("0131")) {
106 				mappingName = "ContactRole";
107 			} else {
108 				continue;
109 			}
110 			
111 			mapper.mapCode(mappingName, coding)
112 				.ifPresent(result -> mappedConcept.addCoding(result.resultCoding()));
113 		}
114 		// Add the mapped codes first, since they are the preferred concepts.
115 		if (!mappedConcept.isEmpty()) {
116 			relatedPerson.addRelationship(mappedConcept);
117 		}
118 		// Also add the original codes.
119 		relatedPerson.addRelationship(relationship);
120 	}
