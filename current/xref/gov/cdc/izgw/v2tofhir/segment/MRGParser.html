3   import java.util.ArrayList;
4   import java.util.List;
5   
6   import org.hl7.fhir.instance.model.api.IBaseResource;
7   import org.hl7.fhir.r4.model.Account;
8   import org.hl7.fhir.r4.model.Encounter;
9   import org.hl7.fhir.r4.model.HumanName;
10  import org.hl7.fhir.r4.model.Identifier;
11  import org.hl7.fhir.r4.model.Patient;
12  import gov.cdc.izgw.v2tofhir.annotation.ComesFrom;
13  import gov.cdc.izgw.v2tofhir.annotation.Produces;
14  import gov.cdc.izgw.v2tofhir.converter.MessageParser;
15  import gov.cdc.izgw.v2tofhir.utils.ParserUtils;
16  import lombok.extern.slf4j.Slf4j;
17  
18  /**
19   * Parser for MRG Segments
20   * 
21   * The MRG segment populates a Patient resource with identifiers and demographics for
22   * a patient which would need to be merged into a single patiennt. 
23   * 
24   * @author Audacious Inquiry
25   */
26  @Produces(segment="MRG", resource=Patient.class, extra = { Account.class, Encounter.class })
27  @Slf4j
28  public class MRGParser extends AbstractSegmentParser {
29  	private Patient patient;
30  	private Account account;
31  	private Encounter encounter;
32  	
33  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
34  
35  	static {
36  		log.debug("{} loaded", MRGParser.class.getName());
37  	}
38  	
39  	/**
40  	 * Construct a new MRGParser
41  	 * 
42  	 * @param messageParser	The messageParser to construct this parser for.
43  	 */
44  	public MRGParser(MessageParser messageParser) {
45  		super(messageParser, "MRG");
46  		if (fieldHandlers.isEmpty()) {
47  			FieldHandler.initFieldHandlers(this, fieldHandlers);
48  		}
49  	}
50  
51  	
52  	@Override
53  	public List<FieldHandler> getFieldHandlers() {
54  		return fieldHandlers;
55  	}
56  	
57  	@Override
58  	public IBaseResource setup() {
59  		// The MRG segment is used with an existing patient recorded
60  		// in the PID segment, so get the most recently created Patient
61  		// resource.
62  		patient = getLastResource(Patient.class);
63  		if (patient == null) {
64  			patient = createResource(Patient.class);
65  		}
66  		account = null;
67  		encounter = null;
68  		return patient;
69  	}
70  	
71  	/*
72  	The MRG segment provides receiving applications with information necessary to initiate the merging of patient data as well as groups of records. It is intended that this segment be used throughout the Standard to allow the merging of registration, accounting, and clinical records within specific applications.
73  
74  	FIELD	LENGTH	DATA TYPE	OPTIONALITY	REPEATABILITY	TABLE	
75  	MRG.1 - Prior Patient Identifier List	250	CX	R	∞		
76  	MRG.2 - Prior Alternate Patient ID	250	CX	B	∞		
77  	MRG.3 - Prior Patient Account Number	250	CX	O	-		
78  	MRG.4 - Prior Patient ID	250	CX	B	-		
79  	MRG.5 - Prior Visit Number	250	CX	O	-		
80  	MRG.6 - Prior Alternate Visit ID	250	CX	O	-		
81  	MRG.7 - Prior Patient Name	250	XPN
82  	*/
83  	/**
84  	 * Add a patient identifier
85  	 * @param ident the patient identifier
86  	 */
87  	@ComesFrom(path="Patient.identifier", field = 1, comment = "Prior Patient Identifier List")
88  	@ComesFrom(path="Patient.identifier", field = 2, comment = "Prior Alternate Patient ID")
89  	@ComesFrom(path="Patient.identifier", field = 4, comment = "Prior Patient ID")
90  	public void addPatientIdentifier(Identifier ident) {
91  		patient.addIdentifier(ident);
92  	}
93  	
94  	/**
95  	 * Add a patient account identifier
96  	 * Creates a new account.  Adds a reference from the Encounter to the account.
97  	 * Adds a reference from the encounter and the account to the patient as the subject.
98  	 * @param accountId	The account identifier
99  	 */
100 	@ComesFrom(path="Account.identifier", field = 3, comment = "Prior Patient Account Number")
101 	public void addPatientAccount(Identifier accountId) {
102 		account = createResource(Account.class);
103 		account.addIdentifier(accountId);
104 		account.addSubject(ParserUtils.toReference(patient, account, "subject", "patient"));
105 		account.getMeta().addTag();
106 		encounter = getEncounter();
107 		encounter.addAccount(ParserUtils.toReference(account, encounter, "account"));
108 	}
109 
110 	/**
111 	 * Create or return the encounter associated with this segment.
112 	 * If an encounter has already been created, returns it. Otherwise
113 	 * creates the encounter and connects the encounter to the patient.
114 	 * @return The encounter associated with this segment
115 	 */
116 	private Encounter getEncounter() {
117 		if (encounter == null) {
118 			encounter = createResource(Encounter.class);
119 			encounter.setSubject(ParserUtils.toReference(patient, encounter, "subject", "patient"));
120 		}
121 		return encounter;
122 	}
123 	
124 	/**
125 	 * Add an identifier to the encounter.
126 	 * Creates or gets the existing encounter and adds the identifier to it.
127 	 * @param encounterId The encounter identifier.
128 	 */
129 	@ComesFrom(path="Encounter.identifier", field = 5, comment = "Prior Visit Number")
130 	@ComesFrom(path="Encounter.identifier", field = 6, comment = "Prior Alternate Visit ID")
131 	public void addEncounterIdentifier(Identifier encounterId) {
132 		encounter = getEncounter();
133 		encounter.addIdentifier(encounterId);
134 	}
135 		
136 	/**
137 	 * Add the name to the patient.
138 	 * @param humanName The patient name
139 	 */
140 	@ComesFrom(path="Patient.name", field = 7, comment = "Patient Name")
141 	public void addPatientName(HumanName humanName) {
142 		patient.addName();
143 	}
144 }
