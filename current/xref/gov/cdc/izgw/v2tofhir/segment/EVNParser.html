3   import java.util.ArrayList;
4   import java.util.List;
5   
6   import org.hl7.fhir.instance.model.api.IBaseResource;
7   import org.hl7.fhir.r4.model.CodeableConcept;
8   import org.hl7.fhir.r4.model.DateTimeType;
9   import org.hl7.fhir.r4.model.Identifier;
10  import org.hl7.fhir.r4.model.InstantType;
11  import org.hl7.fhir.r4.model.Location;
12  import org.hl7.fhir.r4.model.MessageHeader;
13  import org.hl7.fhir.r4.model.Practitioner;
14  import org.hl7.fhir.r4.model.Provenance;
15  
16  import gov.cdc.izgw.v2tofhir.annotation.ComesFrom;
17  import gov.cdc.izgw.v2tofhir.annotation.Produces;
18  import gov.cdc.izgw.v2tofhir.converter.MessageParser;
19  import gov.cdc.izgw.v2tofhir.utils.ParserUtils;
20  import lombok.extern.slf4j.Slf4j;
21  
22  /**
23   * The EVNParser extracts information from the EVN segment and adds it to the
24   * ProvenanceResource for the MessageHeader.
25   *
26   * @author Audacious Inquiry
27   *
28   */
29  @Produces(segment = "EVN", resource = Provenance.class)
30  @Slf4j
31  public class EVNParser extends AbstractSegmentParser {
32  	Provenance provenance;
33  	private static List<FieldHandler> fieldHandlers = new ArrayList<>();
34  	static {
35  		log.debug("{} loaded", EVNParser.class.getName());
36  	}
37  
38  	/**
39  	 * Construct a new DSC Parser for the given messageParser.
40  	 * 
41  	 * @param messageParser The messageParser using this QPDParser
42  	 */
43  	public EVNParser(MessageParser messageParser) {
44  		super(messageParser, "EVN");
45  		if (fieldHandlers.isEmpty()) {
46  			FieldHandler.initFieldHandlers(this, fieldHandlers);
47  		}
48  	}
49  
50  	@Override
51  	public List<FieldHandler> getFieldHandlers() {
52  		return fieldHandlers;
53  	}
54  
55  	public IBaseResource setup() {
56  		MessageHeader mh = getFirstResource(MessageHeader.class);
57  		if (mh != null) {
58  			provenance = (Provenance) mh.getUserData(Provenance.class.getName());
59  		} else {
60  			// Create a standalone provenance resource for segment testing
61  			provenance = createResource(Provenance.class);
62  		}
63  		return provenance;
64  	}
65  
66  	/**
67  	 * Set the recorded date time.
68  	 * @param recordedDateTime The time the event was recorded.
69  	 */
70  	@ComesFrom(path = "Provenance.recorded", field = 2, comment = "Recorded Date/Time")
71  	public void setRecordedDateTime(InstantType recordedDateTime) {
72  		provenance.setRecordedElement(recordedDateTime);
73  	}
74  
75  	/**
76  	 * Set the reason for the event
77  	 * @param eventReasonCode	the reason for the event
78  	 */
79  	@ComesFrom(path = "Provenance.reason", field = 4, comment = "Event Reason Code")
80  	public void setEventReasonCode(CodeableConcept eventReasonCode) {
81  		provenance.addReason(eventReasonCode);
82  	}
83  
84  	/**
85  	 * Set the operator
86  	 * @param operatorId	The operator
87  	 */
88  	@ComesFrom(path = "Provenance.agent.who.Practitioner", field = 5, comment = "Operator ID")
89  	public void setOperatorID(Practitioner operatorId) {
90  		provenance.addAgent().setWho(ParserUtils.toReference(operatorId, provenance, "agent"));
91  	}
92  
93  	/**
94  	 * Set the date/time of the event
95  	 * @param eventOccurred the occurence date/time of the event
96  	 */
97  	@ComesFrom(path = "Provenance.occurredDateTime", field = 6, comment = "Event Occurred")
98  	public void setEventOccurred(DateTimeType eventOccurred) {
99  		provenance.setOccurred(eventOccurred);
100 	}
101 
102 	/**
103 	 * Set the identifier of the facility where the event occurred
104 	 * @param eventFacility The event facility identifier
105 	 */
106 	@ComesFrom(path = "Provenance.location", field = 7, comment = "Event Facility")
107 	public void setEventFacility(Identifier eventFacility) {
108 		if (eventFacility.hasSystem()) {
109 			Location location = createResource(Location.class);
110 			location.setName(eventFacility.getSystem());
111 			location.addIdentifier(eventFacility);
112 			provenance.setLocation(ParserUtils.toReference(location, provenance, "location"));
113 		}
114 	}
115 }
