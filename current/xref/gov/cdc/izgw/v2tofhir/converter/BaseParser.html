33  /**
34   * The base parser class for V2toFHIR conversion. Abstracts the parsing from the message and message component detail.
35   * 
36   * @author Audacious Inquiry
37   *
38   * @param <U>	The message unit for conversion.
39   * @param <S>	The subunits that get converted.
40   */
41  @Slf4j
42  public abstract class BaseParser<U,S> implements Parser<U,S> {

44  	static final Context<Parser<?, ?>> defaultContext = new Context<>(null);
45  	/** The originalText extension */
46  	public static final String ORIGINAL_TEXT = "http://hl7.org/fhir/StructureDefinition/originalText";
47  	/** Locations to look for parser packages */
48  	protected static List<String> packageSources = new ArrayList<>();
49  	/**
50  	 * Enable or disable storing of Provenance information for created resources.
51  	 * 
52  	 * When enabled, the MessageParser will create Provenance resources for each created resource
53  	 * that document the sources of information used to create resources in the generated bundle.
54  	 * These provenance records tie the information in the resource back to the specific segments
55  	 * or groups within the message that sourced the data in the generated resource. 
56  	 * 
57  	 * @param storingProvidence	Set to true to enable generation of Provenance resources, or false to disable them.
58  	 */
59  	public static void setStoringProvenance(boolean storingProvidence) {
60  		defaultContext.setStoringProvenance(storingProvidence);
61  	}
62  	/**
63  	 * Returns true if Provenance resources are to be created, false otherwise.
64  	 * @return true if Provenance resources are to be created, false otherwise.
65  	 */
66  	public static boolean isStoringProvenance() {
67  		return defaultContext.isStoringProvenance();
68  	}

129 		Bundle b = getContext().getBundle();
130 		if (b == null) {
131 			b = new Bundle();
132 			b.setId(new IdType(b.fhirType() + "/" + idGenerator.get()));
133 			b.setType(BundleType.MESSAGE);

155 	/**
156 	 * Create a Bundle by parsing a group of structures
157 	 * @param structures	The structures to parse.
158 	 * @return	The generated Bundle
159 	 */
160 	public Bundle createBundle(Iterable<S> structures) {
161 		Bundle b = getBundle();
162 		processed.clear();
163 		for (S structure: structures) {
164 			updated.clear();
165 			processUnit(structure);
166 		}
167 		Normalizer.normalizeResources(b);
168 		return sortProvenance(b);
169 	}

171 	/**
172 	 * Process the content of a unit
173 	 * @param structure The structure to parse
174 	 * @return the constructed FHIR Resource or type
175 	 */
176 	public IBase processUnit(S structure) {
177 		if (processed.containsKey(structure)) {
178 			// This structure was already processed
179 			log.info("{} processed by {}", getName(structure), processed.get(structure));
180 			return null;
181 		}
182 		
183 		processor = getParser(getName(structure));
184 		if (processor == null) {
185 			return null;
186 		} 
187 		
188 		IBase b = null;
189 		try {
190 			beforeParsing(structure);
191 			b = processor.parse(structure);
192 		} catch (Exception e) {
193 			warn("Unexpected {} parsing {}: {}", e.getClass().getSimpleName(), getName(structure), e.getMessage(), e);
194 		} finally {
195 			afterParsing(structure);
196 			addProcessed(structure);
197 			processor.finish();
198 		}
199 
200 		if (getContext().isStoringProvenance()) {
201 			// Provenance is updated by segments when we did work and are storing provenance
202 			updateProvenance(structure);
203 		}
204 		return b;
205 	}

242 	/**
243 	 * Update the provenance of a component 
244 	 * @param segment The component whose provenance needs to be updated
245 	 */
246 	protected void updateProvenance(S segment) {
247 		for (IBaseResource r: resourcesNeedingProvenance) {
248 			Provenance p = (Provenance) r.getUserData(Provenance.class.getName());
249 			if (p == null) {
250 				continue;
251 			}
252 			ProvenanceEntityComponent entity = p.getEntityFirstRep();
253 			try {
254 				String encodedSegment = encode(segment);
255 				StringType originalText = new StringType(encode(segment)); // Was new StringType(context.getHl7DataId()+"#"+TerserPathUtils.getTerserPath(segment))
256 				if (entity.getExtension().stream()
257 						.noneMatch(ext -> 
258 							BaseParser.ORIGINAL_TEXT.equals(ext.getUrl()) && 
259 							encodedSegment.equals(ext.getValue().toString())
260 						)
261 				) {
262 					entity.addExtension().setUrl(BaseParser.ORIGINAL_TEXT).setValue(originalText);
263 					IBaseMetaType meta = r.getMeta();
264 					if (meta instanceof Meta m4) {
265 						m4.getSourceElement().addExtension().setUrl(BaseParser.ORIGINAL_TEXT).setValue(originalText);
266 					} 
267 				}
268 			} catch (Exception e) {
269 				warn("Unexpected {} updating provenance for {} segment: {}", e.getClass().getSimpleName(), getName(segment), e.getMessage());
270 			}
271 		}
272 		resourcesNeedingProvenance.clear();
273 	}

275 	/**
276 	 * This method sorts Provenance resources to the end.
277 	 * @param b	The bundle to sort
278 	 * @return	The sorted bundle
279 	 */
280 	public Bundle sortProvenance(Bundle b) {
281 		if (getContext().isStoringProvenance()) {
282 			List<BundleEntryComponent> list = b.getEntry();
283 			int len = list.size();
284 			for (int i = 0; i < len; ++i) {
285 				BundleEntryComponent c = list.get(i);
286 				if (c == null || !c.hasResource()) {
287 					continue;
288 				}
289 				if ("Provenance".equals(c.getResource().fhirType())) {
290 					list.add(c);	// Add this provenance to the end
291 					list.remove(i);	// And remove it from the current position
292 					--i;			// NOSONAR: Backup a step to reprocess the new resource in this position.
293 					--len;			// We don't need to look at entries we just shuffled around to the end
294 				}
295 			}
296 		}
297 		return b;
298 	}
299 	/**
300 	 * Load a parser for a specified segment type.
301 	 * @param segment The segment to get a parser for
302 	 * @return The parser for that segment.
303 	 */
304 	public Processor<U, S> getParser(String segment) {
305 		Class<Processor<U, S>> p = parsers.get(segment);
306 		@SuppressWarnings("unchecked")
307 		Class<Processor<U, S>> safeNull = (Class<Processor<U,S>>) Processor.NULL_PROCESSOR.getClass();
308 		if (p == null) {
309 			p = loadParser(segment);
310 			if (p == null) {
311 				// Report inability to load ONCE if context says to.
312 				if (getContext().isWarnOnFailedLoad()) {
313 					log.warn("Cannot load parser for {}", segment);
314 				}
315 				// Save the fact that there is no parser for this segment, have to do this dynamically
316 				// to store a do-nothing marker class.
317 				parsers.put(segment, safeNull);
318 				return null;
319 			}
320 			parsers.put(segment, p);
321 		} else if (Processor.NULL_PROCESSOR_CLASS.equals(p)) {
322 			return null;
323 		}
324 		
325 		try {
326 			Constructor<?>[] ctors = p.getDeclaredConstructors();
327 			for (Constructor<?> ctor : ctors) {
328 				if (ctor.getParameterCount() == 1 && ctor.getParameterTypes()[0].isInstance(this)) {
329 					Object x = ctor.newInstance(this);
330 					@SuppressWarnings("unchecked")
331 					Processor<U,S> instance = (Processor<U,S>)x;
332 					return instance;
333 				}
334 			}
335 			log.error("No constructor found using {} while creating {} for {}", this.getClass().getSimpleName(), p.getName(), segment);
336 			parsers.put(segment, safeNull);
337 			return null;
338 		} catch (InstantiationException | IllegalAccessException | IllegalArgumentException
339 				| InvocationTargetException | SecurityException | ClassCastException e) {
340 			log.error("Unexpected {} while creating {} for {}", e.getClass().getSimpleName(), p.getName(), segment, e);
341 			return null;
342 		}
343 	}

345 	/** A cache of loaded parser classes */
346 	private Map<String, Class<Processor<U,S>>> parserClasses = new TreeMap<>();
347 
348 	/**
349 	 * Load a parser for the specified segment or group
350 	 * @param name	The name of the segment or group to find a parser for
351 	 * @return	A StructureParser for the segment or group, or null if none found. 
352 	 */
353 	public Class<Processor<U,S>> loadParser(String name) {
354 		name = WordUtils.capitalize(name);
355 		Class<Processor<U,S>> clazz = parserClasses.get(name);
356 		if (clazz != null) {
357 			return clazz;
358 		}
359 		
360 		ClassLoader loader = MessageParser.class.getClassLoader();
361 		for (String packageName : BaseParser.packageSources) {
362 			try {
363 				@SuppressWarnings("unchecked") 
364 				Class<Processor<U,S>> clazz2 = (Class<Processor<U,S>>) loader.loadClass(packageName + "." + name + "Parser");
365 				parserClasses.put(name, clazz2);
366 				return clazz2;
367 			} catch (ClassNotFoundException ex) {
368 				// Keep looking
369 			} catch (Exception ex) {
370 				log.error("{}.{} cannot be loaded: {}", packageName, name, ex.getMessage(), ex);
371 			}
372 		}
373 		return null;
374 	}
