1   package gov.cdc.izgw.v2tofhir.annotation;
2   
3   import java.lang.annotation.Documented;
4   import java.lang.annotation.ElementType;
5   import java.lang.annotation.Repeatable;
6   import java.lang.annotation.Retention;
7   import java.lang.annotation.RetentionPolicy;
8   import java.lang.annotation.Target;
9   
10  import org.hl7.fhir.instance.model.api.IBase;
11  
12  /**
13   * This annotation is used to document V2 to FHIR segment conversions.
14   * 
15   * In the future, it may be used to automate such conversions.
16   * 
17   * @author Audacious Inquiry
18   */
19  
20  @Repeatable(ComesFrom.List.class)
21  @Target({ElementType.TYPE, ElementType.METHOD })
22  @Retention(RetentionPolicy.RUNTIME)
23  @Documented 
24  public @interface ComesFrom {
25  	/**
26  	 * The FHIR Path of the content in the bundle
27  	 * @return	The FHIR Path expression that extracts the content from the Bundle
28  	 */
29  	String path();
30  	/**
31  	 * The FHIR Paths of other content in the bundle
32  	 * @return	The FHIR Path expressions that extract the content from the Bundle
33  	 */
34  	String[] also() default {};
35  
36  	/**
37  	 * The terser paths of the objects that construct the object referenced in path
38  	 * @return	HAPI V2 terser path expressions indicating where the data came from
39  	 */
40  	String[] source() default {};
41  	
42  	/**
43  	 * The 1-based index of the field to extract from.
44  	 * @return the 1-based index of the field to extract from, or 0 if a field 
45  	 */
46  	int field() default 0;
47  	/**
48  	 * The 1-based index of the component to extract from.
49  	 * @return the 1-based index of the component to extract from, or 0 if the component is not used.
50  	 */
51  	int component() default 0;
52  	
53  	/**
54  	 * A comment about the conversion
55  	 * @return The comment
56  	 */
57  	String comment() default "";
58  	
59  	/**
60  	 * The name of the concept map to use when mapping between code systems.
61  	 * @return The name of the concept mape
62  	 */
63  	String map() default "";
64  	
65  	/**
66  	 * The name of the HL7 Table used with this field.
67  	 * @return theh name of the HL7 Table
68  	 */
69  	String table() default "";
70  	
71  	/**
72  	 * A fixed value
73  	 * @return	The fixed value.
74  	 */
75  	String fixed() default "";
76  	
77  	/**
78  	 * The type to convert to if other than the type determined from the path
79  	 * in the FHIR resource. 
80  	 * 
81  	 * @return	The type to convert to, or IBase.class to convert to the 
82  	 * type determined by the path.
83  	 */
84  	Class<? extends IBase> fhir() default IBase.class;
85  	
86  	/**
87  	 * The expected V2 datatype.
88  	 * This may need to be specified in cases where a message has pre-adopted features
89  	 * from a newer version, e.g., as for IHE profiles and immunization messages where
90  	 * some 2.7.1 features were used.
91  	 * 
92  	 * @return The expected V2 datatype.
93  	 */
94  	String type() default "";
95  	
96  	/**
97  	 * Set the priority of this mapping.
98  	 * Mappings are processed in field and component order, unless the priority is changed
99  	 * The higher the priority, the earlier in processing the mapping is performed.
100 	 * @return The priority of the mapping.
101 	 */
102 	int priority() default 0;
103     /**
104      * A list of mapping from FHIR types to HL7 V2 Terser paths
105      */
106     @Retention(RetentionPolicy.RUNTIME)
107     @Target({ ElementType.TYPE, ElementType.METHOD })
108     @Documented
109     @interface List {
110     	/**
111     	 * A list of ComesFrom annotations.
112     	 * @return The values
113     	 */
114         ComesFrom[] value();
115     }
116 }
