1   package gov.cdc.izgw.v2tofhir.utils;
2   
3   import java.lang.ref.WeakReference;
4   import java.util.Arrays;
5   
6   import lombok.extern.slf4j.Slf4j;
7   
8   /**
9    * 
10   * The ErrorReporter interface provides a mechanism for reporting warning messages during processing, so
11   * that different implementations can handle warnings in a manner appropriate to their context.
12   * 
13   * @author Audacious Inquiry
14   */
15  public interface ErrorReporter {
16  	/**
17  	 * The default error reporter just logs the warning message using SLF4J.
18  	 * @author Audacious Inquiry
19  	 */
20  	@Slf4j
21  	static class DefaultErrorReporter {
22  		/** Thread local to allow different threads to have different error reporters.
23  		 *  If not set, the default instance is used, which just logs the warning.
24  		 */
25  		private DefaultErrorReporter() {
26  			// Private constructor to prevent instantiation
27  		}
28  		private static final ThreadLocal<WeakReference<ErrorReporter>> ERROR_REPORTER = new ThreadLocal<>();
29  		/** The default error reporter. It logs each message as a warning. */
30  		public static final ErrorReporter INSTANCE = new ErrorReporter() {
31  			@Override
32  			public void warn(String message, Object... args) {
33  				// Truncate the stack trace to only include relevant frames.
34  				if (args != null && args.length > 0 && args[args.length - 1] instanceof Throwable t) {
35  					truncateStackTrace(t, 5, "com.ainq.", "gov.cdc.", "test.", "org.junit.");
36  				}
37  				// Default is to log the warning.
38  				log.warn(message, args);
39  			}
40  		};
41  	}
42  	
43  	/**
44  	 * Truncate the stack trace of the given throwable to only include frames up to and including
45  	 * the first frame that matches one of the given prefixes, plus a specified number of frames after that.
46  	 * 
47  	 * @param t The throwable whose stack trace is to be truncated.
48  	 * @param framesAfterMatch The number of frames to include after the matching frame.
49  	 * @param prefixes The class name prefixes to match.
50  	 */
51  	default void truncateStackTrace(Throwable t, int framesAfterMatch, String... prefixes) {
52  		StackTraceElement[] trace = t.getStackTrace();
53  		// Default to just the first 10 frames if no match is found.
54  		int newLength = Math.min(trace.length, 10); 
55  		for (int i = 0; i < trace.length; i++) {
56  			StackTraceElement ste = trace[i];
57  			String className = ste.getClassName();
58  			if (Arrays.asList(prefixes).stream().anyMatch(className::startsWith)) {
59  				newLength = Math.min(i + framesAfterMatch, trace.length);
60  				break;
61  			}
62  		}
63  		StackTraceElement[] newTrace = Arrays.copyOfRange(trace, 0, newLength);
64  		t.setStackTrace(newTrace);
65  	}	
66  	/**
67  	 * Set the error reporter for the current thread. If not set, the default reporter is used, which just logs the warning.
68  	 * @param reporter	The error reporter to use. If null, the default reporter is used.
69  	 */
70  	public static void set(ErrorReporter reporter) {
71  		// Use a weak reference to avoid memory leaks if the reporter is no longer needed.
72  		if (reporter == null) {
73  			DefaultErrorReporter.ERROR_REPORTER.remove();
74  			return;
75  		}
76  		WeakReference<ErrorReporter> ref = new WeakReference<>(reporter);
77  		DefaultErrorReporter.ERROR_REPORTER.set(ref);
78  	}
79  	
80  	/**
81  	 * Get the error reporter for the current thread. If not set, the default reporter is used, which just logs the warning.
82  	 * @return The error reporter for the current thread.
83  	 */
84  	public static ErrorReporter get() {
85  		WeakReference<ErrorReporter> ref = DefaultErrorReporter.ERROR_REPORTER.get();
86  		if (ref != null) {
87  			ErrorReporter r = ref.get();
88  			if (r != null) {
89  				return r;
90  			}
91  		}
92  		return DefaultErrorReporter.INSTANCE;
93  	}
94  	
95  	/**
96  	 * Handle a warning message. Implementations may choose to log the message, or throw a runtime exception,
97  	 * to ensure that the calling code is aware of the warning.
98  	 * 
99  	 * @param message The warning message
100 	 * @param args The arguments. If the last argument is a Throwable, it is treated as the cause.
101 	 */
102 	default void warn(String message, Object ... args) {
103 		Object x = args != null && args.length > 0 ? args[args.length - 1] : null;
104 		if (x instanceof Throwable t) {
105 			// Trim the stack trace to only include relevant frames.
106 			StackTraceElement[] stackTrace = t.getStackTrace();
107 			for (int i = 0; i < stackTrace.length; i++) {
108 				StackTraceElement ste = stackTrace[i];
109 				String className = ste.getClassName();
110 				if (className.startsWith("com.ainq") || className.startsWith("gov.cdc")) {
111 					int newLength = Math.min(i + 5, stackTrace.length);
112 					StackTraceElement[] newStackTrace = Arrays.copyOfRange(stackTrace, 0, newLength);
113 					t.setStackTrace(newStackTrace);
114 					break;
115 				}
116 			}
117 		}
118 		get().warn(message, args);
119 	}
120 }
