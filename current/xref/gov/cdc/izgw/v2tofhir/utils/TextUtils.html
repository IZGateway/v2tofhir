34  /**
35   * Utility class to convert FHIR and V2 Objects to and from strings used in text.
36   * The toText() Methods in this class attempt to create human readable strings 
37   * that contain the information in the type in a standardized form as would be commonly used in 
38   * human readable text. 
39   * 
40   * The to{fhirType} and to{v2Type} methods convert strings created by these method (and possibly
41   * other sources) to the specified type.
42   * 
43   * If t = TextUtils.toType(TextUtils.toText(x)), then t is generally equivalent to x but may be missing
44   * some data.  
45   */
46  public class TextUtils {
47  	private TextUtils() {}

49  	/**
50  	 * Convert parts of an address into a text string. This method is the same one used to convert both FHIR and V2 address types
51  	 * into a string so that there is a common string representation.
52  	 * 
53  	 * @param country	The country
54  	 * @param city		The city
55  	 * @param state		The state
56  	 * @param postalCode	The postalCode
57  	 * @param district	The district, county or other geographic designator for the address
58  	 * @param lines	The address lines
59  	 * @return A string representing the address
60  	 */
61  	public static String addressToText(String country, String city, String state, String postalCode, String district, Object ... lines) {
62  		StringBuilder b = new StringBuilder();
63  		for (Object line : lines) {
64  			appendIfNonBlank(b, line, "\n");
65  		}
66  		appendIfNonBlank(b, district, "\n");
67  		
68  		if (Strings.CS.equalsAny(StringUtils.upperCase(country), "MX", "MEX", "MEXICO")) {
69  			appendIfNonBlank(b, postalCode, "  ");
70  			appendIfNonBlank(b, city, ", ");
71  			appendIfNonBlank(b, state, null);
72  		} else {
73  			appendIfNonBlank(b, city, ", ");
74  			appendIfNonBlank(b, state, "  ");
75  			appendIfNonBlank(b, postalCode, null);
76  		}
77  		if (Strings.CS.endsWithAny(b, ", ", "  ")) {
78  			b.setLength(b.length() - 2);
79  		} else if (Strings.CS.endsWith(b, " ")) {
80  			b.setLength(b.length() - 1);
81  		}
82  		if (StringUtils.isNotBlank(b)) {
83  			b.append("\n");
84  		}
85  		// Country goes on a line of its own.
86  		appendIfNonBlank(b, country, "\n");
87  		return b.toString();
88  		
89  	}

91  	/**
92  	 * Convert parts of a coding (or multiple codings) into a text string. This method is the same one used to convert 
93  	 * both FHIR and V2 address types into a string so that there is a common string representation.
94  	 * 
95  	 * The very first element in parts is the "original text" for the coding, and will be the first text that is output, then
96  	 * the coded strings.
97  	 * 
98  	 * The next, and each subsequent group of parts is in the form code, display, system
99  	 * 
100 	 * @param text	The original text that was coded
101 	 * @param parts	The parts of the coding(s). Each coding can have three strings, any of which can be null; the display name,
102 	 * the code, and the system.  These are written in the form:
103 	 * 
104 	 * 	Original Text, Display Name (Coding-System Code) [, Display Name (Coding-System Code)] 
105 	 * 
106 	 * so that a concept describing a heart attack in ICD9CM and SNOMED-CT for Heart Attack would be converted to text as:
107 	 * 
108 	 * 	Heart Attack, Myocardial Infarction (ICD9CM 410.41), Myocardial infarction (SNOMEDCT 22298006)
109 	 * 
110 	 * If original text is missing, it would appear as 
111 	 * 
112 	 * 	Myocardial Infarction (ICD9CM 410.41), Myocardial infarction (SNOMEDCT 22298006)
113 	 * 
114 	 * If the Display name is missing is missing for any code only the text between () is written for each code found 
115 	 * e.g., (ICD9CM 410.41), (SNOMEDCT 22298006)
116 	 * If the coding system is missing for any value, and the code is present, this would be written as (410.41)
117 	 * If the code is missing, this would be written as (ICD9CM ) (note the trailing space).
118 	 * If both coding system and code are missing, nothing appears. 
119 	 * 
120 	 * @return	A string representation of the coding(s)
121 	 */
122 
123 	public static String codingToText(String text, String ... parts) {
124 		if (parts.length == 0) {
125 			return "";
126 		}
127 		
128 		StringBuilder b = new StringBuilder();
129 		appendIfNonBlank(b, text, null);
130 		
131 		for (int i = 0; i < parts.length; i += 3) {
132 			String code = getPart(parts, i);
133 			String display = getPart(parts, i + 1); 
134 			String system = getPart(parts, i + 2);
135 			if (StringUtils.isAllEmpty(code, display, system)) {
136 				continue;
137 			}
138 			if (!b.isEmpty()) {
139 				b.append(", ");
140 			}
141 			appendIfNonBlank(b, display, null);
142 			if (StringUtils.isNotBlank(code) && StringUtils.isNotBlank(system)) {
143 				// For converting to Strings, if system is a URL, make it 
144 				// a more human readable name like people would expect.
145 				system = getHumanReadableSystemName(system);
146 				b.append(" (").append(system).append(" ").append(code).append(")");
147 			} else if (StringUtils.isNotBlank(code)) {
148 				// The space after ( makes it easily parsed as a code with an empty system
149 				b.append(" ( ").append(code).append(")"); 
150 			}
151 		}
152 		String value = b.toString();
153 		if (value.endsWith(",") || value.endsWith(", ")) {
154 			value = StringUtils.substringBeforeLast(value, ",");
155 		}
156 		return value;
157 	}

159 	private static String getHumanReadableSystemName(String system) {
160 		if (!Strings.CS.contains(system, ":")) {
161 			return system;
162 		}
163 		List<String> l = gov.cdc.izgw.v2tofhir.terminology.Systems.getSystemNames(system);
164 		// First name is always preferred name, second is the V2 common name
165 		if (!l.isEmpty()) {
166 			system = l.get(0);
167 			if (l.size() > 1) {
168 				// If there is a V2 name, use that.
169 				system = l.get(1);
170 			}
171 		}
172 		return system;
173 	}

180 	 * Convert a human name to text in the form {prefix} {givens} {family} suffix
181 	 * 
182 	 * @param prefix	The prefix or null if there is none
183 	 * @param family	The family name or null if there is none
184 	 * @param suffix	The suffix or null if there is none
185 	 * @param givens	The given names or null or an empty array if there are none
186 	 * @return a string in the form {prefix} {givens} {family} suffix
187 	 */
188 	public static String humanNameToText(String prefix, String family, String suffix, Object ... givens) {
189 		StringBuilder b = new StringBuilder();
190 		appendIfNonBlank(b, prefix, " ");
191 		if (givens != null) {
192 			for (Object given: givens) {
193 				appendIfNonBlank(b, given, " ");
194 			}
195 		}
196 		appendIfNonBlank(b, family, " ");
197 		appendIfNonBlank(b, suffix, null);
198 		return rightTrim(b).toString();
199 	}

231 	/**
232 	 * Convert a quantity to text in the general form {value} {unit}
233 	 * 
234 	 * @param value	The value
235 	 * @param code The code (presently not used)
236 	 * @param unit The units
237 	 * @return	text in the general form {value} {unit}
238 	 */
239 	public static String quantityToText(String value, String code, String unit) {
240 		StringBuilder b = new StringBuilder();
241 		if (value != null) {
242 			if (value.startsWith("-.")) {
243 				value = "-0." + value.substring(2);
244 			} else if (value.startsWith("+.")) {
245 				value = "0." + value.substring(2);
246 			} else if (value.startsWith(".")) {
247 				value = "0." + value.substring(1);
248 			}
249 		}
250 		appendIfNonBlank(b, value, " ");
251 		if (StringUtils.isBlank(unit)) {
252 			appendIfNonBlank(b, code, null);
253 		} else {
254 			appendIfNonBlank(b, unit, null);
255 		}
256 		return rightTrim(b).toString();
257 	}

259 	/**
260 	 * Convert a V2 Composite Type to a standardize string output: case possible
261 	 * with multiple lines of text representing the content of the type.
262 	 * 
263 	 * @param v2Type The type to convert
264 	 * @return A string representation of that type
265 	 */
266 	public static String toString(ca.uhn.hl7v2.model.Type v2Type) {
267 		Composite comp = null;
268 		if (v2Type instanceof Varies v) {
269 			v2Type = v.getData();
270 		}
271 		try {
272 			if (v2Type == null || v2Type.isEmpty()) {
273 				return "";
274 			}
275 		} catch (HL7Exception e1) {
276 			return "";
277 		}
278 		if (v2Type instanceof Composite c) {
279 			comp = c;
280 		}
281 		int offset = 0;
282 		switch (v2Type.getName()) {
283 		case "AD":
284 			return addressToText(
285 					ParserUtils.toString(comp, 5), 
286 					ParserUtils.toString(comp, 2),
287 					ParserUtils.toString(comp, 3),
288 					ParserUtils.toString(comp, 4),
289 					ParserUtils.toString(comp, 7),
290 					ParserUtils.toString(comp, 1)
291 				);
292 		case "XAD":
293 			return addressToText(
294 				ParserUtils.toString(comp, 5), 
295 				ParserUtils.toString(comp, 2),
296 				ParserUtils.toString(comp, 3),
297 				ParserUtils.toString(comp, 4),
298 				ParserUtils.toString(comp, 8),
299 				ParserUtils.toString(comp, 0),
300 				ParserUtils.toString(comp, 1)
301 			);
302 		case "PL", "LA1", "LA2":
303 			return StringUtils.normalizeSpace(
304 				StringUtils.joinWith(" ", 
305 					ParserUtils.toString(comp, 2), 
306 					ParserUtils.toString(comp, 1),
307 					ParserUtils.toString(comp, 0),
308 					ParserUtils.toString(comp, 7),
309 					ParserUtils.toString(comp, 6),
310 					ParserUtils.toString(comp, 3)
311 				)
312 			);
313 			
314 		case "SAD":
315 			return ParserUtils.toString(comp);
316 		case "TN":
317 			return ParserUtils.toString(comp);
318 		case "XTN":
319 			assert comp != null;
320 			return xtnToText(comp.getComponents());
321 		case "CE", "CF", "CNE", "CWE":
322 			return codingToText(
323 				ParserUtils.toString(comp, 8),
324 				ParserUtils.toStrings(comp, 0, 1, 2, 3, 4, 5, 9, 10, 11)
325 			);
326 		case "CX":
327 			String type = ParserUtils.toString(comp, 4);
328 			if (StringUtils.isBlank(type)) {
329 				type = ParserUtils.toString(comp, 3);
330 			}
331 			return identifierToText(type, ParserUtils.toString(comp, 0), ParserUtils.toString(comp, 1));
332 		case "NM":
333 			return quantityToText(ParserUtils.toString(v2Type), null, null);
334 		case "CQ":
335 			return quantityToText(ParserUtils.toString(comp, 0), ParserUtils.toString(comp, 1), ParserUtils.toString(comp, 1, 1));
336 		case "HD":
337 			return Objects.toString(ParserUtils.toString(comp), ParserUtils.toString(comp, 1));
338 		case "EI":
339 			return identifierToText(ParserUtils.toString(comp, 1), ParserUtils.toString(comp, 0), null);
340 		case "ID", "IS", "ST", "EIP", "XON":
341 			return ParserUtils.toString(v2Type);
342 		
343 		case "CNN", "XCN":
344 			offset = 1;
345 			// fall through, CNN and XCN are like XPN but start one component later
346 		case "XPN":
347 			int professional = "XPN".equals(v2Type.getName()) ? 13 : 20;
348 			String[] sufixes = { ParserUtils.toString(comp, offset + 3), ParserUtils.toString(comp, offset + 5), ParserUtils.toString(comp, professional) };
349 			Object[] givens = { ParserUtils.toString(comp, offset + 1), ParserUtils.toString(comp, offset + 2) };
350 			
351 			return TextUtils.humanNameToText(ParserUtils.toString(comp, offset + 4), ParserUtils.toString(comp, offset + 0), 
352 					StringUtils.normalizeSpace(StringUtils.join(sufixes, " ")), givens);
353 		case "ERL", "MSG":
354 		default:
355 			try {
356 				return v2Type.encode();
357 			} catch (HL7Exception e) {
358 				return null;
359 			}
360 		}
361 	}

363 	private static String xtnToText(ca.uhn.hl7v2.model.Type[] types) {
364 		for (int i : Arrays.asList(1, 4, 5, 12)) {
365 			String value = i == 5 ? ContactPointParser.fromXTNparts(types) : ParserUtils.toString(types, i-1);
366 			if (StringUtils.isNotBlank(value)) {
367 				return value;
368 			}
369 		}
370 		return "";
371 	}

373 	/**
374 	 * Convert a FHIR Type to a standardized string output: case possible with
375 	 * multiple lines of text representing the content of the type.
376 	 * 
377 	 * @param fhirType The type to convert
378 	 * @return A string representation of the FHIR type.
379 	 */
380 	public static String toString(IBase fhirType) {
381 		if (fhirType == null || fhirType.isEmpty()) {
382 			return "";
383 		}
384 		if (fhirType instanceof PrimitiveType<?> pt) {
385 			if (pt instanceof BaseDateTimeType) {
386 				return DatatypeConverter.removeIsoPunct(pt.asStringValue());
387 			} 
388 			return StringUtils.defaultIfBlank(pt.asStringValue(), "");
389 		}
390 		switch (fhirType.fhirType()) {
391 		case "Address":
392 			return toText((Address) fhirType);
393 		case "CodeableConcept":
394 			return toText((CodeableConcept) fhirType);
395 		case "ContactPoint":
396 			return toText((ContactPoint) fhirType);
397 		case "Coding":
398 			return toText((Coding)fhirType);
399 		case "HumanName":
400 			return toText((HumanName)fhirType);
401 		case "Identifier":
402 			return toText((Identifier)fhirType);
403 		case "Location":
404 			return ((Location)fhirType).getName();
405 		case "Organization":
406 			return ((Organization)fhirType).getName();
407 		case "Patient":
408 			return toText(((Patient)fhirType).getNameFirstRep());
409 		case "Practitioner":
410 			return toText(((Practitioner)fhirType).getNameFirstRep());
411 		case "Quantity":
412 			Quantity quantity = (Quantity)fhirType;
413 			return toText(quantity);
414 		case "Reference":
415 			Reference ref = (Reference)fhirType;
416 			return toText(ref);
417 		case "RelatedPerson":
418 			return toText(((RelatedPerson)fhirType).getNameFirstRep());
419 		case "Extension":
420 			StringBuilder b = new StringBuilder();
421 			Extension ext = (Extension)fhirType;
422 			b.append("extension[")
423 			 .append(StringUtils.substringAfterLast("/" + ext.getUrl(),"/"))
424 			 .append("]");
425 			if (ext.hasValue()) {
426 				b.append("=").append(toString(ext.getValue())).append(" ");
427 			} else if (ext.hasExtension()) {
428 				// Walk the tree
429 				for (Extension ext2: ext.getExtension()) {
430 					b.append(toString(ext2));
431 				}
432 			}
433 			return b.toString();
434 		default:
435 			return "";
436 		}
437 	}

503 	/**
504 	 * Convert an Identifier to text. 
505 	 * @param identifier The identifier to convert
506 	 * @return	A text representation of the identifier
507 	 * @see #identifierToText
508 	 */
509 	public static String toText(Identifier identifier) {
510 		if (identifier == null || identifier.isEmpty()) {
511 			return "";
512 		}
513 		String type = null;
514 		if (identifier.hasType()) {
515 			CodeableConcept cc = identifier.getType();
516 			Coding coding = cc.getCodingFirstRep();
517 			type = Objects.toString(cc.getText(),  
518 							Objects.toString(coding.getCode(), 
519 								gov.cdc.izgw.v2tofhir.terminology.Systems.toTextName(identifier.getSystem())));
520 		} else {
521 			List<String> types = gov.cdc.izgw.v2tofhir.terminology.Systems.getSystemNames(identifier.getSystem());
522 			if (!types.isEmpty()) {
523 				type = types.get(0);
524 				if (types.size() > 1) {
525 					type = types.get(1);
526 				}
527 			} else {
528 				type = identifier.getSystem();
529 			}
530 			
531 			if (type != null &&
532 				// If type is an OID
533 				isOID(type) &&
534 				// and there's an assigner with a display name
535 				identifier.hasAssigner() && identifier.getAssigner().hasDisplay()
536 			) {
537 				// Use the display name for type in the text representation
538 				type = identifier.getAssigner().getDisplay();
539 			}
540 		}
541 		return identifierToText(type, identifier.getValue(), null);
542 	}

544 	private static boolean isOID(String type) {
545 		boolean dotSeen = false;
546 		for (int i = 0; i < type.length(); ++i) {
547 			char c = type.charAt(i);
548 			if (c == '.') {
549 				if (dotSeen || i == 0) {
550 					return false;
551 				}
552 				dotSeen = true;
553 			} else if (!Character.isDigit(c)) {
554 				return false;
555 			} else {
556 				dotSeen = false;
557 			}
558 		}
559 		return true;
560 	}

573 	 * Append the first string and any extra strings to b if the first string is not blank
574 	 * @param b	The StringBuilder to append to
575 	 * @param first	The string to append if non-empty
576 	 * @param extra	The extra text that will be appended if both first and extra are non-empty
577 	 */
578 	private static void appendIfNonBlank(StringBuilder b, Object first, String extra) {
579 		if (first != null && StringUtils.isNotBlank(first.toString())) {
580 			b.append(first);
581 			if (extra != null) {
582 				b.append(extra);
583 			}
584 		}
585 	}

587 	/**
588 	 * Remove any trailing whitespace from a StringBuilder
589 	 * @param b	The string builder
590 	 * @return	The adjusted string builder
591 	 */
592 	private static StringBuilder rightTrim(StringBuilder b) {
593 		int len = b.length();
594 		while (len > 0 && Character.isWhitespace(b.charAt(len - 1))) {
595 			--len;
596 		}
597 		b.setLength(len);
598 		return b;
599 	}

601 	/** 
602 	 * Convert a string generated from {@link #toText(CodeableConcept)} back to
603 	 * a CodeableConcept
604 	 *   
605 	 * @param actualString	The string to convert
606 	 * @return	A CodeableConcept created by parsing the string
607 	 */
608 	public static CodeableConcept toCodeableConcept(String actualString) {
609 		String[] parts = actualString.split(", ");
610 		CodeableConcept cc = new CodeableConcept();
611 		
612 		for (String part: parts) {
613 			String display = StringUtils.substringBefore(part, "(");
614 			String codeAndSystem = StringUtils.substringBetween(part, "(", ")");
615 			String system = StringUtils.substringBefore(codeAndSystem, " ");
616 			String code = StringUtils.substringAfter(codeAndSystem, " ");
617 			if (!cc.hasText() && 
618 				StringUtils.isNotEmpty(display) && 
619 				StringUtils.isAllEmpty(code, system)
620 			) {
621 				cc.setText(display);
622 			} else {
623 				Coding coding = new Coding(system, code, display);
624 				if (!coding.isEmpty()) {
625 					cc.addCoding(coding);
626 				}
627 			}
628 		}
629 		return cc;
630 	}
