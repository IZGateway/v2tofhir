9   import org.apache.commons.lang3.StringUtils;
10  import org.hl7.fhir.r4.model.Binary;
11  import org.hl7.fhir.r4.model.Resource;
12  import org.springframework.http.HttpHeaders;
13  import org.springframework.http.HttpInputMessage;

22  import lombok.extern.slf4j.Slf4j;

24  /**
25   * This is a converter to and from FHIR for SpringBoot applications that are
26   * not using the HAPI on FHIR native web server.
27   */
28  @Slf4j
29  public class FhirConverter implements HttpMessageConverter<Resource> {
30  	/** Create a FHIR message converter. */
31  	public FhirConverter() {
32  		// Nothing to initialize
33  	}
34  
35  	@Override
36  	public boolean canRead(Class<?> clazz, MediaType mediaType) {
37  		mediaType = mediaType == null ? null : new MediaType(StringUtils.lowerCase(mediaType.getType()), StringUtils.lowerCase(mediaType.getSubtype()));
38  		return Resource.class.isAssignableFrom(clazz) && (mediaType == null || ContentUtils.FHIR_MEDIA_TYPES.contains(mediaType));
39  	}
40  
41  	@Override
42  	public boolean canWrite(Class<?> clazz, MediaType mediaType) {
43  		String subtype = mediaType != null ? StringUtils.lowerCase(mediaType.getSubtype()) : "";
44  		if (Binary.class.isAssignableFrom(clazz) && mediaType != null && (subtype.contains("cda") || subtype.contains("hl7v2"))) {
45  			return true;
46  		}
47  		return canRead(clazz, mediaType);
48  	}
49  
50  	@Override
51  	public List<MediaType> getSupportedMediaTypes() {
52  		return ContentUtils.HL7_MEDIA_TYPES;
53  	}
54  
55  	@Override
56  	public Resource read(Class<? extends Resource> clazz, HttpInputMessage inputMessage)
57  			throws IOException, HttpMessageNotReadableException {
58  		
59  		IParser parser = null;
60  		String contentType = inputMessage.getHeaders().getFirst(HttpHeaders.CONTENT_TYPE);
61  		MediaType mediaType = null;
62  		
63  		BufferedInputStream bis = IOUtils.buffer(inputMessage.getBody());
64  		if (StringUtils.isBlank(contentType)) {
65  			mediaType = ContentUtils.guessMediaType(bis);
66  		} else {
67  			try {
68  				mediaType = MediaType.parseMediaType(contentType);
69  				// Simplify it, stripping any parameters such as charset.
70  				mediaType = new MediaType(mediaType.getType(), mediaType.getSubtype());
71  			} catch (InvalidMediaTypeException e) {
72  				log.warn("Unparseable Content-Type '{}', guessing media type from content", contentType, e);
73  				mediaType = ContentUtils.guessMediaType(bis);
74  			}
75  		}
76  		parser = ContentUtils.selectParser(mediaType);
77  		// HAPI's typed parseResource(Class, ...) scans the target class and rejects
78  		// abstract classes / interfaces (e.g. org.hl7.fhir.r4.model.Resource) with
79  		// "HAPI-1682: Can not scan abstract or interface class". Spring passes the
80  		// *declared* @RequestBody type, so a handler declaring an abstract type such
81  		// as Resource fails here before the controller runs. When the requested type
82  		// is abstract/interface, use the auto-detecting overload, which reads
83  		// resourceType from the payload and resolves the concrete class.
84  		if (Modifier.isAbstract(clazz.getModifiers()) || clazz.isInterface()) {
85  			Resource parsed = (Resource) parser.parseResource(bis);
86  			if (!clazz.isInstance(parsed)) {
87  				throw new HttpMessageNotReadableException(
88  					"Parsed resource type '" + parsed.fhirType()
89  						+ "' is not assignable to requested type " + clazz.getName(),
90  					inputMessage);
91  			}
92  			return parsed;
93  		}
94  		return parser.parseResource(clazz, bis);
95  	}
96  
97  	@Override
98  	public void write(Resource resource, MediaType contentType, HttpOutputMessage outputMessage)
99  			throws IOException, HttpMessageNotWritableException {
100 		if (resource instanceof Binary b && ContentUtils.FHIR_MEDIA_TYPES.stream().noneMatch(t -> t.includes(contentType))) {
101 			writeBinary(b, outputMessage);
102 			return;
103 		}
104 		IParser parser = ContentUtils.selectParser(contentType);
105 		OutputStreamWriter w = null;
106 		try {  // NOSONAR: It's up to the caller to determine whether the body gets closed when this is done.
107 			w = new OutputStreamWriter(outputMessage.getBody());
108 			parser.encodeResourceToWriter(resource, w);
109 		} finally {
110 			if (w != null) {
111 				w.flush();
112 			}
113 		}
114 	}
115 
116 	private void writeBinary(Binary b, HttpOutputMessage outputMessage) throws IOException {
117 		OutputStreamWriter w = null;
118 		try {  // NOSONAR: It's up to the caller to determine whether the body gets closed when this is done.
119 			w = new OutputStreamWriter(outputMessage.getBody());
120 			for (byte data: b.getData()) {
121 				w.write((char)data);
122 			}
123 		} finally {
124 			if (w != null) {
125 				w.flush();
126 			}
127 		}
128 		
129 	}
130 }
