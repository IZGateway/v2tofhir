28  /**
29   * Parser that supports parsing a String, or HL7 Version into a FHIR ContactPoint.
30   * 
31   * @author Audacious Inquiry
32   */
33  @Slf4j
34  public class ContactPointParser implements DatatypeParser<ContactPoint> {
35  	static {
36  		log.debug("{} loaded", ContactPointParser.class.getName());
37  	}
38  
39  	static final String AREA_CODE = "\\(\\s*\\d{3}\s*\\)";
40  	static final String COUNTRY_CODE = "\\+\\s*\\d{1,3}";
41  	static final String PHONE_CHAR = "\\-+0123456789(). []{},#";
42  	static final String PHONE_NUMBER = "(-|\\.|\\d+)+";
43  	// [NN] [(999)]999-9999[X99999][B99999][C any text]
44  	static final Pattern TN_PATTERN = Pattern.compile(
45  		"((\\d{2,3})?(\\(\\d{3}\\))?\\d{3}-?\\d{4}(X\\d{1,5})?(B\\d1,5)?)(C.*)?$"
46  	);
47  	static final String CONTACTPOINT_COMMENT = "http://hl7.org/fhir/StructureDefinition/contactpoint-comment";
48  	
49  	private static final EmailValidator EMAIL_VALIDATOR = EmailValidator.getInstance();
50  	static StringBuilder appendIfNotBlank(StringBuilder b, String prefix, String string, String suffix) {
51  		if (StringUtils.isBlank(string)) {
52  			return b;
53  		}
54  		return b.append(StringUtils.defaultString(prefix))
55  				.append(StringUtils.defaultString(string))
56  				.append(StringUtils.defaultString(suffix));
57  	}

66  	/**
67  	 * Convert an HL7 V2 XTN into a list of contacts.
68  	 * @param xtn	The HL7 V2 xtn (or similarly shaped composite).
69  	 * @return	A list of ContactPoint objects from the XTN.
70  	 */
71  	public List<ContactPoint> convert(Composite xtn) {
72  		Type[] types = xtn.getComponents();
73  		List<ContactPoint> cps = new ArrayList<>();
74  		if (types.length > 0) {
75  			cps.add(convert(types[0]));	// Convert XTN-1 to a phone number.
76  		}
77  		cps.add(fromEmail(ParserUtils.toString(types, 3)));
78  		String value = fromXTNparts(types);
79  		if (StringUtils.isNotBlank(value)) {
80  			cps.add(new ContactPoint().setValue(value).setSystem(ContactPointSystem.PHONE));
81  		}
82  		String unformatted = ParserUtils.toString(types, 11);
83  		if (StringUtils.isNotBlank(unformatted)) {
84  			cps.add(new ContactPoint().setValue(unformatted).setSystem(ContactPointSystem.PHONE));
85  		}
86  		String comment = ParserUtils.toString(types, 8);
87  		String use = StringUtils.defaultIfBlank(ParserUtils.toString(types, 1), "");
88  		String type = StringUtils.defaultIfBlank(ParserUtils.toString(types, 2), "");
89  		DateTimeType start = types.length > 12 ? DatatypeConverter.toDateTimeType(types[12]) : null;
90  		DateTimeType end = types.length > 13 ? DatatypeConverter.toDateTimeType(types[13]) : null;
91  		IntegerType order = types.length > 17 ? DatatypeConverter.toIntegerType(types[17]) : null;
92  		
93  		Period period = null;
94  		if (start != null || end != null) {
95  			period = new Period().setStartElement(start).setEndElement(end);
96  		}
97  		
98  		cleanupContactPoints(cps, comment, use, type, order, period);
99  		return cps;
100 	}

107 	@Override
108 	public ContactPoint convert(Type type) {
109 		
110 		if ((type = DatatypeConverter.adjustIfVaries(type)) == null) {
111 			return null;
112 		}
113 		
114 		if (type instanceof Primitive pt) {
115 			if (DatatypeConverter.isDeleted(pt)) {
116 				return DatatypeConverter.markDeleted(new ContactPoint());
117 			}
118 
119 			return fromString(ParserUtils.toString(type)); 
120 		}
121 		
122 		if ("XTN".equals(type.getName())) {
123 			List<ContactPoint> cps = convert((Composite)type);
124 			return cps.isEmpty() ? null : cps.get(0);
125 		}
126 		
127 		return null;
128 	}

130 	@Override
131 	public ContactPoint fromString(String value) {
132 		ContactPoint cp = fromPhone(value);
133 		if (cp != null && !cp.isEmpty()) {
134 			return cp;
135 		}
136 		cp = fromEmail(value);
137 		if (cp != null && !cp.isEmpty()) {
138 			return cp;
139 		}
140 		cp = fromUrl(value);
141 		if (cp != null && !cp.isEmpty()) {
142 			return cp;
143 		}
144 		return null;
145 	}

147 	/**
148 	 * Create a ContactPoint from a string representing an e-mail address.
149 	 * 
150 	 * This method recognizes mailto: urls, and RFC-2822 email addresses
151 	 * @see #fromEmail(String)
152 	 * 
153 	 * @param value	The email address.
154 	 * @return	A ContactPoint object representing this e-mail address. 
155 	 */
156 	public static ContactPoint fromEmail(String value) {
157 		if (StringUtils.isBlank(value)) {
158 			return null;
159 		}
160 		value = StringUtils.strip(value).split("[;,]")[0];
161 		String name = null;
162 		String email = null;
163 		if (value.contains("<")) {
164 			name = StringUtils.substringBefore(value, "<");
165 			email = StringUtils.substringBetween(value, "<", ">");
166 		} else {
167 			email = value; 
168 		}
169 		if (Strings.CS.startsWith(email, "mailto:")) {
170 			email = email.substring(7);
171 		}
172 		/*
173 		 * Simplfied RFC-2822 grammar
174 		 * mailbox         = name-addr / addr-spec
175 		 * name-addr       = [display-name] angle-addr
176 		 * angle-addr      = [CFWS] "<" addr-spec ">" [CFWS]
177 		 * display-name	   = word | quoted-string
178 		 * addr-spec       = local-part "@" domain
179 		 * local-part      = dot-atom / quoted-string / obs-local-part
180 		 * domain          = dot-atom / domain-literal / obs-domain
181 		 * domain-literal  = [CFWS] "[" *([FWS] dcontent) [FWS] "]" [CFWS]
182 		 * dcontent        = dtext / quoted-pair
183 		 * dtext           = NO-WS-CTL /     ; Non white space controls
184 	     *                   %d33-90 /       ; The rest of the US-ASCII
185 	     *                   %d94-126        ;  characters not including "[", "]", or "\"
186 		 */
187 		if (EMAIL_VALIDATOR.isValid(email)) {
188 			ContactPoint cp = new ContactPoint();
189 			cp.setSystem(ContactPointSystem.EMAIL);
190 			cp.setValue(email);
191 			if (StringUtils.isNotBlank(name)) {
192 				cp.addExtension()
193 					.setUrl(CONTACTPOINT_COMMENT)
194 					.setValue(new StringType(name));
195 			}
196 			return cp;
197 		}
198 		return null;
199 	}

201 	/**
202 	 * Create a list of ContactPoints 
203 	 * @param values	A string providing the list of email addresses, separated by semi-colons, commas, or newlines.
204 	 * @return	The list of contact points.
205 	 */
206 	public static List<ContactPoint> fromEmails(String values) {
207 		if (StringUtils.isBlank(values)) {
208 			return Collections.emptyList();
209 		}
210 		List<ContactPoint> cps = new ArrayList<>();
211 		for (String value: StringUtils.strip(values).split("[;,\n\r]+")) {
212 			ContactPoint cp = fromEmail(value);
213 			if (cp != null && !cp.isEmpty()) {
214 				cps.add(cp);
215 			}
216 		}
217 		return cps;
218 	}

220 	/**
221 	 * Create a ContactPoint from a string representing a phone number.
222 	 * 
223 	 * This method recognizes tel: urls and other strings containing phone dialing
224 	 * characters and punctuation.
225 	 * 
226 	 * @param value	The phone number.
227 	 * @return	A ContactPoint object representing the phone number. 
228 	 */
229 	public static ContactPoint fromPhone(String value) {
230 		if (StringUtils.isBlank(value)) {
231 			return null;
232 		}
233 		ContactPoint cp = null;
234 		value = StringUtils.strip(value);
235 		if (value.startsWith("tel:")) {
236 			return new ContactPoint().setValue(value.substring(4)).setSystem(ContactPointSystem.PHONE);
237 		}
238 		if (value.startsWith("fax:")) {
239 			return new ContactPoint().setValue(value.substring(4)).setSystem(ContactPointSystem.FAX);
240 		}
241 		if (value.startsWith("sms:")) {
242 			return new ContactPoint().setValue(value.substring(4)).setSystem(ContactPointSystem.SMS);
243 		}
244 		
245 		Matcher m = TN_PATTERN.matcher(value);
246 		if (m.matches()) {
247 			cp = new ContactPoint();
248 			// Could be fax or pager or SMS or other but we won't know without context.
249 			cp.setSystem(ContactPointSystem.PHONE);
250 			cp.setValue(m.group(1));
251 			String comment = StringUtils.substringAfter(value, "C");
252 			if (StringUtils.isNotBlank(comment)) {
253 				cp.addExtension()
254 					.setUrl(CONTACTPOINT_COMMENT)
255 					.setValue(new StringType(comment));
256 			}
257 			return cp;
258 		}
259 		if (StringUtils.containsOnly(value, PHONE_CHAR) ||
260 			value.matches(COUNTRY_CODE) || value.matches(AREA_CODE) || value.matches(PHONE_NUMBER)
261 		) {
262 			cp = new ContactPoint();
263 			cp.setSystem(ContactPointSystem.PHONE);
264 			cp.setValue(value.replace(" ", ""));
265 			return cp;
266 		}
267 		return null;
268 	}

270 	/**
271 	 * Create a ContactPoint from a URL
272 	 * @param url The URL to convert to a contact point
273 	 * @return A ContactPoint object representing the URL.
274 	 */
275 	public static ContactPoint fromUrl(String url) {
276 		url = StringUtils.strip(url);
277 		if (StringUtils.isEmpty(url)) {
278 			return null;
279 		}
280 		String scheme = StringUtils.substringBefore(url, ":");
281 		ContactPointSystem system = ContactPointSystem.URL;
282 		if (StringUtils.isNotEmpty(scheme)) {
283 			if ("mailto".equalsIgnoreCase(scheme))
284 				system = ContactPointSystem.EMAIL;
285 			else if ("tel".equalsIgnoreCase(scheme))
286 				system = ContactPointSystem.PHONE;
287 			else if ("fax".equalsIgnoreCase(scheme))
288 				system = ContactPointSystem.FAX;
289 			else if ("sms".equalsIgnoreCase(scheme))
290 				system = ContactPointSystem.SMS;
291 			return new ContactPoint()
292 					.setValue(url)
293 					.setSystem(system);
294 		}
295 		if (url.matches("^([\\-a-zA-Z0-9]+|\\d{1,3})([.\\-a-zA-Z0-9]+|\\d{1,3})+(/.*)?$")) {
296 			if (url.startsWith("www") || url.contains("/")) {
297 				url = "http://" + url;
298 				system = ContactPointSystem.URL;
299 			} else {
300 				system = ContactPointSystem.OTHER;
301 			}
302 			return new ContactPoint()
303 					.setValue(url)
304 					.setSystem(system);
305 		}
306 		return null;
307 	}

314 	private static void cleanupContactPoints(List<ContactPoint> cps, String comment, String use, String type,
315 			IntegerType order, Period period) {
316 		// Cleanup list after parsing.
317 		Iterator<ContactPoint> it = cps.iterator();
318 		while (it.hasNext()) {
319 			ContactPoint cp = it.next();
320 			if (cp == null || cp.isEmpty() || StringUtils.isBlank(cp.getValue())) {
321 				it.remove();
322 				continue;
323 			}
324 			if (StringUtils.isNotBlank(comment)) {
325 				cp.addExtension()
326 					.setUrl(CONTACTPOINT_COMMENT)
327 					.setValue(new StringType(comment));
328 			}
329 			mapUseCode(use, cp);
330 			mapTypeCode(type, cp);
331 			if (period != null && !period.isEmpty()) {
332 				cp.setPeriod(period);
333 			}
334 			if (order != null && !order.isEmpty()) {
335 				cp.setRank(order.getValue());
336 			}
337 		}
338 	}

340 	/**
341 	 * Create a phone number as a String from the components of XTN datatype.
342 	 * @param types	The components of the XTN datatype
343 	 * @return	The phone number as a string.
344 	 */
345 	public static String fromXTNparts(Type[] types) {
346 		String country = ParserUtils.toString(types, 4);
347 		String area = ParserUtils.toString(types, 5);
348 		String local = ParserUtils.toString(types, 6);
349 		String extension  = ParserUtils.toString(types, 7);
350 		String extPrefix = ParserUtils.toString(types, 9);
351 		
352 		if (!StringUtils.isAllBlank(country, area, local, extension)) {
353 			StringBuilder b = new StringBuilder();
354 			appendIfNotBlank(b, "+", country, " ");
355 			appendIfNotBlank(b, "(", area, ") ");
356 			appendIfNotBlank(b, null, local, null);
357 			appendIfNotBlank(b, StringUtils.defaultIfBlank(extPrefix, "#"), extension, null);
358 			return b.toString();
359 		}
360 		return null;
361 	}

363 	private static void mapTypeCode(String type, ContactPoint cp) {
364 		switch (type) {
365 		case "BP": cp.setSystem(ContactPointSystem.PAGER); break;
366 		case "CP": 
367 			cp.setSystem(ContactPointSystem.PHONE);
368 			cp.setUse(ContactPointUse.MOBILE);
369 			break;
370 		case "FX": cp.setSystem(ContactPointSystem.FAX); break;
371 		case "Internet": cp.setSystem(ContactPointSystem.URL); break;
372 		case "MD": cp.setSystem(ContactPointSystem.OTHER); break;
373 		case "PH", "SAT": 
374 			cp.setSystem(ContactPointSystem.PHONE); break;
375 		case "TDD", "TTY": cp.setSystem(ContactPointSystem.OTHER); break;
376 		case "X.400": cp.setSystem(ContactPointSystem.EMAIL); break;
377 		default: break;
378 		}
379 	}
380 
381 	private static void mapUseCode(String use, ContactPoint cp) {
382 		switch (use) {
383 		case "PRS": // Personal Number
384 			cp.setUse(ContactPointUse.MOBILE); break;
385 		case "ORN",	// Other Residence Number
386 			 "PRN", // Primary Residence Number
387 			 "VHN": // Vacation Home Number
388 			cp.setUse(ContactPointUse.HOME); break;
389 		case "WPN": // Work Number
390 			cp.setUse(ContactPointUse.WORK); break;
391 		case "NET", // Network (email) Address
392 			 "ASN",	// Answering Service Number
393 			 "BPN",	// Beeper Number
394 			 "EMR": // Emergency Number
395 		default: break;
396 		}
