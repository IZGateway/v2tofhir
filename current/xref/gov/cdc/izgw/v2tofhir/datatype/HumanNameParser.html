75  	/**
76  	 * Parse a human name from a string. The parser recognizes common prefixes and
77  	 * suffixes and puts them in the prefix and suffix fields of the human name. It
78  	 * puts the first space separated string into the first given name, and any
79  	 * susbsequent strings except the last.
80  	 * 
81  	 * @param name The name to parse
82  	 * @return The parsed name in a HumanName object
83  	 */

89  	/**
90  	 * Parse a human name from a string. The parser recognizes common prefixes and
91  	 * suffixes and puts them in the prefix and suffix fields of the human name. It
92  	 * puts the first space separated string into the first given name, and any
93  	 * susbsequent strings except the last.
94  	 * 
95  	 * @param name The name to parse
96  	 * @return The parsed name in a HumanName object
97  	 */
98  	public static HumanName computeFromString(String name) {
99  		if (name == null) {
100 			return null;
101 		}
102 		HumanName hn = new HumanName();
103 		hn.setText(name);
104 		String[] parts = name.split("\\s");
105 		StringBuilder familyName = new StringBuilder();
106 		for (String part : parts) {
107 			updateNamePart(hn, familyName, part);
108 		}
109 		List<StringType> given = hn.getGiven();
110 		if (familyName.isEmpty() && !given.isEmpty()) {
111 			String family = given.get(given.size() - 1).toString();
112 			hn.setFamily(family);
113 			given.remove(given.size() - 1);
114 			hn.setGiven(null);
115 			for (StringType g : given) {
116 				hn.addGiven(g.toString());
117 			}
118 		} else if (!familyName.isEmpty()) {
119 			familyName.setLength(familyName.length() - 1); // Remove terminal " "
120 			hn.setFamily(familyName.toString());
121 		}
122 		if (hn.isEmpty()) {
123 			return null;
124 		}
125 		return hn;
126 	}

128 	private static void updateNamePart(HumanName hn, StringBuilder familyName, String part) {
129 		NamePart classification = classifyNamePart(part);
130 		switch (classification) {
131 		case PREFIXANDSUFFIX:
132 			if (!hn.hasGiven() && familyName.isEmpty()) {
133 				hn.addPrefix(part);
134 				return;
135 			} 
136 			if (familyName.isEmpty()) { 
137 				// could be a prefix or a suffix, but we haven't yet started
138 				// on suffixes and are beyond prefixes.  
139 				// Only thing that is both is "sr".
140 				// We add it as a given name.
141 				hn.addGiven(part);
142 				return;
143 			}
144 			// Fall through to add suffix if we have some part of familyName
145 		case SUFFIX:
146 			hn.addSuffix(part);
147 			return;
148 		case PREFIX:
149 			if (!hn.hasGiven() && familyName.isEmpty()) {
150 				hn.addPrefix(part);
151 				return;
152 			} 
153 			familyName.append(part).append(" ");
154 			return;
155 		case AFFIX:
156 			if (!hn.hasGiven()) {
157 				hn.addGiven(part);
158 				return;
159 			} 
160 			familyName.append(part).append(" ");
161 			return;
162 		case NAME:
163 			if (familyName.isEmpty()) {
164 				hn.addGiven(part);
165 				return;
166 			} 
167 			familyName.append(part).append(" ");
168 			return;
169 		default:
170 			return;
171 		}
172 	}

174 	private static NamePart classifyNamePart(String part) {
175 		if (isPrefix(part)) {
176 			return isSuffix(part) ? NamePart.PREFIXANDSUFFIX : NamePart.PREFIX;
177 		}
178 		if (isAffix(part)) {
179 			return NamePart.AFFIX;
180 		}
181 		if (isSuffix(part) || isDegree(part)) {
182 			return NamePart.SUFFIX;
183 		}
184 		return NamePart.NAME;
185 	}

187 	@Override
188 	public HumanName convert(Type t) {
189 		t = DatatypeConverter.adjustIfVaries(t);
190 		if (t instanceof Primitive pt) {
191 			if (t instanceof Primitive p && DatatypeConverter.isDeleted(p)) {
192 				return DatatypeConverter.markDeleted(new HumanName());
193 			}
194 
195 			return fromString(pt.getValue());
196 		}
197 
198 		if (t instanceof Composite comp) {
199 			Type[] types = comp.getComponents();
200 			switch (t.getName()) {
201 			case "CNN":
202 				return HumanNameParser.parse(types, 1, -1);
203 			case "XCN":
204 				HumanName hn = HumanNameParser.parse(types, 1, 9);
205 				if (hn != null && types.length > 20 && types[20] != null) {
206 					String suffix = ParserUtils.toString(types[20]);
207 					hn.addSuffix(suffix);
208 				}
209 				return hn;
210 			case "XPN":
211 				return HumanNameParser.parse(types, 0, 6);
212 			default:
213 				break;
214 			}
215 		}
216 		return null;
217 	}

224 	static HumanName parse(Type[] types, int offset, int nameTypeLoc) {
225 		HumanName hn = new HumanName();
226 		for (int i = 0; i < 6; i++) {
227 			String part = ParserUtils.toString(types[i + offset]);
228 			if (StringUtils.isBlank(part)) {
229 				continue;
230 			}
231 			switch (i) {
232 			case 0: // Family Name (as ST in HL7 2.3)
233 				hn.setFamily(part);
234 				break;
235 			case 1: // Given Name
236 				hn.addGiven(part);
237 				break;
238 			case 2: // Second and Further Given Name or Initials
239 					// Thereof
240 				hn.addGiven(part);
241 				break;
242 			case 3: // Suffix
243 				hn.addSuffix(part);
244 				break;
245 			case 4: // Prefix
246 				hn.addPrefix(part);
247 				break;
248 			case 5: // Degree
249 				hn.addSuffix(part);
250 				break;
251 			default:
252 				break;
253 			}
254 		}
255 		Type type = DatatypeConverter.adjustIfVaries(types, nameTypeLoc);
256 
257 		if (type instanceof Primitive pt) {
258 			String nameType = pt.getValue();
259 			hn.setUse(toNameUse(nameType));
260 		}
261 		if (hn.isEmpty()) {
262 			return null;
263 		}
264 		return hn;
265 	}

267 	private static NameUse toNameUse(String nameType) {
268 		if (nameType == null) {
269 			return null;
270 		}
271 		switch (StringUtils.upperCase(nameType)) {
272 		case "A": // Assigned
273 			return NameUse.USUAL;
274 		case "D": // Customary Name
275 			return NameUse.USUAL;
276 		case "L": // Official Registry Name
277 			return NameUse.OFFICIAL;
278 		case "M": // Maiden Name
279 			return NameUse.MAIDEN;
280 		case "N": // Nickname
281 			return NameUse.NICKNAME;
282 		case "NOUSE": // No Longer To Be Used
283 			return NameUse.OLD;
284 		case "R": // Registered Name
285 			return NameUse.OFFICIAL;
286 		case "TEMP": // Temporary Name
287 			return NameUse.TEMP;
288 		case "B", // Birth name
289 				"BAD", // Bad Name
290 				"C", // Adopted Name
291 				"I", // Licensing Name
292 				"K", // Business name
293 				"MSK", // Masked
294 				"NAV", // Temporarily Unavailable
295 				"NB", // Newborn Name
296 				"P", // Name of Partner/Spouse
297 				"REL", // Religious
298 				"S", // Pseudonym
299 				"T", // Indigenous/Tribal
300 				"U": // Unknown
301 		default:
302 			return null;
303 		}
304 	}

306 	/**
307 	 * Sees if the string is an affix
308 	 * @param part	The string to check
309 	 * @return	True if it is an affix to the name.
310 	 */
311 	public static boolean isAffix(String part) {
312 		return AFFIXES.contains(StringUtils.lowerCase(part));
313 	}
314 
315 	/**
316 	 * Sees if the string is an prefix
317 	 * @param part	The string to check
318 	 * @return	True if it is an prefix to the name.
319 	 */
320 	public static boolean isPrefix(String part) {
321 		return PREFIXES.contains(Strings.CS.replace(StringUtils.lowerCase(part), ".", ""));
322 	}
323 
324 	/**
325 	 * Sees if the string is an suffix
326 	 * @param part	The string to check
327 	 * @return	True if it is an suffix to the name.
328 	 */
329 	public static boolean isSuffix(String part) {
330 		return SUFFIXES.contains(Strings.CS.replace(StringUtils.lowerCase(part), ".", ""));
331 	}
332 
333 	/**
334 	 * Sees if the string is a degree (e.g., MD)
335 	 * @param part	The string to check
336 	 * @return	True if it is a degree following the name
337 	 */
338 	public static boolean isDegree(String part) {
339 		return DEGREES.contains(Strings.CS.replace(StringUtils.lowerCase(part), ".", ""));
340 	}
