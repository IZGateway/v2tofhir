5   /**
6    * A parser for a FHIR Datatype
7    * @param <T>	The FHIR datatype parsed by this object.
8    * 
9    * @author Audacious Inquiry
10   */
11  public interface DatatypeParser<T extends org.hl7.fhir.r4.model.Type> {
12  	/**
13  	 * The name of the type this parser works on.
14  	 * @return The name of the type this parser works on.
15  	 */
16  	Class<? extends T> type();
17  	/**
18  	 * Convert a string to a FHIR type.
19  	 * 
20  	 * @param value	The string to convert.
21  	 * @return	An item of the specified type or null if the conversion could not be performed or when value is null or empty.
22  	 * @throws UnsupportedOperationException If this type cannot be converted from a string.
23  	 */
24  	T fromString(String value) throws UnsupportedOperationException;
25  	/**
26  	 * Convert a V2 type to a FHIR type.
27  	 * 
28  	 * @param type The V2 type to convert.
29  	 * @return	An item of the required type or null if the conversion could not be performed or value is null or empty.
30  	 */
31  	T convert(Type type);
32  	/**
33  	 * Convert a FHIR type to a V2 type
34  	 * @param fhirType	The FHIR type to convert
35  	 * @return The HL7 v2 type.
36  	 */
37  	Type convert(T fhirType);
38  }
