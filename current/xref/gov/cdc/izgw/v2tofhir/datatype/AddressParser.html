222 	@Override
223 	public Address convert(Type type) {
224 		Address addr = null;
225 		type = DatatypeConverter.adjustIfVaries(type);
226 
227 		if (type instanceof Primitive pt) {
228 			if (DatatypeConverter.isDeleted(pt)) {
229 				return DatatypeConverter.markDeleted(new Address());
230 			}
231 			addr = fromString(pt.getValue());
232 		} else if (type instanceof Composite comp) {
233 			switch (type.getName()) {
234 			case "AD", "XAD":
235 				addr = parse(comp.getComponents(), 0);
236 				break;
237 			case "SAD":
238 				addr = new Address().addLine(ParserUtils.toString(comp, 0));
239 				break;
240 			case "LA1":
241 				Type[] types = comp.getComponents();
242 				if (types.length > 8) {
243 					// Recursive call to parse LA1-8 which is of type AD
244 					addr = convert(types[8]);
245 				}
246 				break;
247 			case "LA2":
248 				addr = parse(comp.getComponents(), 8);
249 				break;
250 			default:
251 				break;
252 			}
253 		} 
254 		
255 		removeEmptyLines(addr);
256 		if (addr == null || addr.isEmpty()) {
257 			return null;
258 		}
259 		return addr;
260 	}

262 	private static void removeEmptyLines(Address addr) {
263 		if (addr != null && addr.hasLine()) {
264 			Iterator<StringType> it = addr.getLine().iterator();
265 			while (it.hasNext()) {
266 				StringType line = it.next(); 
267 				if (line == null || line.isEmpty()) {
268 					it.remove();
269 				}
270 			}
271 		}
272 	}

278 	@Override
279 	public Address fromString(String value) {
280 		if (StringUtils.isBlank(value)) {
281 			return null;
282 		}
283 		String[] parts = value.split("[\\n\\r]");
284 		String[] originalParts = parts;
285 		if (parts.length == 1) {
286 			parts = value.split(",");
287 		}
288 		Address addr = new Address();
289 		addr.setText(value);
290 		for (int i = 0; i < parts.length; i++) {
291 			String part = StringUtils.normalizeSpace(parts[i]);
292  			if (part.isEmpty()) {
293 				continue;
294 			}
295 			if (!addr.hasLine()) {
296 				getLineOrCsz(originalParts, addr, part);
297 				continue;
298 			} 
299 			
300 			if (isAddressLine(part)) {
301 				addr.addLine(part);
302 				continue;
303 			} 
304 			
305 			if (isCountry(addr, part)) {
306 				addr.setCountry(part);
307 				continue;
308 			} 
309 			
310 			if (!addr.hasPostalCode()) {
311 				getCityStatePostalCode(addr, part);
312 			}
313 		}
314 		return addr.isEmpty() ? null : addr;
315 	}
316 	private static void getLineOrCsz(String[] originalParts, Address addr, String part) {
317 		// First line is usually an address line, but
318 		// sometimes it might just be a city, state and postal code
319 		// in cases of partial representations.
320 		if (isCszOnly(originalParts)) {
321 			getCityStatePostalCode(addr, part);
322 		} else {
323 			addr.addLine(part);
324 		}
325 	}

328 	 * Returns true if part is a possible country name
329 	 * @param addr	The addr that may need a country
330 	 * @param part	The part to examine
331 	 * @return	true if addr needs country and part is a country name
332 	 */
333 	private static boolean isCountry(Address addr, String part) {
334 		return !addr.hasCountry()
335 				&& countryPattern.matcher(part).matches();
336 	}
337 	/**
338 	 * Returns true if an address part matches patterns indicating it is an address line.
339 	 * 
340 	 * NOTE: Not all address lines will be matched, only those containing certain patterns
341 	 * that match (e.g., those with a street name or abbreviation, a directional indicator, or a unit number)
342 	 * @param part	The string to match
343 	 * @return true if it appears to be an address line, false if it doesn't match any known patterns.
344 	 */
345 	private static boolean isAddressLine(String part) {
346 		return streetPattern.matcher(part).matches()
347 				|| directionalPattern.matcher(part).matches()
348 				|| unitPattern.matcher(part).matches();
349 	}

351 	/**
352 	 * Check for odd cases like:
353 	 * Just CSZ: Chicago, IL, 65932
354 	 * Just CS and Country: Myfaircity, GA\nUSA
355 	 * Just State: Indiana
356 	 * @return true if it's an odd case.
357 	 */
358 	private static boolean isCszOnly(String[] parts) {
359 		if (parts.length > 2) {
360 			return false;
361 		}
362 		if (parts.length == 2 && 
363 			!countryPattern.matcher(StringUtils.trim(parts[1])).matches()
364 		) {
365 			return false;
366 		}
367 		return isCszOnly(StringUtils.trim(parts[0]));
368 	}

370 	private static boolean isCszOnly(String line) {
371 		boolean hasCa1Part = false;
372 		boolean hasPostalCode = false;
373 		boolean hasState = false;
374 		boolean hasAnythingAfterState = false;
375 		for (String part : line.split("[ ,]+")) {
376 			if (postalCodePattern.matcher(part).matches()) {
377 				hasPostalCode = true;
378 			} else if (caPostalCode1.matcher(part).matches()) {
379 				hasCa1Part = true;
380 			} else if (caPostalCode2.matcher(part).matches()) {
381 				hasPostalCode = hasCa1Part;
382 			} else if (statePattern.matcher(part).matches()) {
383 				hasState = true;
384 			} else if (hasState) {
385 				// Pennsylvania Ave would result in a false match if we didn't check
386 				// for anything else on the line other that postalCode and state
387 				hasAnythingAfterState = true;
388 			}
389 		}
390 		return (hasState || hasPostalCode) && !hasAnythingAfterState;
391 	}

393 	private static void getCityStatePostalCode(Address addr, String part) {
394 		// could be an address line, country, or some combination of
395 		// city, state, and postal code
396 
397 		String[] lineParts = part.split("[, ]+");
398 		lineParts = getPostalCode(addr, lineParts);
399 		lineParts = getState(addr, lineParts);
400 		lineParts = getCountry(addr, lineParts);
401 		// City is anything left
402 		if (lineParts.length != 0) {
403 			addr.setCity(StringUtils.joinWith(" ", (Object[])lineParts));
404 		}
405 	}
406 	private static String[] getCountry(Address addr, String[] lineParts) {
407 		int position = 0;

428 	private static String[] getPostalCode(Address addr, String[] lineParts) {
429 		String postalPart1 = null;
430 		int position = 0;
431 		for (String linePart : lineParts) {
432 			if (postalCodePattern.matcher(linePart).matches()) {
433 				addr.setPostalCode(linePart);
434 				return ParserUtils.removeArrayElement(lineParts, position);
435 			} else if (postalPart1 == null && caPostalCode1.matcher(linePart).matches()) {
436 				postalPart1 = linePart;
437 			} else if (postalPart1 != null && caPostalCode2.matcher(linePart).matches()) {
438 				addr.setPostalCode(postalPart1 + " " + linePart);
439 				// Remove the part we just matched
440 				lineParts = ParserUtils.removeArrayElement(lineParts, position);
441 				// And it's predecessor.
442 				return ParserUtils.removeArrayElement(lineParts, position - 1);
443 			} else {
444 				postalPart1 = null;
445 			}
446 			++position;
447 		}
448 		return lineParts;
449 	}

451 	/**
452 	 * Parse a composite made up of types into an address.
453 	 * @param types	The components of the address
454 	 * @param offset The offset to the address line
455 	 * @return	A new FHIR Address populated from the values in types.
456 	 */
457 	public static Address parse(Type[] types, int offset) {
458 		Address addr = new Address();
459 		for (int i = 0; i < 14; i++) {
460 			
461 			if (types.length + offset <= i) {
462 				break;
463 			}
464 			Type t = DatatypeConverter.adjustIfVaries(types, i + offset);
465 			if (i == 0) {
466 				addr.addLine(ParserUtils.toString(t));
467 			} else if (t instanceof Primitive part) {
468 				switch (i) {
469 					case 1 :
470 						addr.addLine(part.getValue());
471 						break;
472 					case 2 :
473 						addr.setCity(part.getValue());
474 						break;
475 					case 3 :
476 						addr.setState(part.getValue());
477 						break;
478 					case 4 :
479 						addr.setPostalCode(part.getValue());
480 						break;
481 					case 5 :
482 						addr.setCountry(part.getValue());
483 						break;
484 					case 6 :
485 						addr.setUse(getUse(part.getValue()));
486 						break;
487 					case 8 :
488 						addr.setDistrict(part.getValue());
489 						break;
490 					default:
491 						break;
492 					}
493 			}
494 		}
495 		if (addr.isEmpty()) {
496 			return null;
497 		}
498 		return addr;
499 	}
500 	private static AddressUse getUse(String value) {
501 		if (StringUtils.isBlank(value)) {
502 			return null;
503 		}
504 		switch (value.trim().toUpperCase()) {
505 			case "B", // Firm/Business
506 				 "O" : // Office/Business
507 				return AddressUse.WORK;
508 			case "BI" : // Billing Address
509 				return AddressUse.BILLING;
510 			case "C" : // Current Or Temporary
511 				return AddressUse.TEMP;
512 			case "H" : // Home
513 				return AddressUse.HOME;
514 			case "BA", // Bad address
515 				 "BDL", // Birth delivery location (address where birth occurred)
516 				 "BR", // Residence at birth (home address at time of birth)
517 				 "F", // Country Of Origin
518 				 "L", // Legal Address
519 				 "M", // Mailing
520 				 "N", // Birth (nee) (birth address, not otherwise specified)
521 				 "P", // Permanent
522 				 "RH", // Registry home
523 				 "S", // Service Location
524 				 "SH", // Shipping Address
525 				 "TM", // Tube Address
526 				 "V" : // Vacation
527 			default :
528 				return null;
529 		}
