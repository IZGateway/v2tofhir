110 	void testCompositeConversionsForCodings(Type t) throws HL7Exception {
111 		CodeableConcept cc = DatatypeConverter.toCodeableConcept(t);
112 		Coding coding = DatatypeConverter.toCoding(t);
113 		if (cc != null && coding != null) {
114 			assertEquals(TestUtils.toString(cc.getCodingFirstRep()), TestUtils.toString(coding));
115 		}
116 		
117 		String first = ParserUtils.toString(t);
118 		if ("HD".equals(t.getName())) {
119 			// HD has System but not code
120 			assertFalse(coding.hasCode());
121 			assertTrue(coding.hasSystem());
122 		} else {
123 			String code = coding == null ? null : coding.getCode();
124 			assertEquals(StringUtils.isBlank(first), StringUtils.isBlank(code));
125 			if (!StringUtils.isBlank(first)) {
126 				assertEquals(first, code);
127 			}
128 		}
129 		
130 		if (coding != null) {
131 			String display = coding.hasUserData("originalDisplay") ? (String) coding.getUserData("originalDisplay") : coding.getDisplay();
132 			assertEquals(StringUtils.isBlank(getComponent(t, 2)), StringUtils.isBlank(display));
133 		
134 			if (hasDisplay(t) && hasComponent(t, 2)) {
135 				// Either both are blank or both are filled.
136 				if (!StringUtils.isBlank(first)) {
137 					// If both are filled, check values.
138 					String[] a = { getComponent(t, 2), TerminologyMapperFactory.get().getDisplay(coding.getSystem(), coding.getCode()).orElse(null) };
139 					List<String> l = Arrays.asList(a);
140 					Supplier<Boolean> test = null;
141 					
142 					if (StringUtils.isNotBlank(coding.getDisplay())) {
143 						// Display came from component 2, or it was properly mapped.
144 						test = () -> l.contains(coding.getDisplay());
145 					} else {
146 						// Display is empty and there are no good values to use.
147 						test = () -> StringUtils.isAllEmpty(a);
148 					}
149 					assertEquals(Boolean.TRUE, test.get(),
150 						"Display value " + coding.getDisplay() + " expected to be from " + l);
151 				}
152 			}
153 		}
154 		String encoded = ParserUtils.unescapeV2Chars(t.encode());
155 		if (coding != null && coding.hasSystem()) {
156 			if (coding.getSystem().contains(":")) {
157 				// There is a proper URI, verify we got it correctly.
158 				// TerminologyMapper has no system alias lookup yet (IGDD-3747), so this test still
159 				// uses the deprecated utils.Systems stub.
160 				@SuppressWarnings("deprecation")
161 				Collection<String> names = Systems.getSystemNames(coding.getSystem());
162 				if (coding.hasUserData("originalSystem")) {
163 					String originalSystem = (String)coding.getUserData("originalSystem");
164 					originalSystem = StringUtils.trim(originalSystem); 
165 					if (!names.contains(originalSystem)) {
166 						names.add(originalSystem);
167 					}
168 				}
169 				names = names.stream().map(StringUtils::upperCase).toList();
170 				// We know about this URI
171 				assertNotNull(names);
172 				List<String> fields = Arrays.asList(encoded.split("\\^")).stream()
173 						.map(StringUtils::trim).map(StringUtils::upperCase).toList();
174 				List<String> f = names.stream().filter(n -> fields.contains(n)).toList();
175 				// At least one of the names for the system is in the message fields.
176 				assertTrue(f.size() > 0, "Could not find any of " + names + " in " + fields);
177 				
178 			} else {
179 				assertTrue(
180 					Strings.CS.contains(encoded, coding.getSystem()), 
181 					t.encode() + " does not contain " + coding.getSystem()
182 				);
183 			}
184 		}
185 		log.debug("{}", encoded);
186 		log.debug("{}", TestUtils.toString(coding));
187 	}
188 	private boolean hasDisplay(Type t) {
189 		return Arrays.asList("CE", "CNE", "CWE").contains(t.getName());
190 	}
191 	private boolean hasComponent(Type t, int index) {
192 		if (t instanceof Composite comp) {
193 			Type types[] = comp.getComponents();
194 			try {
195 				return types.length >= index && !types[index].isEmpty();
196 			} catch (HL7Exception e) {
197 				return false;
198 			}
199 		}
200 		return false;
201 	}
202 	private String getComponent(Type t, int index) throws HL7Exception {
203 		if (t instanceof Varies v) {
204 			t = v.getData();
205 		}
206 		if (t instanceof Composite comp) {
207 			Type types[] = comp.getComponents();
208 			return ParserUtils.unescapeV2Chars(types[index-1].encode());
209 		}
210 		return null;
211 	}

300 	private String testMicrosoftConverterResponse(InputStream is) throws IOException, ParseException {
301 		String response = IOUtils.toString(is, StandardCharsets.UTF_8);
302 		StreamTokenizer t = new StreamTokenizer(new StringReader(response));
303 		t.lowerCaseMode(false);
304 		t.quoteChar('"');
305 		t.whitespaceChars(0, ' ');
306 		t.wordChars('!', 0xFF);
307 		t.eolIsSignificant(false);
308 		
309 		StringBuilder b = new StringBuilder();
310 		String fhirResult = null;
311 		while (t.nextToken() != StreamTokenizer.TT_EOF) {
312 			if (t.ttype == '"') { 
313 				switch (t.sval) {
314 				case "FhirResource":
315 					skipColon(t);
316 					fhirResult = getJsonContent(t, b);
317 					break;
318 				case "Status":
319 					skipColon(t);
320 					assertEquals('"', t.nextToken());
321 					assertEquals("OK", t.sval);
322 					break;
323 				}
324 			}
325 		}
326 		return fhirResult;
327 	}

329 	private void skipColon(StreamTokenizer t) throws IOException, ParseException {
330 		if (t.nextToken() == StreamTokenizer.TT_WORD) {
331 			if (":".equals(t.sval))
332 				return;
333 		}
334 		throw new ParseException("A colon ':' was expected.", t.lineno());
335 	}
336 	private String getJsonContent(StreamTokenizer t, StringBuilder b) throws IOException, ParseException {
337 		int braceCount = 0;
338 		while (t.nextToken() != StreamTokenizer.TT_EOF) {
339 			switch (t.ttype) {
340 			case StreamTokenizer.TT_WORD:
341 				b.append(t.sval);
342 				if ("{".equals(t.sval)) {
343 					braceCount++;
344 				} else if ("}".equals(t.sval)) {
345 					if (--braceCount == 0) {
346 						return b.toString();
347 					}
348 				}
349 				break;
350 			case '"':
351 				b.append('"');
352 				b.append(StringEscapeUtils.escapeJava(t.sval));
353 				b.append('"');
354 				break;
355 			}
356 			// Insert whitespace between each token
357 			b.append(' ');
358 		}
359 		throw new ParseException("Unexpected EOF", t.lineno());
360 	}

362 	private HttpURLConnection getUrlConnection() throws IOException {
363 		URL url;
364 		try {
365 			url = new URI(base).toURL();
366 		} catch (URISyntaxException e) {
367 			throw new IOException(e.getMessage(), e);
368 		}
369 		HttpURLConnection con = (HttpURLConnection) url.openConnection();
370 		con.setRequestMethod("POST");
371 		con.setDoInput(true);
372 		con.setDoOutput(true);
373 		return con;
374 	}
