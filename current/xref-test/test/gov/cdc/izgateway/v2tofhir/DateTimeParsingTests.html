123 	static int listComparator(List<String> a, List<String> b) {
124 		if (a == b) {
125 			return 0;
126 		}
127 		if ((a == null || a.isEmpty()) && (b == null || b.isEmpty())) {
128 			return 0;
129 		}
130 		if (a == null || a.isEmpty()) {
131 			return -1;
132 		}
133 		if (b == null) {
134 			return 1;
135 		}
136 		int comp = a.getClass().getName().compareTo(b.getClass().getName());
137 		if (comp != 0) {
138 			return comp;
139 		}
140 		comp = Integer.compare(a.size(), b.size());
141 		if (comp != 0) {
142 			return comp;
143 		}
144 		// Test some arbitrary locations (first, last and middle)
145 		int[] tests = { 0, a.size() - 1, a.size() / 2 };
146 		for (int t : tests) {
147 			comp = a.get(t).compareTo(b.get(t));
148 			if (comp != 0) {
149 				return comp;
150 			}
151 		}
152 		// Two different array lists aren't equal even if they contain the same element
153 		// data,
154 		// choose an arbitrary value. We aren't sorting these keys.
155 		return 1;
156 	}

158 	private static NamedArrayList choose(NamedArrayList list) {
159 		if (list.size() == 0) {
160 			return NamedArrayList.find("empty");
161 		}
162 		Integer choice = choices.computeIfAbsent(list, k -> Integer.valueOf(0));
163 		if (choice >= list.size()) {
164 			choice = 0;
165 		}
166 		NamedArrayList result = NamedArrayList.l(list.name() + "[" + choice + "]", list.get(choice));
167 		choices.put(list, choice + 1);
168 		return result;
169 	}

207 	private static void addTimeZones(NamedArrayList tests, boolean isValid, boolean withPunct) {
208 		NamedArrayList l = NamedArrayList.find(tests.name() + "withTZ()");
209 		NamedArrayList tzSource = isValid ? validTimeZones : invalidTimeZones;
210 		if (l.isEmpty()) {
211 			l.addAll(tests); // Take a snapshot of tests
212 			
213 			// Modify list by adding time zones.
214 			int pos = 0;
215 			for (int i = 0; i < l.size(); i++) {
216 				String vt = l.get(i);
217 				String tz = tzSource.get(pos);
218 				if (!withPunct) {
219 					tz.replace(":", ""); // Remove colons in TZ
220 				}
221 				pos = (pos + 1) %  tzSource.size();
222 				l.set(i, vt + tz);
223 			}
224 		}
225 		tests.addAll(l);
226 	}

350 	static String deleteOne(String s, int position) {
351 		if (s.length() <= 1) {
352 			return "";
353 		}
354 		if (position == 0) {
355 			return s.length() == 1 ? "" : s.substring(position);
356 		}
357 		String newString = s.substring(0, position);
358 		if (position + 1 < s.length()) {
359 			newString += s.substring(position + 1);
360 		}
361 		return newString;
362 	}

364 	static String doubleUp(String s, int position) {
365 		if (s.length() <= 1) {
366 			return s + s;
367 		}
368 		if (position == 0) {
369 			return s.charAt(position) + s;
370 		}
371 
372 		return s.substring(0, position) + s.charAt(position) + s.substring(position);
373 	}

375 	static String transpose(String s, int position) {
376 		if (s.length() < 2) {
377 			return "" + s.charAt(s.length() - 1) + s.charAt(0);
378 		}
379 		position %= s.length();
380 		StringBuffer b = new StringBuffer(s);
381 		char l = b.charAt(position);
382 		char r = b.charAt((position + 1) % s.length());
383 		b.setCharAt(position, r);
384 		b.setCharAt((position + 1) % s.length(), l);
385 		return b.toString();
386 	}

394 	 * Get timestamps combine valid times and dates invalid times with valid dates

396 	 * valid times with fuzzed dates
397 	 * 
398 	 * @param isValid
399 	 * @return
400 	 */
401 	static NamedArrayList getTimestamps(Combo combo, boolean withPunct) {
402 		String key = "getTimestamps(" + combo + "," + withPunct + ")";
403 		NamedArrayList result = NamedArrayList.find(key);
404 		if (!result.isEmpty()) {
405 			return result;
406 		}
407 		char punct = withPunct ? 'T' : (char) 0;
408 		NamedArrayList dates;
409 		switch (combo) {
410 		case VALID_DATE_AND_TIME, VALID_DATE_AND_INVALID_TIME, VALID_DATE_AND_FUZZED_TIME:
411 			dates = choose(getValidDates(withPunct));
412 			break;
413 		case INVALID_DATE_AND_VALID_TIME:
414 			dates = getInvalidDates(withPunct);
415 			break;
416 		case FUZZED_DATE_AND_VALID_TIME:
417 			dates = fuzzStrings(getValidDates(withPunct), DateTimeParsingTests::deleteOne, DateTimeParsingTests::doubleUp,
418 					DateTimeParsingTests::transpose);
419 			break;
420 		default:
421 			dates = null;
422 			break;
423 		}
424 		TestUtils.assertNotEmpty(dates);
425 		NamedArrayList times;
426 		switch (combo) {
427 		case VALID_DATE_AND_TIME:
428 		case INVALID_DATE_AND_VALID_TIME:
429 		case FUZZED_DATE_AND_VALID_TIME:
430 			times = choose(getTimes(withPunct, true, -1, 0, 1, 2, 3, 4));
431 			break;
432 		case VALID_DATE_AND_INVALID_TIME:
433 			times = getTimes(withPunct, false, -1, 0, 1, 2, 3, 4);
434 			break;
435 		case VALID_DATE_AND_FUZZED_TIME:
436 			times = fuzzStrings(getTimes(withPunct, true, -1, 0, 1, 2, 3, 4), DateTimeParsingTests::deleteOne,
437 					DateTimeParsingTests::doubleUp, DateTimeParsingTests::transpose);
438 			break;
439 		default:
440 			times = null;
441 			break;
442 		}
443 		TestUtils.assertNotEmpty(times);
444 
445 		NamedArrayList crossedResult = cross(dates, times, punct);
446 		log.debug("Computed {} test cases", crossedResult.size());
447 		return crossedResult.notEmpty();
448 	}

490 	private <T> void compareToFhir(String input, PrimitiveType<?> actual, Class<T> primitive) {
491 		String expected = null;
492 		try {
493 			BaseDateTimeType fhirType = new DateTimeType();
494 			fhirType.setValueAsString(input);
495 			Date date = fhirType.getValue();
496 			TemporalPrecisionEnum prec = fhirType.getPrecision();
497 			TimeZone tz = fhirType.getTimeZone();
498 			fhirType = new InstantType();
499 			fhirType.setValue(date);
500 			fhirType.setTimeZone(tz);
501 			fhirType.setPrecision(prec);
502 			expected = fhirType.getValueAsString();
503 		} catch (Exception ex) {
504 			// Ignore FHIR exception.
505 		}
506 		if (expected != null) {
507 			String actualValue = actual == null ? null : actual.asStringValue();
508 			assertNotNull(actual);
509 			assertTrue(timesAreEquivalent(input, actualValue), input + " is not equivalent to " + actualValue);
510 			assertTrue(timesAreEquivalent(expected, actualValue), input + " is not equivalent to original FHIR string " + actualValue);
511 		}
512 	}

514 	private boolean timesAreEquivalent(String expectedValue, String actualValue) {
515 		// Let -0: == 00: be OK 
516 		String expected = expectedValue.replace("-0:", "00:").replace(":-0", ":00").replace("+00:00", "Z").replace("-00:00", "Z");
517 		String actual = actualValue.replace("+00:00", "Z").replace("-00:00", "Z");
518 		
519 		if (hasTimeZone(actual) && !hasTimeZone(expected)) {
520 			actual = removeTimeZone(actual);
521 		}
522 		
523 		if (expected.equals(actual)) {  // strings are equal
524 			return true;
525 		}
526 		
527 		String expHours = StringUtils.substringBefore(expected, ":");
528 		String actHours = StringUtils.substringBefore(actual, ":");
529 		if (expHours.equals(actHours)) {
530 			String expMins = StringUtils.substringAfter(expected, ":");
531 			String actMins = StringUtils.substringAfter(actual, ":");
532 
533 			// Second rollover works
534 			if (expMins.contains(":60") && actMins.contains(":00")) {
535 				expMins = StringUtils.substringBefore(expMins, ":60");
536 				actMins = StringUtils.substringBefore(actMins, ":00");
537 				if (StringUtils.isNumeric(expMins) && StringUtils.isNumeric(actMins) &&
538 					(Integer.parseInt(actMins) - Integer.parseInt(expMins) == 1)
539 				) {
540 					return true;
541 				}
542 			}
543 		}
544 		
545 		if (expected.contains(".") && actual.contains(".")) {  // Check for precision enhancement
546 			if (expected.contains(".999") && actual.contains(".000") &&
547 				StringUtils.substringBefore(actual, "T").equals(StringUtils.substringBefore(expected, "T"))
548 			) {
549 				// Truncated to 3-4 digits of precision and rolled over to next second
550 				return true;
551 			}
552 			log.debug("Unadjusted: actual {} {}", expected, actual); 
553 			
554 			expected = adjustPrecision(expected);
555 			actual = adjustPrecision(actual);
556 			if (expected.equals(actual)) {
557 				return true;
558 			}
559 			log.debug("Precision Adjusted: actual {} {}", expected, actual); 
560 		} else if (expected.contains(".") || actual.contains(".")) {
561 			// Allow .000 to be removed
562 			String exp = expected.replace(".000", "").replace("+00:00", "Z").replace("-00:00", "Z");
563 			String act = actual.replace(".000", "").replace("+00:00", "Z").replace("-00:00", "Z");
564 			if (exp.equals(act)) {
565 				return true;
566 			}
567 		}
568 		
569 		if (!hasEquivalentTimeZones(expected, actual)) { // Time zones equivalent
570 			return false;
571 		}
572 		expected = removeTimeZone(expected);
573 		actual = removeTimeZone(actual);
574 		
575 		// Handle 08:00 vs 08:00:00
576 		if (expected.endsWith(":00")) {
577 			expected = StringUtils.substringBeforeLast(expected, ":00");
578 			return expected.equals(actual);
579 		}
580 		return false;
581 	}

583 	private String removeTz(String s) {
584 		// Remove any TZ suffix
585 		s = StringUtils.substringBeforeLast(s, "-");
586 		s = StringUtils.substringBeforeLast(s, "+");
587 		s = StringUtils.substringBeforeLast(s, "Z");
588 		return s;
589 	}
590 	
591 	private String adjustPrecision(String s) {
592 		if (!s.contains(".")) {
593 			return s;
594 		}
595 		String l = StringUtils.substringBefore(s, ".");
596 		String r = StringUtils.substringAfter(s, ".");
597 		String rwotz = removeTz(r);
598 		String tz = r.substring(rwotz.length());
599 		return l + "." + StringUtils.right(rwotz + "000", 3) + tz;
600 	}
601 	private boolean hasEquivalentTimeZones(String expectedValue, String actualValue) {
602 		if (expectedValue.contains("+") && actualValue.contains("+")) {
603 			return StringUtils.substringAfter(expectedValue, "+").equals(
604 					StringUtils.substringAfter(actualValue, "+"));
605 		}
606 		if (expectedValue.contains("-") && actualValue.contains("-")) {
607 			return StringUtils.substringAfter(expectedValue, "+").equals(
608 					StringUtils.substringAfter(actualValue, "+"));
609 		}
610 		return expectedValue.endsWith("Z") && actualValue.endsWith("Z");
611 	}
612 
613 	private boolean hasTimeZone(String value) {
614 		value = StringUtils.substringAfter(value, "T");
615 		return StringUtils.isEmpty(value) || StringUtils.containsAny(value, "+-Z");
616 	}
617 	
618 	private String removeTimeZone(String value) {
619 		String[] parts = value.split("T");
620 		if (parts.length == 1) {
621 			return value;
622 		}
623 		return parts[0] + "T" + StringUtils.substringBefore(parts[1].replace("-", "Z").replace("+", "Z"), "Z");
624 	}

626 	private void compareToString(String s, PrimitiveType<?> actual, boolean isValid) {
627 		if (actual != null) {
628 			// We got a value, but FHIR couldn't.
629 			if (s.length() + 6 == actual.asStringValue().length()) {
630 				assertEquals(s + isoTZ, actual.asStringValue(), actual.asStringValue()
631 						+ " didn't match the original string " + s + " with default timezone " + isoTZ);
632 			} else {
633 				assertEquals(s, actual.asStringValue(),
634 						actual.asStringValue() + " didn't match the original string " + s);
635 			}
636 		} else if (isValid) {
637 			assertNotNull(actual, String.format("%s is supposed to be a valid Timestamp", s));
638 		}
639 	}
