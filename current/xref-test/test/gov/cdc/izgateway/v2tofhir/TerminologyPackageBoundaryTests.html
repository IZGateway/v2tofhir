11  /**
12   * Enforces that the five static utility classes moved to
13   * {@code gov.cdc.izgw.v2tofhir.terminology} are not accessed directly from outside
14   * that package, with the sole exception of the deprecated forwarding stubs in
15   * {@code gov.cdc.izgw.v2tofhir.utils}.
16   *
17   * <p>New code should use {@link gov.cdc.izgw.v2tofhir.terminology.TerminologyMapper}
18   * (obtained via {@link gov.cdc.izgw.v2tofhir.terminology.TerminologyMapperFactory#get()})
19   * for all terminology operations, and
20   * {@link gov.cdc.izgw.v2tofhir.utils.Systems} for namespace/system URI constants.</p>
21   */
22  class TerminologyPackageBoundaryTests {
23  
24      private static final String TERMINOLOGY_PKG = "gov.cdc.izgw.v2tofhir.terminology";
25      private static final String UTILS_PKG        = "gov.cdc.izgw.v2tofhir.utils";
26  
27      private static JavaClasses classes;
28  
29      @BeforeAll
30      static void importClasses() {
31          classes = new ClassFileImporter()
32                  .importPackages("gov.cdc.izgw.v2tofhir", "test.gov.cdc.izgateway.v2tofhir");
33      }
34  
35      @Test
36      void noClassOutsideTerminologyPackageShouldAccessTerminologySystems() {
37          ArchRule rule = noClasses()
38                  .that().resideOutsideOfPackage(TERMINOLOGY_PKG)
39                  .and().resideOutsideOfPackage(UTILS_PKG)
40                  .should().accessClassesThat()
41                  .haveFullyQualifiedName(TERMINOLOGY_PKG + ".Systems");
42          rule.check(classes);
43      }
